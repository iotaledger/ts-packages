20pub(crate) trait BlockVerifier: Send + Sync + 'static {
21    /// Verifies a block's metadata and transactions.
22    /// This is called before examining a block's causal history.
23    fn verify(&self, block: &SignedBlock) -> ConsensusResult<()>;
24
25    /// Verifies a block w.r.t. ancestor blocks.
26    /// This is called after a block has complete causal history locally,
27    /// and is ready to be accepted into the DAG.
28    ///
29    /// Caller must make sure ancestors corresponse to block.ancestors() 1-to-1,
30    /// in the same order.
31    fn check_ancestors(
32        &self,
33        block: &VerifiedBlock,
34        ancestors: &[Option<VerifiedBlock>],
35        gc_enabled: bool,
36        gc_round: Round,
37    ) -> ConsensusResult<()>;
38}
39
40/// `SignedBlockVerifier` checks the validity of a block.
41///
42/// Blocks that fail verification at one honest authority will be rejected by
43/// all other honest authorities as well. The means invalid blocks, and blocks
44/// with an invalid ancestor, will never be accepted into the DAG.
45pub(crate) struct SignedBlockVerifier {
46    context: Arc<Context>,
47    genesis: BTreeSet<BlockRef>,
48    transaction_verifier: Arc<dyn TransactionVerifier>,
49}
50
51impl SignedBlockVerifier {
52    pub(crate) fn new(
53        context: Arc<Context>,
54        transaction_verifier: Arc<dyn TransactionVerifier>,
55    ) -> Self {
56        let genesis = genesis_blocks(&context)
57            .into_iter()
58            .map(|b| b.reference())
59            .collect();
60        Self {
61            context,
62            genesis,
63            transaction_verifier,
64        }
65    }
66
67    pub(crate) fn check_transactions(&self, batch: &[&[u8]]) -> ConsensusResult<()> {
68        let max_transaction_size_limit =
69            self.context.protocol_config.max_transaction_size_bytes() as usize;
70        for t in batch {
71            if t.len() > max_transaction_size_limit && max_transaction_size_limit > 0 {
72                return Err(ConsensusError::TransactionTooLarge {
73                    size: t.len(),
74                    limit: max_transaction_size_limit,
75                });
76            }
77        }
78
79        let max_num_transactions_limit =
80            self.context.protocol_config.max_num_transactions_in_block() as usize;
81        if batch.len() > max_num_transactions_limit && max_num_transactions_limit > 0 {
82            return Err(ConsensusError::TooManyTransactions {
83                count: batch.len(),
84                limit: max_num_transactions_limit,
85            });
86        }
87
88        let total_transactions_size_limit = self
89            .context
90            .protocol_config
91            .max_transactions_in_block_bytes() as usize;
92        if batch.iter().map(|t| t.len()).sum::<usize>() > total_transactions_size_limit
93            && total_transactions_size_limit > 0
94        {
95            return Err(ConsensusError::TooManyTransactionBytes {
96                size: batch.len(),
97                limit: total_transactions_size_limit,
98            });
99        }
100        Ok(())
101    }
102}

104// All block verification logic are implemented below.
105impl BlockVerifier for SignedBlockVerifier {
106    #[instrument(level = "trace", skip_all)]
107    fn verify(&self, block: &SignedBlock) -> ConsensusResult<()> {
108        let committee = &self.context.committee;
109        // The block must belong to the current epoch and have valid authority index,
110        // before having its signature verified.
111        if block.epoch() != committee.epoch() {
112            return Err(ConsensusError::WrongEpoch {
113                expected: committee.epoch(),
114                actual: block.epoch(),
115            });
116        }
117        if block.round() == 0 {
118            return Err(ConsensusError::UnexpectedGenesisBlock);
119        }
120        if !committee.is_valid_index(block.author()) {
121            return Err(ConsensusError::InvalidAuthorityIndex {
122                index: block.author(),
123                max: committee.size() - 1,
124            });
125        }
126
127        // Verify the block's signature.
128        block.verify_signature(&self.context)?;
129
130        // Verify the block's ancestor refs are consistent with the block's round,
131        // and total parent stakes reach quorum.
132        if block.ancestors().len() > committee.size() {
133            return Err(ConsensusError::TooManyAncestors(
134                block.ancestors().len(),
135                committee.size(),
136            ));
137        }
138        if block.ancestors().is_empty() {
139            return Err(ConsensusError::InsufficientParentStakes {
140                parent_stakes: 0,
141                quorum: committee.quorum_threshold(),
142            });
143        }
144        let mut seen_ancestors = vec![false; committee.size()];
145        let mut parent_stakes = 0;
146        for (i, ancestor) in block.ancestors().iter().enumerate() {
147            if !committee.is_valid_index(ancestor.author) {
148                return Err(ConsensusError::InvalidAncestorAuthorityIndex {
149                    block_authority: block.author(),
150                    ancestor_index: ancestor.author,
151                    max: committee.size() - 1,
152                });
153            }
154            if (i == 0 && ancestor.author != block.author())
155                || (i > 0 && ancestor.author == block.author())
156            {
157                return Err(ConsensusError::InvalidAncestorPosition {
158                    block_authority: block.author(),
159                    ancestor_authority: ancestor.author,
160                    position: i,
161                });
162            }
163            if ancestor.round >= block.round() {
164                return Err(ConsensusError::InvalidAncestorRound {
165                    ancestor: ancestor.round,
166                    block: block.round(),
167                });
168            }
169            if ancestor.round == GENESIS_ROUND && !self.genesis.contains(ancestor) {
170                return Err(ConsensusError::InvalidGenesisAncestor(*ancestor));
171            }
172            if seen_ancestors[ancestor.author] {
173                return Err(ConsensusError::DuplicatedAncestorsAuthority(
174                    ancestor.author,
175                ));
176            }
177            seen_ancestors[ancestor.author] = true;
178            // Block must have round >= 1 so checked_sub(1) should be safe.
179            if ancestor.round == block.round().checked_sub(1).unwrap() {
180                parent_stakes += committee.stake(ancestor.author);
181            }
182        }
183        if !committee.reached_quorum(parent_stakes) {
184            return Err(ConsensusError::InsufficientParentStakes {
185                parent_stakes,
186                quorum: committee.quorum_threshold(),
187            });
188        }
189
190        let batch: Vec<_> = block.transactions().iter().map(|t| t.data()).collect();
191
192        self.check_transactions(&batch)?;
193
194        self.transaction_verifier
195            .verify_batch(&batch)
196            .map_err(|e| ConsensusError::InvalidTransaction(format!("{e:?}")))
197    }
198
199    fn check_ancestors(
200        &self,
201        block: &VerifiedBlock,
202        ancestors: &[Option<VerifiedBlock>],
203        gc_enabled: bool,
204        gc_round: Round,
205    ) -> ConsensusResult<()> {
206        if gc_enabled {
207            // TODO: will be removed with new timestamp calculation is in place as all these
208            // will be irrelevant. When gc is enabled we don't have guarantees
209            // that all ancestors will be available. We'll take into account only the passed
210            // gc_round ones for the timestamp check.
211            let mut max_timestamp_ms = BlockTimestampMs::MIN;
212            for ancestor in ancestors.iter().flatten() {
213                if ancestor.round() <= gc_round {
214                    continue;
215                }
216                max_timestamp_ms = max_timestamp_ms.max(ancestor.timestamp_ms());
217                if max_timestamp_ms > block.timestamp_ms() {
218                    return Err(ConsensusError::InvalidBlockTimestamp {
219                        max_timestamp_ms,
220                        block_timestamp_ms: block.timestamp_ms(),
221                    });
222                }
223            }
224        } else {
225            assert_eq!(block.ancestors().len(), ancestors.len());
226            // This checks the invariant that block timestamp >= max ancestor timestamp.
227            let mut max_timestamp_ms = BlockTimestampMs::MIN;
228            for (ancestor_ref, ancestor_block) in block.ancestors().iter().zip(ancestors.iter()) {
229                let ancestor_block = ancestor_block
230                    .as_ref()
231                    .expect("There should never be an empty slot");
232                assert_eq!(ancestor_ref, &ancestor_block.reference());
233                max_timestamp_ms = max_timestamp_ms.max(ancestor_block.timestamp_ms());
234            }
235            if max_timestamp_ms > block.timestamp_ms() {
236                return Err(ConsensusError::InvalidBlockTimestamp {
237                    max_timestamp_ms,
238                    block_timestamp_ms: block.timestamp_ms(),
239                });
240            }
241        }
242        Ok(())
243    }
244}

595                .build();
596            let verified_block = VerifiedBlock::new_for_test(block);
597            assert!(
598                verifier

606            let block = TestBlock::new(11, 0)
607                .set_ancestors(ancestor_refs)

609                .build();
610            let verified_block = VerifiedBlock::new_for_test(block);

617            ));
618        }
