31#[derive(Serialize, Deserialize, Clone, Debug)]
32#[serde(rename_all = "kebab-case")]
33pub struct ArchivalConfig {
34    pub remote_url: String,
35    pub remote_store_options: Vec<(String, String)>,
36    pub commit_file_size: usize,
37    pub commit_duration_seconds: u64,
38}
39
40pub struct ArchivalReducer {
41    remote_store: Box<dyn ObjectStore>,
42    commit_duration_ms: u64,
43}
44
45impl ArchivalReducer {
46    pub async fn new(config: ArchivalConfig) -> anyhow::Result<Self> {
47        let remote_store =
48            create_remote_store_client(config.remote_url, config.remote_store_options, 10)?;
49
50        Ok(Self {
51            remote_store,
52            commit_duration_ms: config.commit_duration_seconds * 1000,
53        })
54    }
55
56    async fn upload(
57        &self,
58        epoch: EpochId,
59        start: CheckpointSequenceNumber,
60        end: CheckpointSequenceNumber,
61        summary_buffer: Vec<u8>,
62        buffer: Vec<u8>,
63    ) -> anyhow::Result<()> {
64        let checkpoint_file_path = format!("epoch_{epoch}/{start}.chk");
65        let chk_bytes = self
66            .upload_file(
67                Path::from(checkpoint_file_path.clone()),
68                CHECKPOINT_FILE_MAGIC,
69                &buffer,
70            )
71            .await?;
72        let summary_file_path = format!("epoch_{epoch}/{start}.sum");
73        let sum_bytes = self
74            .upload_file(
75                Path::from(summary_file_path.clone()),
76                SUMMARY_FILE_MAGIC,
77                &summary_buffer,
78            )
79            .await?;
80        let mut manifest = Self::read_manifest(&self.remote_store).await?;
81        let checkpoint_file_metadata = create_file_metadata_from_bytes(
82            chk_bytes,
83            FileType::CheckpointContent,
84            epoch,
85            start..end,
86        )?;
87        let summary_file_metadata = create_file_metadata_from_bytes(
88            sum_bytes,
89            FileType::CheckpointSummary,
90            epoch,
91            start..end,
92        )?;
93        manifest.update(epoch, end, checkpoint_file_metadata, summary_file_metadata);
94
95        let bytes = finalize_manifest(manifest)?;
96        self.remote_store
97            .put(&Path::from("MANIFEST"), bytes.into())
98            .await?;
99        Ok(())
100    }
101
102    async fn upload_file(
103        &self,
104        location: Path,
105        magic: u32,
106        content: &[u8],
107    ) -> anyhow::Result<Bytes> {
108        let mut buffer = vec![0; 4];
109        BigEndian::write_u32(&mut buffer, magic);
110        buffer.push(StorageFormat::Blob.into());
111        buffer.push(FileCompression::Zstd.into());
112        buffer.extend_from_slice(content);
113        let mut compressed_buffer = vec![];
114        let mut cursor = Cursor::new(buffer);
115        compress(&mut cursor, &mut compressed_buffer)?;
116        self.remote_store
117            .put(&location, Bytes::from(compressed_buffer.clone()).into())
118            .await?;
119        Ok(Bytes::from(compressed_buffer))
120    }
121
122    pub async fn get_watermark(&self) -> anyhow::Result<CheckpointSequenceNumber> {
123        let manifest = Self::read_manifest(&self.remote_store).await?;
124        Ok(manifest.next_checkpoint_seq_num())
125    }
126
127    async fn read_manifest(remote_store: &dyn ObjectStore) -> anyhow::Result<Manifest> {
128        Ok(match remote_store.get(&Path::from("MANIFEST")).await {
129            Ok(resp) => read_manifest_from_bytes(resp.bytes().await?.to_vec())?,
130            Err(err) if err.to_string().contains("404") => Manifest::new(0, 0),
131            Err(err) => Err(err)?,
132        })
133    }
134}
135
136#[async_trait]
137impl Reducer<RelayWorker> for ArchivalReducer {
138    async fn commit(&self, batch: &[Arc<CheckpointData>]) -> Result<(), anyhow::Error> {
139        if batch.is_empty() {
140            bail!("commit batch can't be empty");
141        }
142        let mut summary_buffer = vec![];
143        let mut buffer = vec![];
144        let first_checkpoint = &batch[0];
145        let epoch = first_checkpoint.checkpoint_summary.epoch;
146        let start_checkpoint = first_checkpoint.checkpoint_summary.sequence_number;
147        let mut last_checkpoint = start_checkpoint;
148        for checkpoint in batch {
149            let full_checkpoint_contents = FullCheckpointContents::from_contents_and_execution_data(
150                checkpoint.checkpoint_contents.clone(),
151                checkpoint
152                    .transactions
153                    .iter()
154                    .map(|t| ExecutionData::new(t.transaction.clone(), t.effects.clone())),
155            );
156            let contents_blob = Blob::encode(&full_checkpoint_contents, BlobEncoding::Bcs).unwrap();
157            let summary_blob =
158                Blob::encode(&checkpoint.checkpoint_summary, BlobEncoding::Bcs).unwrap();
159            contents_blob.write(&mut buffer).unwrap();
160            summary_blob.write(&mut summary_buffer).unwrap();
161            last_checkpoint += 1;
162        }
163        self.upload(
164            epoch,
165            start_checkpoint,
166            last_checkpoint,
167            summary_buffer,
168            buffer,
169        )
170        .await
171        .unwrap();
172        Ok(())
173    }
174
175    fn should_close_batch(
176        &self,
177        batch: &[Arc<CheckpointData>],
178        next_item: Option<&Arc<CheckpointData>>,
179    ) -> bool {
180        // never close a batch without a trigger condition
181        if batch.is_empty() || next_item.is_none() {
182            return false;
183        }
184        let first_checkpoint = &batch[0].checkpoint_summary;
185        let next_checkpoint = next_item.expect("invariant's checked");
186        next_checkpoint.checkpoint_summary.epoch != first_checkpoint.epoch
187            || next_checkpoint.checkpoint_summary.timestamp_ms
188                > (self.commit_duration_ms + first_checkpoint.timestamp_ms)
189    }
190}
