15pub(crate) const MAX_SCORE: u64 = u16::MAX as u64 + 1; // Note: must be consistent with MAX_SCORE in validator_set.move in iota-framework.
16const SCALE_FACTOR: u64 = 2_u64.pow(16);
17
18/// Holds all information related to scoring of authorities in the committee.
19pub struct Scorer {
20    // The current metrics counts collected by the authority, i.e., the local view of the node
21    // about the behaviour of the rest of the committee, according to the blocks received.
22    pub(crate) current_local_metrics_count: Arc<VersionedScoringMetrics>,
23    // The metrics counts received from other authorities, i.e., the information contained in the
24    // MisbehaviourReports received by the authority. If an authority has not sent a report, its
25    // entry in this vector will be all zeroed.
26    received_metrics: Vec<VersionedScoringMetrics>,
27    // Indicates whether an authority did not send any misbehavior reports in the epoch. We use
28    // this to differentiate an authority that did not send a report from another one who sent
29    // zeroed reports.
30    has_not_sent_report: Vec<AtomicBool>,
31    // The current scores of the authorities, updated after each received report. This score is
32    // calculated based on the information in the received reports and the validity of the reports
33    // themselves.
34    pub(crate) current_scores: Scores,
35    // The count of invalid reports received from each authority. Validity here must be checked in
36    // a deterministic way, since this information will not be propagated again to the rest of the
37    // committee.
38    invalid_reports_count: Vec<AtomicU64>,
39    // The voting power of each authority in the committee.
40    voting_power: Vec<u64>,
41    // The version of the scorer being used with its parameters.
42    version: ScorerVersion,
43}
44
45impl Scorer {
46    pub fn new(voting_power: Vec<u64>, protocol_config: &ProtocolConfig) -> Self {
47        let committee_size = voting_power.len();
48        match protocol_config.scorer_version_as_option() {
49            None | Some(1) => {
50                // Local metrics count are always initialized as zero.
51                let current_local_metrics_count = Arc::new(VersionedScoringMetrics::new(
52                    committee_size,
53                    protocol_config,
54                ));
55                let (received_metrics, has_not_sent_report, current_scores, invalid_reports_count) =
56                    (0..committee_size)
57                        .map(|_| {
58                            (
59                                // Received metrics initialized to zero.
60                                VersionedScoringMetrics::new(committee_size, protocol_config),
61                                // Initially, none of the authorities had sent any valid report.
62                                AtomicBool::new(true),
63                                // Current scores initialized to max score.
64                                AtomicU64::new(MAX_SCORE),
65                                // Invalid reports count initialized to zero.
66                                AtomicU64::new(0),
67                            )
68                        })
69                        .collect();
70                let parameters = ParametersV1 {
71                    allowances: MisbehaviorsV1 {
72                        faulty_blocks_provable: 1,
73                        faulty_blocks_unprovable: 2,
74                        missing_proposals: 48_000, // roughly 3% of consensus rounds in an epoch
75                        equivocations: 0,
76                    },
77                    maximums: MisbehaviorsV1 {
78                        faulty_blocks_provable: 5,
79                        faulty_blocks_unprovable: 10,
80                        missing_proposals: 160_000, // roughly 10% of consensus rounds in an epoch
81                        equivocations: 1,
82                    },
83                    weights: MisbehaviorsV1 {
84                        faulty_blocks_provable: SCALE_FACTOR * 30 / 100,
85                        faulty_blocks_unprovable: SCALE_FACTOR * 10 / 100,
86                        missing_proposals: SCALE_FACTOR * 35 / 100,
87                        equivocations: 1,
88                    },
89                };
90                // Assert that the allowance for major misbehaviors is 0,
91                // maximum is 1 and weight is 1. This is because major misbehaviors should
92                // reduce the score to 0 is there are any occurrences.
93                // Only equivocation is considered a major misbehavior in this version.
94                assert!(
95                    parameters
96                        .allowances
97                        .iter_major_misbehaviors()
98                        .all(|&a| a == 0)
99                        && parameters
100                            .maximums
101                            .iter_major_misbehaviors()
102                            .all(|&m| m == 1)
103                        && parameters
104                            .weights
105                            .iter_major_misbehaviors()
106                            .all(|&w| w == 1)
107                );
108                // Assert that allowances are compatible with the maximums for all metrics.
109                assert!(
110                    parameters
111                        .allowances
112                        .iter()
113                        .zip(parameters.maximums.iter())
114                        .all(|(&a, &m)| a < m)
115                );
116
117                // Assert that maximums are compatible with MAX_SCORE for all metrics, to
118                // prevent overflows.
119                assert!(
120                    parameters
121                        .maximums
122                        .iter()
123                        .all(|&a| a <= u64::MAX / MAX_SCORE)
124                );
125
126                Self {
127                    current_local_metrics_count,
128                    received_metrics,
129                    has_not_sent_report,
130                    current_scores,
131                    invalid_reports_count,
132                    voting_power,
133                    version: ScorerVersion::V1(parameters),
134                }
135            }
136            _ => panic!("Unsupported scorer version"),
137        }
138    }
139
140    fn get_parameters_v1(&self) -> ParametersV1 {
141        match &self.version {
142            ScorerVersion::V1(params) => params.clone(),
143        }
144    }
145
146    // Boundary checks for this functions are done at a higher level. `authority``
147    // should always be derived from a valid AuthorityIndex
148    pub(crate) fn update_invalid_reports_count(&self, authority: u32) {
149        self.invalid_reports_count[authority as usize].fetch_add(1, Ordering::Relaxed);
150    }
151
152    pub(crate) fn update_scores(&self) {
153        match self.version {
154            ScorerVersion::V1(_) => self.update_scores_v1(),
155        };
156    }
157
158    pub(crate) fn update_received_reports(
159        &self,
160        authority: u32,
161        report: &VersionedMisbehaviorReport,
162    ) {
163        // Update the received metrics for the authority, and mark that we have received
164        // metrics from them. Then, update the scores accordingly.
165        self.received_metrics[authority as usize].update_from_report(report);
166        self.has_not_sent_report[authority as usize].store(false, Ordering::Relaxed);
167    }
168}

170// Methods for ScorerVersion::V1
171impl Scorer {
172    fn update_scores_v1(&self) {
173        // Vector with the highest received reports from each authority and their voting
174        // power. Authorities that did not send reports are filtered out.
175        let highest_received_reports_from_authority = self
176            .received_metrics
177            .iter()
178            .zip(self.voting_power.iter())
179            .zip(self.has_not_sent_report.iter())
180            .filter(|((_, _), is_missing)| !is_missing.load(Ordering::Relaxed))
181            .map(|((metrics, voting_power), _)| (metrics.to_report(), *voting_power))
182            .collect::<Vec<(VersionedMisbehaviorReport, VotingPower)>>();
183        // Ensure that we have at least one report to calculate the scores, otherwise we
184        // do nothing.
185        if highest_received_reports_from_authority.is_empty() {
186        } else {
187            let median_report = calculate_median_report(&highest_received_reports_from_authority);
188            let scores = calculate_scores_v1(median_report, self.get_parameters_v1());
189            for (i, &score) in scores.iter().enumerate() {
190                self.current_scores[i].store(score, Ordering::Relaxed);
191            }
192        }
193    }
194}

196/// Given a vector of pairs (VersionedMisbehaviorReport, VotingPower), calculate
197/// the medians for all metrics in VersionedMisbehaviorReport and authorities:
198///
199/// - Assume we have N authorities in the committee, but n<=N reports R_1, R_2,
200///   ..., R_n from authorities with voting powers VP_1, VP_2, ..., VP_n.
201/// - For each metric M in VersionedMisbehaviorReport, we'll have n vectors of
202///   metric values: M_1, M_2, ..., M_n, where M_i is the vector of metric
203///   values for report R_i.
204/// - Each M_i is a vector of length N, where the j-th value corresponds to the
205///   metric value for authority j.
206///
207/// Example: If we have 4 authorities in the committee, and we receive 3
208/// reports:
209/// - Report R_1 from authority A_1 with voting power VP_1 = 1:
210///     - Metric M1: [0, 0, 0, 0] (values for authorities A_1, A_2, A_3, A_4)
211///     - Metric M2: [0, 0, 0, 0] (values for authorities A_1, A_2, A_3, A_4)
212/// - Report R_2 from authority A_2 with voting power VP_2 = 1:
213///     - Metric M1: [1, 1, 1, 1] (values for authorities A_1, A_2, A_3, A_4)
214///     - Metric M2: [2, 2, 2, 2] (values for authorities A_1, A_2, A_3, A_4)
215/// - Report R_3 from authority A_3 with voting power VP_3 = 1:
216///     - Metric M1: [2, 2, 2, 2] (values for authorities A_1, A_2, A_3, A_4)
217///     - Metric M2: [1, 1, 1, 1] (values for authorities A_1, A_2, A_3, A_4)
218///
219/// For Metric M1, we have that the median metric vector is [1, 1, 1, 1].
220///
221/// This method returns a vector of MedianMetricVec, one per metric in
222/// VersionedMisbehaviorReport
223fn calculate_median_report(
224    reports_and_voting_power: &[(VersionedMisbehaviorReport, VotingPower)],
225) -> MisbehaviorsV1<MedianMetricVec> {
226    // Calls to this method should ensure that we have at least one report to
227    // process.
228    assert!(!reports_and_voting_power.is_empty());
229
230    let number_of_metrics = reports_and_voting_power[0].0.iterate_over_metrics().len();
231
232    // In the case of the example in the method documentation,
233    // reports_and_voting_power_per_metric should be
234    // vec![
235    //      vec![([0, 0, 0, 0],VP_1),([1, 1, 1, 1],VP_2),([2, 2, 2, 2],VP_3)],
236    //      vec![([0, 0, 0, 0],VP_1),([2, 2, 2, 2],VP_2),([1, 1, 1, 1],VP_3)]
237    //      ]
238    let mut reports_and_voting_power_per_metric: Vec<Vec<(MetricVec, VotingPower)>> =
239        vec![vec![]; number_of_metrics];
240    for (versioned_report, voting_power) in reports_and_voting_power.iter() {
241        for (i, metric) in versioned_report.iterate_over_metrics().enumerate() {
242            reports_and_voting_power_per_metric[i].push((metric.clone(), *voting_power));
243        }
244    }
245
246    // Calculate and return the weighted median for each metric
247    let median_report = reports_and_voting_power_per_metric
248        .iter_mut()
249        .map(|vec| calculate_weighted_median(vec.as_mut_slice()))
250        .collect::<MisbehaviorsV1<MedianMetricVec>>();
251    median_report
252}

254// Given a vector of pairs (MetricVec, VotingPower), calculate the weighted
255// median of each entry of MetricVec and returns a MedianMetricVec. Each entry
256// of reports corresponds to a single authority i who sent the report. MetricVec
257// always corresponds to a single metric, and each of its entries corresponds to
258// the number of misbehaviors that i claims to have detected from each authority
259// in the committee.
260fn calculate_weighted_median(reports: &mut [(MetricVec, VotingPower)]) -> MedianMetricVec {
261    // Calls to this method should ensure that we have at least one pair (MetricVec,
262    // VotingPower) to process.
263    assert!(!reports.is_empty());
264
265    // We calculate the weighted median relative to the voting power of the
266    // authorities who actually sent a report.
267    let voting_power_used = reports.iter().map(|(_, vp)| *vp).sum::<VotingPower>();
268    // The caller should also guarantee that the MetricVec in all reports have the
269    // same length (committee_size). This is naturally guaranteed when these data
270    // come from MisbehaviorReports, since they would been considered invalid
271    // otherwise.
272    let committee_size = reports[0].0.len();
273    let mut median_per_validator_being_scored = Vec::new();
274
275    for validator_being_scored in 0..committee_size {
276        let mut accumulated_voting_power = 0;
277        reports.sort_by_key(|(reported_counts, _)| reported_counts[validator_being_scored]);
278        for (reported_counts, voting_power) in reports.iter() {
279            accumulated_voting_power += *voting_power;
280            if accumulated_voting_power * 2 >= voting_power_used {
281                median_per_validator_being_scored.push(reported_counts[validator_being_scored]);
282                break;
283            }
284        }
285    }
286
287    median_per_validator_being_scored
288}

290// Scorer version. Currently, only V1 is implemented, relative to both
291// protocol_config.scorer_version = None or Some(1).
292enum ScorerVersion {
293    V1(ParametersV1),
294}
295
296// Parameters for ScorerVersion::V1
297#[derive(Clone)]
298struct ParametersV1 {
299    // Allowed misbehaviors without any punishment
300    allowances: MisbehaviorsV1<u64>,
301    // Number of misbehaviors that lead to zero score
302    maximums: MisbehaviorsV1<u64>,
303    // Weights for each metric. The sum of minor misbehavior weights + baseline_score =
304    // scale_factor. Major misbehavior weights are either 0 or 1.
305    weights: MisbehaviorsV1<u64>,
306}
307
308// Aliases for better readability.
309pub(crate) type Scores = Vec<Score>;
310pub(crate) type Score = AtomicU64;
311type VotingPower = u64;
312type MedianMetricVec = Vec<u64>;
313type MetricVec = Vec<u64>;
314
315// Given the median reports for all metrics, calculate the final scores. A score
316// is an integer between 0 and max_score. For each metrics, we have an allowance
317// (allowed misbehaviors without any punishment) and a maximum (number of
318// misbehaviors that lead to zero score). Based on those values, we calculate a
319// score per metric, and then combine them into a final score. Each individual
320// score for minor misbeahviors (non-equivocation) is also  an integer between 0
321// and max_score, and the weights used for the combination are such that
322// sum(weights) + baseline_score = scale_factor. Thus, we need
323// max_score*scale_factor < 2^64 to avoid overflows.
324// Major misbehaviors (equivocations) are treated differently, as they
325// multiplicatively impact the final score. Their value is either 0 or 1.
326fn calculate_scores_v1(
327    median_reports: MisbehaviorsV1<MedianMetricVec>,
328    parameters: ParametersV1,
329) -> Vec<u64> {
330    let baseline_score = SCALE_FACTOR - parameters.weights.iter_minor_misbehaviors().sum::<u64>();
331
332    let median_minor_reports_and_parameters = median_reports
333        .iter_minor_misbehaviors()
334        .zip(parameters.allowances.iter_minor_misbehaviors())
335        .zip(parameters.maximums.iter_minor_misbehaviors());
336
337    // Calculate individual metric scores
338    let minor_metric_scores = median_minor_reports_and_parameters
339        .map(
340            |((median_report_for_a_single_metric, metric_allowance), metric_maximum)| {
341                median_report_single_metric_to_score(
342                    median_report_for_a_single_metric,
343                    *metric_allowance,
344                    *metric_maximum,
345                    MAX_SCORE,
346                )
347            },
348        )
349        .collect::<Vec<Vec<u64>>>();
350
351    let median_major_reports_and_parameters = median_reports
352        .iter_major_misbehaviors()
353        .zip(parameters.allowances.iter_major_misbehaviors())
354        .zip(parameters.maximums.iter_major_misbehaviors());
355
356    // Calculate individual metric scores
357    let major_metric_scores = median_major_reports_and_parameters
358        .map(
359            |((median_report_for_a_single_metric, metric_allowance), metric_maximum)| {
360                median_report_single_metric_to_score(
361                    median_report_for_a_single_metric,
362                    *metric_allowance,
363                    *metric_maximum,
364                    1,
365                )
366            },
367        )
368        .collect::<Vec<Vec<u64>>>();
369
370    metrics_scores_to_final_scores(
371        minor_metric_scores,
372        major_metric_scores,
373        parameters.weights,
374        baseline_score,
375        SCALE_FACTOR,
376        MAX_SCORE,
377    )
378}

380fn metrics_scores_to_final_scores(
381    minor_metric_scores: Vec<Vec<u64>>,
382    major_metric_scores: Vec<Vec<u64>>,
383    weights: MisbehaviorsV1<u64>,
384    baseline_score: u64,
385    scale_factor: u64,
386    max_score: u64,
387) -> Vec<u64> {
388    // Initialise the final scores with the baseline score whose value is between 0
389    // and max_score * scale_factor.
390    let committee_size = minor_metric_scores.first().unwrap().len();
391    let mut final_scores = vec![baseline_score * max_score; committee_size];
392    // First, calculate the weights sum of minor misbehavior scores vector. The
393    // values in final_scores will still be between 0 and max_score * scale_factor
394    minor_metric_scores
395        .iter()
396        .zip(weights.iter_minor_misbehaviors())
397        .for_each(|(scores, weight)| {
398            for (i, &score) in scores.iter().enumerate() {
399                final_scores[i] += score * weight;
400            }
401        });
402    // Then, multiply by each major misbehavior score which is a value of either 0
403    // or 1.
404    major_metric_scores.iter().for_each(|scores| {
405        for (i, &score) in scores.iter().enumerate() {
406            final_scores[i] *= score;
407        }
408    });
409    // Finally, divide by the scale factor and scale to max_score
410    for score in final_scores.iter_mut() {
411        *score /= scale_factor;
412    }
413    final_scores
414}

416// Calculate the metric scores for a single metric's median report vector. It
417// returns a vector of values between 0 and the max score for that metric.
418fn median_report_single_metric_to_score(
419    median_report_for_metric: &MedianMetricVec,
420    metric_allowance: u64,
421    metric_max: u64,
422    max_metric_score: u64,
423) -> Vec<u64> {
424    median_report_for_metric
425        .iter()
426        .map(|&report| metric_to_score(report, metric_allowance, metric_max, max_metric_score))
427        .collect()
428}
429
430fn metric_to_score(value: u64, allowance: u64, max: u64, max_score: u64) -> u64 {
431    if value <= allowance {
432        max_score
433    } else if value >= max {
434        0
435    } else {
436        // max - allowance > 0 and the multiplication not overflowing are guaranteed by
437        // assertions done during scorer initialization
438        max.saturating_sub(value).saturating_mul(max_score) / max.saturating_sub(allowance)
439    }
440}
