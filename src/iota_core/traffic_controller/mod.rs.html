37pub const METRICS_INTERVAL_SECS: u64 = 2;
38pub const DEFAULT_DRAIN_TIMEOUT_SECS: u64 = 300;

40type Blocklist = Arc<DashMap<IpAddr, SystemTime>>;
41
42#[derive(Clone)]
43struct Blocklists {
44    clients: Blocklist,
45    proxied_clients: Blocklist,
46}
47
48#[derive(Clone)]
49enum Acl {
50    Blocklists(Blocklists),
51    /// If this variant is set, then we do no tallying or running
52    /// of background tasks, and instead simply block all IPs not
53    /// in the allowlist on calls to `check`. The allowlist should
54    /// only be populated once at initialization.
55    Allowlist(Vec<IpAddr>),
56}
57
58#[derive(Clone)]
59pub struct TrafficController {
60    tally_channel: Option<mpsc::Sender<TrafficTally>>,
61    acl: Acl,
62    metrics: Arc<TrafficControllerMetrics>,
63    dry_run_mode: bool,
64}

66impl Debug for TrafficController {
67    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
68        // NOTE: we do not want to print the contents of the blocklists to logs
69        // given that (1) it contains all requests IPs, and (2) it could be quite
70        // large. Instead, we print lengths of the blocklists. Further, we prefer
71        // to get length from the metrics rather than from the blocklists themselves
72        // to avoid unnecessarily acquiring the read lock.
73        f.debug_struct("TrafficController")
74            .field(
75                "connection_ip_blocklist_len",
76                &self.metrics.connection_ip_blocklist_len.get(),
77            )
78            .field(
79                "proxy_ip_blocklist_len",
80                &self.metrics.proxy_ip_blocklist_len.get(),
81            )
82            .finish()
83    }
84}

86impl TrafficController {
87    pub fn init(
88        policy_config: PolicyConfig,
89        metrics: TrafficControllerMetrics,
90        fw_config: Option<RemoteFirewallConfig>,
91    ) -> Self {
92        match policy_config.allow_list {
93            Some(allow_list) => {
94                let allowlist = allow_list
95                    .into_iter()
96                    .map(|ip_str| {
97                        parse_ip(&ip_str).unwrap_or_else(|| {
98                            panic!("Failed to parse allowlist IP address: {ip_str:?}")
99                        })
100                    })
101                    .collect();
102                Self {
103                    tally_channel: None,
104                    acl: Acl::Allowlist(allowlist),
105                    metrics: Arc::new(metrics),
106                    dry_run_mode: policy_config.dry_run,
107                }
108            }
109            None => Self::spawn(policy_config, metrics, fw_config),
110        }
111    }
112
113    fn spawn(
114        policy_config: PolicyConfig,
115        metrics: TrafficControllerMetrics,
116        fw_config: Option<RemoteFirewallConfig>,
117    ) -> Self {
118        let metrics = Arc::new(metrics);
119        Self::set_policy_config_metrics(&policy_config, metrics.clone());
120        let (tx, rx) = mpsc::channel(policy_config.channel_capacity);
121        // Memoized drainfile existence state. This is passed into delegation
122        // functions to prevent them from continuing to populate blocklists
123        // if drain is set, as otherwise it will grow without bounds
124        // without the firewall running to periodically clear it.
125        let mem_drainfile_present = fw_config
126            .as_ref()
127            .map(|config| config.drain_path.exists())
128            .unwrap_or(false);
129        metrics
130            .deadmans_switch_enabled
131            .set(mem_drainfile_present as i64);
132        let blocklists = Blocklists {
133            clients: Arc::new(DashMap::new()),
134            proxied_clients: Arc::new(DashMap::new()),
135        };
136        let tally_loop_blocklists = blocklists.clone();
137        let clear_loop_blocklists = blocklists.clone();
138        let tally_loop_metrics = metrics.clone();
139        let clear_loop_metrics = metrics.clone();
140        let dry_run_mode = policy_config.dry_run;
141        spawn_monitored_task!(run_tally_loop(
142            rx,
143            policy_config,
144            fw_config,
145            tally_loop_blocklists,
146            tally_loop_metrics,
147            mem_drainfile_present,
148        ));
149        spawn_monitored_task!(run_clear_blocklists_loop(
150            clear_loop_blocklists,
151            clear_loop_metrics,
152        ));
153        Self {
154            tally_channel: Some(tx),
155            acl: Acl::Blocklists(blocklists),
156            metrics,
157            dry_run_mode,
158        }
159    }

191    pub fn tally(&self, tally: TrafficTally) {
192        if let Some(channel) = self.tally_channel.as_ref() {
193            // Use try_send rather than send mainly to avoid creating backpressure
194            // on the caller if the channel is full, which may slow down the critical
195            // path. Dropping the tally on the floor should be ok, as in this case
196            // we are effectively sampling traffic, which we would need to do anyway
197            // if we are overloaded
198            match channel.try_send(tally) {
199                Err(TrySendError::Full(_)) => {
200                    warn!("TrafficController tally channel full, dropping tally");
201                    self.metrics.tally_channel_overflow.inc();
202                    // TODO: once we've verified this doesn't happen under
203                    // normal conditions, we can consider
204                    // dropping the request itself given
205                    // that clearly the system is overloaded
206                }
207                Err(TrySendError::Closed(_)) => {
208                    panic!("TrafficController tally channel closed unexpectedly");
209                }
210                Ok(_) => {}
211            }
212        }
213    }

215    /// Handle check with dry-run mode considered
216    pub async fn check(&self, client: &Option<IpAddr>, proxied_client: &Option<IpAddr>) -> bool {
217        let check_with_dry_run_maybe = |allowed| -> bool {
218            match (allowed, self.dry_run_mode()) {
219                // check succeeded
220                (true, _) => true,
221                // check failed while in dry-run mode
222                (false, true) => {
223                    debug!("Dry run mode: Blocked request from client {:?}", client);
224                    self.metrics.num_dry_run_blocked_requests.inc();
225                    true
226                }
227                // check failed
228                (false, false) => false,
229            }
230        };
231
232        match &self.acl {
233            Acl::Allowlist(allowlist) => {
234                let allowed = client.is_none() || allowlist.contains(&client.unwrap());
235                check_with_dry_run_maybe(allowed)
236            }
237            Acl::Blocklists(blocklists) => {
238                let allowed = self
239                    .check_blocklists(blocklists, client, proxied_client)
240                    .await;
241                check_with_dry_run_maybe(allowed)
242            }
243        }
244    }

246    /// Returns true if the connection is in blocklist, false otherwise
247    async fn check_blocklists(
248        &self,
249        blocklists: &Blocklists,
250        client: &Option<IpAddr>,
251        proxied_client: &Option<IpAddr>,
252    ) -> bool {
253        let client_check = self.check_and_clear_blocklist(
254            client,
255            blocklists.clients.clone(),
256            &self.metrics.connection_ip_blocklist_len,
257        );
258        let proxied_client_check = self.check_and_clear_blocklist(
259            proxied_client,
260            blocklists.proxied_clients.clone(),
261            &self.metrics.proxy_ip_blocklist_len,
262        );
263        let (client_check, proxied_client_check) =
264            futures::future::join(client_check, proxied_client_check).await;
265        client_check && proxied_client_check
266    }
267
268    pub fn dry_run_mode(&self) -> bool {
269        self.dry_run_mode
270    }
271
272    async fn check_and_clear_blocklist(
273        &self,
274        client: &Option<IpAddr>,
275        blocklist: Blocklist,
276        blocklist_len_gauge: &IntGauge,
277    ) -> bool {
278        let client = match client {
279            Some(client) => client,
280            None => return true,
281        };
282        let now = SystemTime::now();
283        // the below two blocks cannot be nested, otherwise we will deadlock
284        // due to acquiring the lock on get, then holding across the remove
285        let (should_block, should_remove) = {
286            match blocklist.get(client) {
287                Some(expiration) if now >= *expiration => (false, true),
288                None => (false, false),
289                _ => (true, false),
290            }
291        };
292        if should_remove {
293            blocklist_len_gauge.dec();
294            blocklist.remove(client);
295        }
296        !should_block
297    }
298}

300/// Although we clear IPs from the blocklist lazily when they are checked,
301/// it's possible that over time we may accumulate a large number of stale
302/// IPs in the blocklist for clients that are added, then once blocked,
303/// never checked again. This function runs periodically to clear out any
304/// such stale IPs. This also ensures that the blocklist length metric
305/// accurately reflects TTL.
306async fn run_clear_blocklists_loop(blocklists: Blocklists, metrics: Arc<TrafficControllerMetrics>) {
307    loop {
308        tokio::time::sleep(Duration::from_secs(3)).await;
309        let now = SystemTime::now();
310        blocklists.clients.retain(|_, expiration| now < *expiration);
311        blocklists
312            .proxied_clients
313            .retain(|_, expiration| now < *expiration);
314        metrics
315            .connection_ip_blocklist_len
316            .set(blocklists.clients.len() as i64);
317        metrics
318            .proxy_ip_blocklist_len
319            .set(blocklists.proxied_clients.len() as i64);
320    }
321}

323async fn run_tally_loop(
324    mut receiver: mpsc::Receiver<TrafficTally>,
325    policy_config: PolicyConfig,
326    fw_config: Option<RemoteFirewallConfig>,
327    blocklists: Blocklists,
328    metrics: Arc<TrafficControllerMetrics>,
329    mut mem_drainfile_present: bool,
330) {
331    let mut spam_policy = TrafficControlPolicy::from_spam_config(policy_config.clone()).await;
332    let mut error_policy = TrafficControlPolicy::from_error_config(policy_config.clone()).await;
333    let spam_blocklists = Arc::new(blocklists.clone());
334    let error_blocklists = Arc::new(blocklists);
335    let node_fw_client = fw_config
336        .as_ref()
337        .map(|fw_config| NodeFWClient::new(fw_config.remote_fw_url.clone()));
338
339    let timeout = fw_config
340        .as_ref()
341        .map(|fw_config| fw_config.drain_timeout_secs)
342        .unwrap_or(DEFAULT_DRAIN_TIMEOUT_SECS);
343    let mut metric_timer = Instant::now();
344
345    loop {
346        tokio::select! {
347            received = receiver.recv() => {
348                metrics.tallies.inc();
349                match received {
350                    Some(tally) => {
351                        // TODO: spawn a task to handle tallying concurrently
352                        if let Err(err) = handle_spam_tally(
353                            &mut spam_policy,
354                            &policy_config,
355                            &node_fw_client,
356                            &fw_config,
357                            tally.clone(),
358                            spam_blocklists.clone(),
359                            metrics.clone(),
360                            mem_drainfile_present,
361                        )
362                        .await {
363                            warn!("Error handling spam tally: {}", err);
364                        }
365                        if let Err(err) = handle_error_tally(
366                            &mut error_policy,
367                            &policy_config,
368                            &node_fw_client,
369                            &fw_config,
370                            tally,
371                            error_blocklists.clone(),
372                            metrics.clone(),
373                            mem_drainfile_present,
374                        )
375                        .await {
376                            warn!("Error handling error tally: {}", err);
377                        }
378                    }
379                    None => {
380                        info!("TrafficController tally channel closed by all senders");
381                        return;
382                    }
383                }
384            }
385            // Dead man's switch - if we suspect something is sinking all traffic to node, disable nodefw
386            _ = tokio::time::sleep(tokio::time::Duration::from_secs(timeout)) => {
387                if let Some(fw_config) = &fw_config {
388                    error!("No traffic tallies received in {} seconds.", timeout);
389                    if mem_drainfile_present {
390                        continue;
391                    }
392                    if !fw_config.drain_path.exists() {
393                        mem_drainfile_present = true;
394                        warn!("Draining Node firewall.");
395                        File::create(&fw_config.drain_path)
396                            .expect("Failed to touch nodefw drain file");
397                        metrics.deadmans_switch_enabled.set(1);
398                    }
399                }
400            }
401        }
402
403        // every N seconds, we update metrics and logging that would be too
404        // spammy to be handled while processing each tally
405        if metric_timer.elapsed() > Duration::from_secs(METRICS_INTERVAL_SECS) {
406            if let TrafficControlPolicy::FreqThreshold(spam_policy) = &spam_policy {
407                if let Some(highest_direct_rate) = spam_policy.highest_direct_rate() {
408                    metrics
409                        .highest_direct_spam_rate
410                        .set(highest_direct_rate.0 as i64);
411                    debug!("Recent highest direct spam rate: {:?}", highest_direct_rate);
412                }
413                if let Some(highest_proxied_rate) = spam_policy.highest_proxied_rate() {
414                    metrics
415                        .highest_proxied_spam_rate
416                        .set(highest_proxied_rate.0 as i64);
417                    debug!(
418                        "Recent highest proxied spam rate: {:?}",
419                        highest_proxied_rate
420                    );
421                }
422            }
423            if let TrafficControlPolicy::FreqThreshold(error_policy) = &error_policy {
424                if let Some(highest_direct_rate) = error_policy.highest_direct_rate() {
425                    metrics
426                        .highest_direct_error_rate
427                        .set(highest_direct_rate.0 as i64);
428                    debug!(
429                        "Recent highest direct error rate: {:?}",
430                        highest_direct_rate
431                    );
432                }
433                if let Some(highest_proxied_rate) = error_policy.highest_proxied_rate() {
434                    metrics
435                        .highest_proxied_error_rate
436                        .set(highest_proxied_rate.0 as i64);
437                    debug!(
438                        "Recent highest proxied error rate: {:?}",
439                        highest_proxied_rate
440                    );
441                }
442            }
443            metric_timer = Instant::now();
444        }
445    }
446}

448async fn handle_error_tally(
449    policy: &mut TrafficControlPolicy,
450    policy_config: &PolicyConfig,
451    nodefw_client: &Option<NodeFWClient>,
452    fw_config: &Option<RemoteFirewallConfig>,
453    tally: TrafficTally,
454    blocklists: Arc<Blocklists>,
455    metrics: Arc<TrafficControllerMetrics>,
456    mem_drainfile_present: bool,
457) -> Result<(), reqwest::Error> {
458    let Some((error_weight, error_type)) = tally.clone().error_info else {
459        return Ok(());
460    };
461    if !error_weight.is_sampled() {
462        return Ok(());
463    }
464    trace!(
465        "Handling error_type {:?} from client {:?}",
466        error_type, tally.direct,
467    );
468    metrics
469        .tally_error_types
470        .with_label_values(&[error_type.as_str()])
471        .inc();
472    let resp = policy.handle_tally(tally);
473    metrics.error_tally_handled.inc();
474    if let Some(fw_config) = fw_config {
475        if fw_config.delegate_error_blocking && !mem_drainfile_present {
476            let client = nodefw_client
477                .as_ref()
478                .expect("Expected NodeFWClient for blocklist delegation");
479            return delegate_policy_response(
480                resp,
481                policy_config,
482                client,
483                fw_config.destination_port,
484                metrics.clone(),
485            )
486            .await;
487        }
488    }
489    handle_policy_response(resp, policy_config, blocklists, metrics).await;
490    Ok(())
491}

493async fn handle_spam_tally(
494    policy: &mut TrafficControlPolicy,
495    policy_config: &PolicyConfig,
496    nodefw_client: &Option<NodeFWClient>,
497    fw_config: &Option<RemoteFirewallConfig>,
498    tally: TrafficTally,
499    blocklists: Arc<Blocklists>,
500    metrics: Arc<TrafficControllerMetrics>,
501    mem_drainfile_present: bool,
502) -> Result<(), reqwest::Error> {
503    if !(tally.spam_weight.is_sampled() && policy_config.spam_sample_rate.is_sampled()) {
504        return Ok(());
505    }
506    let resp = policy.handle_tally(tally.clone());
507    metrics.tally_handled.inc();
508    if let Some(fw_config) = fw_config {
509        if fw_config.delegate_spam_blocking && !mem_drainfile_present {
510            let client = nodefw_client
511                .as_ref()
512                .expect("Expected NodeFWClient for blocklist delegation");
513            return delegate_policy_response(
514                resp,
515                policy_config,
516                client,
517                fw_config.destination_port,
518                metrics.clone(),
519            )
520            .await;
521        }
522    }
523    handle_policy_response(resp, policy_config, blocklists, metrics).await;
524    Ok(())
525}

527async fn handle_policy_response(
528    response: PolicyResponse,
529    policy_config: &PolicyConfig,
530    blocklists: Arc<Blocklists>,
531    metrics: Arc<TrafficControllerMetrics>,
532) {
533    let PolicyResponse {
534        block_client,
535        block_proxied_client,
536    } = response;
537    let PolicyConfig {
538        connection_blocklist_ttl_sec,
539        proxy_blocklist_ttl_sec,
540        ..
541    } = policy_config;
542    if let Some(client) = block_client {
543        if blocklists
544            .clients
545            .insert(
546                client,
547                SystemTime::now() + Duration::from_secs(*connection_blocklist_ttl_sec),
548            )
549            .is_none()
550        {
551            // Only increment the metric if the client was not already blocked
552            debug!("Blocking client: {:?}", client);
553            metrics.requests_blocked_at_protocol.inc();
554            metrics.connection_ip_blocklist_len.inc();
555        }
556    }
557    if let Some(client) = block_proxied_client {
558        if blocklists
559            .proxied_clients
560            .insert(
561                client,
562                SystemTime::now() + Duration::from_secs(*proxy_blocklist_ttl_sec),
563            )
564            .is_none()
565        {
566            // Only increment the metric if the client was not already blocked
567            debug!("Blocking proxied client: {:?}", client);
568            metrics.requests_blocked_at_protocol.inc();
569            metrics.proxy_ip_blocklist_len.inc();
570        }
571    }
572}

574async fn delegate_policy_response(
575    response: PolicyResponse,
576    policy_config: &PolicyConfig,
577    node_fw_client: &NodeFWClient,
578    destination_port: u16,
579    metrics: Arc<TrafficControllerMetrics>,
580) -> Result<(), reqwest::Error> {
581    let PolicyResponse {
582        block_client,
583        block_proxied_client,
584    } = response;
585    let PolicyConfig {
586        connection_blocklist_ttl_sec,
587        proxy_blocklist_ttl_sec,
588        ..
589    } = policy_config;
590    let mut addresses = vec![];
591    if let Some(client_id) = block_client {
592        debug!("Delegating client blocking to firewall");
593        addresses.push(BlockAddress {
594            source_address: client_id.to_string(),
595            destination_port,
596            ttl: *connection_blocklist_ttl_sec,
597        });
598    }
599    if let Some(ip) = block_proxied_client {
600        debug!("Delegating proxied client blocking to firewall");
601        addresses.push(BlockAddress {
602            source_address: ip.to_string(),
603            destination_port,
604            ttl: *proxy_blocklist_ttl_sec,
605        });
606    }
607    if addresses.is_empty() {
608        Ok(())
609    } else {
610        metrics
611            .blocks_delegated_to_firewall
612            .inc_by(addresses.len() as u64);
613        match node_fw_client
614            .block_addresses(BlockAddresses { addresses })
615            .await
616        {
617            Ok(()) => Ok(()),
618            Err(err) => {
619                metrics.firewall_delegation_request_fail.inc();
620                Err(err)
621            }
622        }
623    }
624}

627pub struct TrafficSimMetrics {
628    pub num_requests: u64,
629    pub num_blocked: u64,
630    pub time_to_first_block: Option<Duration>,
631    pub abs_time_to_first_block: Option<Duration>,
632    pub total_time_blocked: Duration,
633    pub num_blocklist_adds: u64,
634}
635
636impl Default for TrafficSimMetrics {
637    fn default() -> Self {
638        Self {
639            num_requests: 0,
640            num_blocked: 0,
641            time_to_first_block: None,
642            abs_time_to_first_block: None,
643            total_time_blocked: Duration::from_micros(0),
644            num_blocklist_adds: 0,
645        }
646    }
647}
648
649impl Add for TrafficSimMetrics {
650    type Output = Self;
651
652    fn add(self, other: Self) -> Self {
653        Self {
654            num_requests: self.num_requests + other.num_requests,
655            num_blocked: self.num_blocked + other.num_blocked,
656            time_to_first_block: match (self.time_to_first_block, other.time_to_first_block) {
657                (Some(a), Some(b)) => Some(a + b),
658                (Some(a), None) => Some(a),
659                (None, Some(b)) => Some(b),
660                (None, None) => None,
661            },
662            abs_time_to_first_block: match (
663                self.abs_time_to_first_block,
664                other.abs_time_to_first_block,
665            ) {
666                (Some(a), Some(b)) => Some(a.min(b)),
667                (Some(a), None) => Some(a),
668                (None, Some(b)) => Some(b),
669                (None, None) => None,
670            },
671            total_time_blocked: self.total_time_blocked + other.total_time_blocked,
672            num_blocklist_adds: self.num_blocklist_adds + other.num_blocklist_adds,
673        }
674    }
675}

681impl TrafficSim {
682    pub async fn run(
683        policy: PolicyConfig,
684        num_clients: u8,
685        per_client_tps: usize,
686        duration: Duration,
687        report: bool,
688    ) -> TrafficSimMetrics {
689        assert!(
690            per_client_tps <= 10_000,
691            "per_client_tps must be less than 10,000. For higher values, increase num_clients"
692        );
693        assert!(num_clients < 20, "num_clients must be greater than 0");
694        assert!(num_clients > 0);
695        assert!(per_client_tps > 0);
696        assert!(duration.as_secs() > 0);
697
698        let controller = TrafficController::init_for_test(policy.clone(), None);
699        let tasks = (0..num_clients).map(|task_num| {
700            tokio::spawn(Self::run_single_client(
701                controller.clone(),
702                duration,
703                task_num,
704                per_client_tps,
705            ))
706        });
707
708        let status_task = if report {
709            Some(tokio::spawn(async move {
710                println!(
711                    "Running naive traffic simulation for {} seconds",
712                    duration.as_secs()
713                );
714                println!("Policy: {policy:#?}");
715                println!("Num clients: {num_clients}");
716                println!("TPS per client: {per_client_tps}");
717                println!(
718                    "Target total TPS: {}",
719                    per_client_tps * num_clients as usize
720                );
721                println!("\n");
722                for _ in 0..duration.as_secs() {
723                    print!(".");
724                    tokio::time::sleep(Duration::from_secs(1)).await;
725                }
726                println!();
727            }))
728        } else {
729            None
730        };
731
732        let metrics = futures::future::join_all(tasks).await.into_iter().fold(
733            TrafficSimMetrics::default(),
734            |acc, run_client_ret| match run_client_ret {
735                Ok(metrics) => acc + metrics,
736                Err(err) => {
737                    error!("Error running traffic sim client: {:?}", err);
738                    acc
739                }
740            },
741        );
742
743        if report {
744            status_task.unwrap().await.unwrap();
745            Self::report_metrics(metrics.clone(), duration, per_client_tps, num_clients);
746        }
747        metrics
748    }
749
750    async fn run_single_client(
751        controller: TrafficController,
752        duration: Duration,
753        task_num: u8,
754        per_client_tps: usize,
755    ) -> TrafficSimMetrics {
756        // Do an initial sleep for a random amount of time to smooth
757        // out the traffic. This shouldn't be strictly necessary and
758        // we can remove if we want more determinism
759        let sleep_time = Duration::from_micros(rand::thread_rng().gen_range(0..100));
760        tokio::time::sleep(sleep_time).await;
761
762        // collectors
763        let mut num_requests = 0;
764        let mut num_blocked = 0;
765        let mut time_to_first_block = None;
766        let mut total_time_blocked = Duration::from_micros(0);
767        let mut num_blocklist_adds = 0;
768        // state variables
769        let mut currently_blocked = false;
770        let mut time_blocked_start = Instant::now();
771        let start = Instant::now();
772
773        // we use ticker instead of sleep to be as close to the target TPS as possible,
774        let sleep_time = Duration::from_micros(1_000_000 / per_client_tps as u64);
775        let mut interval_ticker = time::interval(sleep_time);
776
777        while start.elapsed() < duration {
778            let client = Some(IpAddr::V4(Ipv4Addr::new(127, 0, 0, task_num)));
779            let allowed = controller.check(&client, &None).await;
780            if allowed {
781                if currently_blocked {
782                    total_time_blocked += time_blocked_start.elapsed();
783                    currently_blocked = false;
784                }
785                controller.tally(TrafficTally::new(
786                    client,
787                    // TODO add proxy IP for testing
788                    None,
789                    // TODO add weight adjustments
790                    None,
791                    Weight::one(),
792                ));
793            } else {
794                if !currently_blocked {
795                    time_blocked_start = Instant::now();
796                    currently_blocked = true;
797                    num_blocklist_adds += 1;
798                    if time_to_first_block.is_none() {
799                        time_to_first_block = Some(start.elapsed());
800                    }
801                }
802                num_blocked += 1;
803            }
804            num_requests += 1;
805
806            interval_ticker.tick().await;
807        }
808        TrafficSimMetrics {
809            num_requests,
810            num_blocked,
811            time_to_first_block,
812            abs_time_to_first_block: time_to_first_block,
813            total_time_blocked,
814            num_blocklist_adds,
815        }
816    }
817
818    fn report_metrics(
819        metrics: TrafficSimMetrics,
820        duration: Duration,
821        per_client_tps: usize,
822        num_clients: u8,
823    ) {
824        println!("TrafficSim metrics:");
825        println!("-------------------");
826        // The below two should be near equal
827        println!(
828            "Num expected requests: {}",
829            per_client_tps * (num_clients as usize) * duration.as_secs() as usize
830        );
831        println!("Num actual requests: {}", metrics.num_requests);
832        // This reflects the number of requests that were blocked, but note that once a
833        // client is added to the blocklist, all subsequent requests from that
834        // client are blocked until ttl is expired.
835        println!("Num blocked requests: {}", metrics.num_blocked);
836        // This metric on the other hand reflects the number of times a client was added
837        // to the blocklist and thus can be compared an the expectation based on
838        // the policy block threshold and ttl
839        println!(
840            "Num times added to blocklist: {}",
841            metrics.num_blocklist_adds
842        );
843        // This averages the duration for the first request to be blocked across all
844        // clients, which is useful for understanding if the policy is rate
845        // limiting based on expectation
846        let avg_first_block_time = metrics
847            .time_to_first_block
848            .map(|ttf| ttf / num_clients as u32);
849        println!("Average time to first block: {avg_first_block_time:?}");
850        // This is the time it took for the first request to be blocked across all
851        // clients, and is instead more useful for understanding false positives
852        // in terms of rate and magnitude.
853        println!(
854            "Absolute time to first block (across all clients): {:?}",
855            metrics.abs_time_to_first_block
856        );
857        // Useful for ensuring that TTL is respected
858        let avg_time_blocked = if metrics.num_blocklist_adds > 0 {
859            metrics.total_time_blocked.as_millis() as u64 / metrics.num_blocklist_adds
860        } else {
861            0
862        };
863        println!(
864            "Average time blocked (ttl): {:?}",
865            Duration::from_millis(avg_time_blocked)
866        );
867    }
868}

870pub fn parse_ip(ip: &str) -> Option<IpAddr> {
871    ip.parse::<IpAddr>().ok().or_else(|| {
872        ip.parse::<SocketAddr>()
873            .ok()
874            .map(|socket_addr| socket_addr.ip())
875            .or_else(|| {
876                error!("Failed to parse value of {:?} to ip address or socket.", ip,);
877                None
878            })
879    })
