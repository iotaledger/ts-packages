4use core::str::FromStr;
5
6use crypto::hashes::{Digest, blake2b::Blake2b256};
7use packable::{PackableExt, bounded::BoundedU16};
8
9use crate::block::{Error, TransactionId, output::OUTPUT_INDEX_RANGE};
10
11pub(crate) type OutputIndex =
12    BoundedU16<{ *OUTPUT_INDEX_RANGE.start() }, { *OUTPUT_INDEX_RANGE.end() }>;
13
14/// The identifier of an [`Output`](crate::block::output::Output).
15#[derive(Clone, Copy, Eq, PartialEq, Hash, Ord, PartialOrd, packable::Packable)]
16#[packable(unpack_error = Error)]
17pub struct OutputId {
18    transaction_id: TransactionId,
19    #[packable(unpack_error_with = Error::InvalidInputOutputIndex)]
20    index: OutputIndex,
21}
22
23impl OutputId {
24    /// The length of a [`OutputId`].
25    pub const LENGTH: usize = TransactionId::LENGTH + core::mem::size_of::<u16>();
26
27    /// Creates a new [`OutputId`].
28    pub fn new(transaction_id: TransactionId, index: u16) -> Result<Self, Error> {
29        index
30            .try_into()
31            .map(|index| Self {
32                transaction_id,
33                index,
34            })
35            .map_err(Error::InvalidInputOutputIndex)
36    }
37
38    /// Returns the [`TransactionId`] of an [`OutputId`].
39    #[inline(always)]
40    pub fn transaction_id(&self) -> &TransactionId {
41        &self.transaction_id
42    }
43
44    /// Returns the index of an [`OutputId`].
45    #[inline(always)]
46    pub fn index(&self) -> u16 {
47        self.index.get()
48    }
49
50    /// Creates a null [`OutputId`].
51    pub fn null() -> Self {
52        Self {
53            transaction_id: TransactionId::null(),
54            // Unwrap is fine because index is already known and valid.
55            index: 0u16.try_into().unwrap(),
56        }
57    }
58
59    /// Splits an [`OutputId`] into its [`TransactionId`] and index.
60    #[inline(always)]
61    pub fn split(self) -> (TransactionId, u16) {
62        (self.transaction_id, self.index())
63    }
64
65    /// Hash the [`OutputId`] with BLAKE2b-256.
66    #[inline(always)]
67    pub fn hash(&self) -> [u8; 32] {
68        Blake2b256::digest(self.pack_to_vec()).into()
69    }
70}
71
72#[cfg(feature = "serde")]
73string_serde_impl!(OutputId);
74
75impl TryFrom<[u8; Self::LENGTH]> for OutputId {
76    type Error = Error;
77
78    fn try_from(bytes: [u8; Self::LENGTH]) -> Result<Self, Self::Error> {
79        let (transaction_id, index) = bytes.split_at(TransactionId::LENGTH);
80
81        Self::new(
82            // Unwrap is fine because size is already known and valid.
83            From::<[u8; TransactionId::LENGTH]>::from(transaction_id.try_into().unwrap()),
84            // Unwrap is fine because size is already known and valid.
85            u16::from_le_bytes(index.try_into().unwrap()),
86        )
87    }
88}
89
90impl FromStr for OutputId {
91    type Err = Error;
92
93    fn from_str(s: &str) -> Result<Self, Self::Err> {
94        Self::try_from(prefix_hex::decode::<[u8; Self::LENGTH]>(s).map_err(Error::Hex)?)
95    }
96}
97
98impl core::fmt::Display for OutputId {
99    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
100        let mut buffer = [0u8; Self::LENGTH];
101        let (transaction_id, index) = buffer.split_at_mut(TransactionId::LENGTH);
102        transaction_id.copy_from_slice(self.transaction_id.as_ref());
103        index.copy_from_slice(&self.index().to_le_bytes());
104        write!(f, "{}", prefix_hex::encode(buffer))
105    }
106}
107
108impl core::fmt::Debug for OutputId {
109    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
110        write!(f, "OutputId({self})")
111    }
