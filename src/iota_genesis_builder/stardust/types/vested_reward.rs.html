20/// All basic outputs whose IDs start with this prefix represent vested rewards
21/// that were created during the stardust upgrade on IOTA mainnet.
22pub const VESTED_REWARD_ID_PREFIX: &str =
23    "0xb191c4bc825ac6983789e50545d5ef07a1d293a98ad974fc9498cb18";
24
25#[derive(Debug, thiserror::Error)]
26pub enum VestedRewardError {
27    #[error("failed to create genesis move object, owner: {owner}, timelock: {timelock:#?}")]
28    ObjectCreation {
29        owner: IotaAddress,
30        timelock: TimeLock<Balance>,
31        source: ExecutionError,
32    },
33    #[error("a vested reward must not contain native tokens")]
34    NativeTokensNotSupported,
35    #[error("a basic output is not a vested reward")]
36    NotVestedReward,
37    #[error("a vested reward must have two unlock conditions")]
38    UnlockConditionsNumberMismatch,
39    #[error("only timelocked vested rewards can be migrated as `TimeLock<Balance<IOTA>>`")]
40    UnlockedVestedReward,
41}
42
43/// Checks if an output is a timelocked vested reward.
44pub fn is_timelocked_vested_reward(
45    output_id: OutputId,
46    basic_output: &BasicOutput,
47    target_milestone_timestamp_sec: u32,
48) -> bool {
49    is_vested_reward(output_id, basic_output)
50        && basic_output
51            .unlock_conditions()
52            .is_time_locked(target_milestone_timestamp_sec)
53}
54
55/// Checks if an output is a vested reward, if it has a specific ID prefix,
56/// and if it contains a timelock unlock condition,
57/// and if an output has no native tokens,
58/// and if an output has only 2 unlock conditions and their address.
59pub fn is_vested_reward(output_id: OutputId, basic_output: &BasicOutput) -> bool {
60    let has_vesting_prefix = output_id.to_string().starts_with(VESTED_REWARD_ID_PREFIX);
61
62    has_vesting_prefix
63        && basic_output.unlock_conditions().timelock().is_some()
64        && basic_output.native_tokens().is_empty()
65        && basic_output.unlock_conditions().len() == 2
66        && basic_output.unlock_conditions().address().is_some()
67}
68
69/// Creates a `TimeLock<Balance<IOTA>>` from a Stardust-based Basic Output
70/// that represents a vested reward.
71pub fn try_from_stardust(
72    output_id: OutputId,
73    basic_output: &BasicOutput,
74    target_milestone_timestamp_sec: u32,
75) -> Result<TimeLock<Balance>, VestedRewardError> {
76    if !is_vested_reward(output_id, basic_output) {
77        return Err(VestedRewardError::NotVestedReward);
78    }
79
80    if !basic_output
81        .unlock_conditions()
82        .is_time_locked(target_milestone_timestamp_sec)
83    {
84        return Err(VestedRewardError::UnlockedVestedReward);
85    }
86
87    if basic_output.unlock_conditions().len() != 2 {
88        return Err(VestedRewardError::UnlockConditionsNumberMismatch);
89    }
90
91    if !basic_output.native_tokens().is_empty() {
92        return Err(VestedRewardError::NativeTokensNotSupported);
93    }
94
95    let id = UID::new(ObjectID::new(output_id.hash()));
96    let locked = Balance::new(basic_output.amount());
97
98    // We already checked the existence of the timelock unlock condition at this
99    // point.
100    let timelock_uc = basic_output
101        .unlock_conditions()
102        .timelock()
103        .expect("a vested reward should contain a timelock unlock condition");
104    let expiration_timestamp_ms = Into::<u64>::into(timelock_uc.timestamp()) * 1000;
105
106    let label = Option::Some(label_struct_tag_to_string(stardust_upgrade_label_type()));
107
108    Ok(TimeLock::new(id, locked, expiration_timestamp_ms, label))
109}

111/// Creates a genesis object from a time-locked balance.
112pub fn to_genesis_object(
113    timelock: TimeLock<Balance>,
114    owner: IotaAddress,
115    protocol_config: &ProtocolConfig,
116    tx_context: &TxContext,
117    version: SequenceNumber,
118) -> Result<Object, VestedRewardError> {
119    let move_object = {
120        MoveObject::new_from_execution(
121            MoveObjectType::timelocked_iota_balance(),
122            version,
123            timelock.to_bcs_bytes(),
124            protocol_config,
125        )
126        .map_err(|source| VestedRewardError::ObjectCreation {
127            owner,
128            timelock,
129            source,
130        })?
131    };
132
133    Ok(Object::new_from_genesis(
134        Data::Move(move_object),
135        Owner::AddressOwner(owner),
136        tx_context.digest(),
137    ))
138}

167    fn vested_reward_output(amount: u64, expiration_time_sec: u32) -> BasicOutput {
168        BasicOutputBuilder::new_with_amount(amount)
169            .add_unlock_condition(AddressUnlockCondition::new(
170                Ed25519Address::from_str(
171                    "0xebe40a263480190dcd7939447ee01aefa73d6f3cc33c90ef7bf905abf8728655",
172                )
173                .unwrap(),
174            ))
175            .add_unlock_condition(TimelockUnlockCondition::new(expiration_time_sec).unwrap())
176            .finish()
177            .unwrap()
178    }

180    #[test]
181    fn timelock_ser_deser_roundtrip() {
182        let id = UID::new(ObjectID::random());
183        let balance = Balance::new(100);
184        let expiration_timestamp_ms = 10;
185        let label = Option::Some(label_struct_tag_to_string(stardust_upgrade_label_type()));
186
187        let timelock = TimeLock::new(id, balance, expiration_timestamp_ms, label);
188
189        let timelock_bytes = timelock.to_bcs_bytes();
190        let deserialized_timelock: TimeLock<Balance> = bcs::from_bytes(&timelock_bytes).unwrap();
191
192        assert_eq!(deserialized_timelock.id(), timelock.id());
193        assert_eq!(deserialized_timelock.locked(), timelock.locked());
194        assert_eq!(
195            deserialized_timelock.expiration_timestamp_ms(),
196            timelock.expiration_timestamp_ms()
197        );
198        assert_eq!(deserialized_timelock.label(), timelock.label());
199    }
