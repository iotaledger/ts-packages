27#[derive(Debug, Clone)]
28pub struct InnerTemporaryStore {
29    pub input_objects: ObjectMap,
30    pub mutable_inputs: BTreeMap<ObjectID, (VersionDigest, Owner)>,
31    // All the written objects' sequence number should have been updated to the lamport version.
32    pub written: WrittenObjects,
33    pub loaded_runtime_objects: BTreeMap<ObjectID, DynamicallyLoadedObjectMetadata>,
34    pub events: TransactionEvents,
35    pub binary_config: BinaryConfig,
36    pub runtime_packages_loaded_from_db: BTreeMap<ObjectID, PackageObject>,
37    pub lamport_version: SequenceNumber,
38}

40impl InnerTemporaryStore {
41    pub fn get_output_keys(&self, effects: &TransactionEffects) -> Vec<InputKey> {
42        let mut output_keys: Vec<_> = self
43            .written
44            .iter()
45            .map(|(id, obj)| {
46                if obj.is_package() {
47                    InputKey::Package { id: *id }
48                } else {
49                    InputKey::VersionedObject {
50                        id: *id,
51                        version: obj.version(),
52                    }
53                }
54            })
55            .collect();
56
57        let deleted: HashMap<_, _> = effects
58            .deleted()
59            .iter()
60            .map(|oref| (oref.0, oref.1))
61            .collect();
62
63        // add deleted shared objects to the outputkeys that then get sent to
64        // notify_commit
65        let deleted_output_keys = deleted
66            .iter()
67            .filter(|(id, _)| {
68                self.input_objects
69                    .get(id)
70                    .is_some_and(|obj| obj.is_shared())
71            })
72            .map(|(id, seq)| InputKey::VersionedObject {
73                id: *id,
74                version: *seq,
75            });
76        output_keys.extend(deleted_output_keys);
77
78        // For any previously deleted shared objects that appeared mutably in the
79        // transaction, synthesize a notification for the next version of the
80        // object.
81        let smeared_version = self.lamport_version;
82        let deleted_accessed_objects = effects.deleted_mutably_accessed_shared_objects();
83        for object_id in deleted_accessed_objects.into_iter() {
84            let key = InputKey::VersionedObject {
85                id: object_id,
86                version: smeared_version,
87            };
88            output_keys.push(key);
89        }
90
91        output_keys
92    }
93}

95pub struct TemporaryModuleResolver<'a, R> {
96    temp_store: &'a InnerTemporaryStore,
97    fallback: R,
98}
99
100impl<'a, R> TemporaryModuleResolver<'a, R> {
101    pub fn new(temp_store: &'a InnerTemporaryStore, fallback: R) -> Self {
102        Self {
103            temp_store,
104            fallback,
105        }
106    }
107}
108
109impl<R> GetModule for TemporaryModuleResolver<'_, R>
110where
111    R: GetModule<Item = Arc<CompiledModule>, Error = anyhow::Error>,
112{
113    type Error = anyhow::Error;
114    type Item = Arc<CompiledModule>;
115
116    fn get_module_by_id(&self, id: &ModuleId) -> anyhow::Result<Option<Self::Item>, Self::Error> {
117        let obj = self.temp_store.written.get(&ObjectID::from(*id.address()));
118        if let Some(o) = obj {
119            if let Some(p) = o.data.try_as_package() {
120                return Ok(Some(Arc::new(p.deserialize_module(
121                    &id.name().into(),
122                    &self.temp_store.binary_config,
123                )?)));
124            }
125        }
126        self.fallback.get_module_by_id(id)
127    }
128}

130impl BackingPackageStore for InnerTemporaryStore {
131    fn get_package_object(&self, package_id: &ObjectID) -> IotaResult<Option<PackageObject>> {
132        Ok(self
133            .written
134            .get(package_id)
135            .cloned()
136            .map(PackageObject::new))
137    }
138}

140pub struct PackageStoreWithFallback<P, F> {
141    primary: P,
142    fallback: F,
143}
144
145impl<P, F> PackageStoreWithFallback<P, F> {
146    pub fn new(primary: P, fallback: F) -> Self {
147        Self { primary, fallback }
148    }
149}
150
151impl<P, F> BackingPackageStore for PackageStoreWithFallback<P, F>
152where
153    P: BackingPackageStore,
154    F: BackingPackageStore,
155{
156    fn get_package_object(&self, package_id: &ObjectID) -> IotaResult<Option<PackageObject>> {
157        if let Some(package) = self.primary.get_package_object(package_id)? {
158            Ok(Some(package))
159        } else {
160            self.fallback.get_package_object(package_id)
161        }
162    }
