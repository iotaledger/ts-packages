45/// This type captures the minimum amount of information from IotaSystemState
46/// needed by a validator to run the protocol. This allows us to decouple from
47/// the actual IotaSystemState type, and hence do not need to evolve it when we
48/// upgrade the IotaSystemState type. Evolving EpochStartSystemState is also a
49/// lot easier in that we could add optional fields and fill them with None for
50/// older versions. When we absolutely must delete fields, we could also add new
51/// db tables to store the new version. This is OK because we only store one
52/// copy of this as part of EpochStartConfiguration for the most recent epoch in
53/// the db.
54#[derive(Serialize, Deserialize, Clone, Debug, Eq, PartialEq)]
55#[enum_dispatch(EpochStartSystemStateTrait)]
56pub enum EpochStartSystemState {
57    V1(EpochStartSystemStateV1),
58    V2(EpochStartSystemStateV2),
59}

166// Helper macro to implement get_committee functions for different committee
167// types.
168macro_rules! impl_get_committee {
169    (
170        // function signature
171        fn $fn_name:ident -> $committee_ty:path,
172        // Committee authority type
173        authority = $authority_ty:ident,
174        // Config module where the committee authority public key types are defined
175        cfg = $cfg:ident,
176        // Label for logging purposes
177        label = $label:expr
178    ) => {
179        fn $fn_name(&self) -> $committee_ty {
180            let mut authorities: Vec<$authority_ty> =
181                Vec::with_capacity(self.committee_validators.len());
182
183            for validator in self.committee_validators.iter() {
184                authorities.push($authority_ty {
185                    stake: validator.voting_power as $cfg::Stake,
186                    address: validator.primary_address.clone(),
187                    hostname: validator.hostname.clone(),
188                    authority_key: <$cfg::AuthorityPublicKey>::new(
189                        validator.authority_pubkey.clone(),
190                    ),
191                    protocol_key: <$cfg::ProtocolPublicKey>::new(validator.protocol_pubkey.clone()),
192                    network_key: <$cfg::NetworkPublicKey>::new(validator.network_pubkey.clone()),
193                });
194            }
195
196            // Sort the authorities by their authority (public) key in ascending order, same
197            // as the order in the IOTA committee returned from get_iota_committee().
198            authorities.sort_by(|a1, a2| a1.authority_key.cmp(&a2.authority_key));
199
200            for ((i, authority), iota_authority_name) in authorities
201                .iter()
202                .enumerate()
203                .zip(self.get_iota_committee().names())
204            {
205                if iota_authority_name.0 != authority.authority_key.to_bytes() {
206                    error!(
207                        "Mismatched authority order between IOTA and {}! \
208                        Index {}, {} authority {:?}\nIota authority name {}",
209                        $label, i, $label, authority, iota_authority_name
210                    );
211                }
212            }
213
214            <$committee_ty>::new(self.epoch as $cfg::Epoch, authorities)
215        }
216    };
217}

251    fn get_iota_committee_with_network_metadata(&self) -> CommitteeWithNetworkMetadata {
252        let validators = self
253            .committee_validators
254            .iter()
255            .map(|validator| {
256                (

258                    (
259                        validator.voting_power,
260                        NetworkMetadata {

266                )

268            .collect();
269

271    }

274        let voting_rights = self
275            .committee_validators
276            .iter()

279        Committee::new(self.epoch, voting_rights)

296    fn get_validator_as_p2p_peers(&self, excluding_self: AuthorityName) -> Vec<PeerInfo> {
297        self.committee_validators
298            .iter()
299            .filter(|validator| validator.authority_name() != excluding_self)
300            .map(|validator| {
301                let address = validator
302                    .p2p_address
303                    .to_anemo_address()
304                    .into_iter()
305                    .collect::<Vec<_>>();
306                let peer_id = PeerId(validator.network_pubkey.0.to_bytes());
307                if address.is_empty() {
308                    warn!(
309                        ?peer_id,
310                        "Peer has invalid p2p address: {}", &validator.p2p_address
311                    );
312                }
313                PeerInfo {
314                    peer_id,
315                    affinity: PeerAffinity::High,
316                    address,
317                }
318            })
319            .collect()
320    }
321
322    fn get_authority_names_to_peer_ids(&self) -> HashMap<AuthorityName, PeerId> {
323        self.committee_validators
324            .iter()
325            .map(|validator| {
326                let name = validator.authority_name();
327                let peer_id = PeerId(validator.network_pubkey.0.to_bytes());
328
329                (name, peer_id)
330            })
331            .collect()
332    }
333
334    fn get_authority_names_to_hostnames(&self) -> HashMap<AuthorityName, String> {
335        self.committee_validators
336            .iter()
337            .map(|validator| {
338                let name = validator.authority_name();
339                let hostname = validator.hostname.clone();
340
341                (name, hostname)
342            })
343            .collect()
344    }
345
346    fn get_active_validators(&self) -> Vec<AuthorityPublicKey> {
347        // for V1 the committee and active validators are the same as there is no
348        // additional committee selection
349        self.committee_validators
350            .iter()
351            .map(|validator| validator.authority_pubkey.clone())
352            .collect()
353    }
354}

563        // GIVEN
564        let mut committee_validators = vec![];
565        let mut non_committee_validators = vec![];
566
567        for i in 0..10 {
568            let (iota_address, authority_key): (IotaAddress, AuthorityKeyPair) = get_key_pair();
569            let protocol_network_key = NetworkKeyPair::generate(&mut thread_rng());

576                iota_net_address: Multiaddr::empty(),
577                p2p_address: Multiaddr::empty(),
578                primary_address: Multiaddr::empty(),
579                voting_power: 1_000,
580                hostname: format!("committee-{i}").to_string(),
581            });
582
583            let (iota_address, authority_key): (IotaAddress, AuthorityKeyPair) = get_key_pair();
584            let protocol_network_key = NetworkKeyPair::generate(&mut thread_rng());
585
586            non_committee_validators.push(EpochStartValidatorInfoV1 {
587                iota_address,
588                authority_pubkey: authority_key.public().clone(),
589                network_pubkey: protocol_network_key.public().clone(),
590                protocol_pubkey: protocol_network_key.public().clone(),
591                iota_net_address: Multiaddr::empty(),
592                p2p_address: Multiaddr::empty(),
593                primary_address: Multiaddr::empty(),
594                voting_power: 500,
595                hostname: format!("non-committee-{i}").to_string(),
596            });
597        }
598
599        // Create active_validators list containing all validators in the desired order
600        let mut active_validators = committee_validators.clone();
601        active_validators.extend(non_committee_validators.clone());
602
603        let state = EpochStartSystemState::new_v2(
604            10,
605            ProtocolVersion::MAX.as_u64(),
606            0,
607            false,
608            0,
609            0,
610            committee_validators.clone(),
611            active_validators,
612        );
613
614        // WHEN
615        let iota_committee = state.get_iota_committee();
616        let consensus_committee = state.get_consensus_committee();
617        let active_validators = state.get_active_validators();
618
619        // THEN
620        // Assert committee validators details
621        assert_eq!(iota_committee.num_members(), 10);
622        assert_eq!(iota_committee.num_members(), consensus_committee.size());
623        assert_eq!(
624            iota_committee.validity_threshold(),
625            consensus_committee.validity_threshold()
626        );
627        assert_eq!(
628            iota_committee.quorum_threshold(),
629            consensus_committee.quorum_threshold()
630        );
631
632        // Verify committee validators are correctly mapped
633        for (authority_index, consensus_authority) in consensus_committee.authorities() {
634            let iota_authority_name = iota_committee
635                .authority_by_index(authority_index.value() as u32)
636                .unwrap();
637
638            assert_eq!(
639                consensus_authority.authority_key.to_bytes(),
640                iota_authority_name.0,
641                "IOTA Foundation & IOTA committee member of same index correspond to different public key"
642            );
643            assert_eq!(
644                consensus_authority.stake,
645                iota_committee.weight(iota_authority_name),
646                "IOTA Foundation & IOTA committee member stake differs"
647            );
648        }
649
650        // Verify active validators (should include all: committee + non-committee)
651        assert_eq!(active_validators.len(), 20); // 10 committee + 10 non-committee
652
653        // Verify order is preserved - active_validators should contain all validators
654        // in the expected order First committee validators, then non-committee
655        // validators
656        let expected_order: Vec<_> = committee_validators
657            .iter()
658            .chain(non_committee_validators.iter())
659            .collect();
660
661        for (i, expected_validator) in expected_order.iter().enumerate() {
662            let found_pubkey = &active_validators[i];
663            assert_eq!(
664                found_pubkey.as_bytes(),
665                expected_validator.authority_pubkey.as_bytes(),
666                "Order not preserved: expected validator at index {i}",
667            );
