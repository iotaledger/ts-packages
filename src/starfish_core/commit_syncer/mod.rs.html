5//! CommitSyncer implements efficient synchronization of committed data.
6//!
7//! During the operation of a committee of authorities for consensus, one or
8//! more authorities can fall behind the quorum in their received and accepted
9//! blocks. This can happen due to network disruptions, host crash, or other
10//! reasons. Authorities fell behind need to catch up to the quorum to be able
11//! to vote on the latest leaders. So efficient synchronization is necessary
12//! to minimize the impact of temporary disruptions and maintain smooth
13//! operations of the network.
14//! CommitSyncer achieves efficient synchronization by relying on the following:
15//! when blocks are included in commits with >= 2f+1 certifiers by stake, these
16//! blocks must have passed verifications on some honest validators, so
17//! re-verifying them is unnecessary. In fact, the quorum certified commits
18//! themselves can be trusted to be sent to IOTA directly, but for simplicity
19//! this is not done. Blocks from trusted commits still go through Core and
20//! committer.
21//!
22//! Another way CommitSyncer improves the efficiency of synchronization is
23//! parallel fetching: commits have a simple dependency graph (linear), so it is
24//! easy to fetch ranges of commits in parallel.
25//!
26//! Commit synchronization is an expensive operation, involving transferring
27//! large amount of data via the network. And it is not on the critical path of
28//! block processing. So the heuristics for synchronization, including triggers
29//! and retries, should be chosen to favor throughput and efficient resource
30//! usage, over faster reactions.

67pub(crate) enum CommitSyncType {
68    Fast,
69    Regular,

72impl CommitSyncType {
73    pub(crate) fn commit_sync_batch_size(&self, context: &Context) -> u32 {
74        match self {
75            CommitSyncType::Fast => context.parameters.fast_commit_sync_batch_size,
76            CommitSyncType::Regular => context.parameters.commit_sync_batch_size,
77        }
78    }
79
80    pub(crate) fn as_str(&self) -> &'static str {
81        match self {
82            CommitSyncType::Fast => "fast_commit_sync",
83            CommitSyncType::Regular => "commit_sync",
84        }
85    }
86
87    pub(crate) fn should_schedule(
88        &self,
89        gap: u32,
90        commit_sync_gap_threshold: u32,
91        consensus_fast_commit_sync: bool,
92    ) -> bool {
93        match self {
94            // Fast syncer requires consensus_fast_commit_sync to be enabled
95            CommitSyncType::Fast => consensus_fast_commit_sync && gap > commit_sync_gap_threshold,
96            // Regular syncer handles all gaps when consensus_fast_commit_sync is disabled,
97            // otherwise only handles small gaps
98            CommitSyncType::Regular => {
99                !consensus_fast_commit_sync || gap <= commit_sync_gap_threshold
100            }
101        }
102    }
103}

105/// Verifies that fetched block headers match the requested block refs.
106/// Returns verified headers or an error if count/reference mismatch.
107pub(crate) fn verify_fetched_headers(
108    peer: AuthorityIndex,
109    request_block_refs: &[BlockRef],
110    serialized_block_headers: Vec<Bytes>,
111) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
112    // 1. Verify count matches
113    if request_block_refs.len() != serialized_block_headers.len() {
114        return Err(ConsensusError::UnexpectedNumberOfHeadersFetched {
115            authority: peer,
116            requested: request_block_refs.len(),
117            received_headers: serialized_block_headers.len(),
118        });
119    }
120
121    // 2. Verify each header's reference matches requested
122    serialized_block_headers
123        .into_iter()
124        .zip(request_block_refs)
125        .map(|(serialized, requested_ref)| {
126            let header = VerifiedBlockHeader::new_from_bytes(serialized)?;
127            if *requested_ref != header.reference() {
128                return Err(ConsensusError::UnexpectedBlockHeaderForCommit {
129                    peer,
130                    requested: *requested_ref,
131                    received: header.reference(),
132                });
133            }
134            Ok(header)
135        })
136        .collect()
137}

139// Handle to stop the CommitSyncer loop.
140pub(crate) struct CommitSyncerHandle {
141    schedule_task: JoinHandle<()>,
142    tx_shutdown: oneshot::Sender<()>,
143}
144
145impl CommitSyncerHandle {
146    pub(crate) async fn stop(self) {
147        let _ = self.tx_shutdown.send(());
148        // Do not abort schedule task, which waits for fetches to shut down.
149        if let Err(e) = self.schedule_task.await {
150            if e.is_panic() {
151                std::panic::resume_unwind(e.into_panic());
152            }
153        }
154    }
155}

168impl<C: NetworkClient> Inner<C> {
169    /// Calculates the threshold for unhandled commits to apply backpressure.
170    /// When the gap between synced and scheduled commits exceeds this
171    /// threshold, scheduling new fetches should pause to let the handler
172    /// catch up.
173    pub(crate) fn unhandled_commits_threshold(&self) -> CommitIndex {
174        self.context.parameters.commit_sync_batch_size
175            * (self.context.parameters.commit_sync_batches_ahead as u32)
176    }
177
178    /// Verifies the commits and also certifies them using the provided vote
179    /// blocks for the last commit. The method returns the trusted commits
180    /// and the verified voting block headers.
181    pub(crate) fn verify_commits(
182        &self,
183        peer: AuthorityIndex,
184        commit_range: CommitRange,
185        serialized_commits: Vec<Bytes>,
186        serialized_vote_blocks_headers: Vec<Bytes>,
187        max_commits: usize,
188    ) -> ConsensusResult<(Vec<TrustedCommit>, Vec<VerifiedBlockHeader>)> {
189        // Validate response size - peer should not return more than max_commits
190        if serialized_commits.len() > max_commits {
191            return Err(ConsensusError::TooManyCommitsFromPeer {
192                peer,
193                count: serialized_commits.len() as CommitIndex,
194                limit: max_commits as CommitIndex,
195            });
196        }
197
198        // Parse and verify commits.
199        let mut commits = Vec::new();
200        for serialized in &serialized_commits {
201            let commit: Commit =
202                bcs::from_bytes(serialized).map_err(ConsensusError::MalformedCommit)?;
203            let digest = TrustedCommit::compute_digest(serialized);
204            if commits.is_empty() {
205                // start is inclusive, so first commit must be at the start index.
206                if commit.index() != commit_range.start() {
207                    return Err(ConsensusError::UnexpectedStartCommit {
208                        peer,
209                        start: commit_range.start(),
210                        commit: Box::new(commit),
211                    });
212                }
213            } else {
214                // Verify next commit increments index and references the previous digest.
215                let (last_commit_digest, last_commit): &(CommitDigest, Commit) =
216                    commits.last().unwrap();
217                if commit.index() != last_commit.index() + 1
218                    || &commit.previous_digest() != last_commit_digest
219                {
220                    return Err(ConsensusError::UnexpectedCommitSequence {
221                        peer,
222                        prev_commit: Box::new(last_commit.clone()),
223                        curr_commit: Box::new(commit),
224                    });
225                }
226            }
227            commits.push((digest, commit));
228        }
229        let Some((end_commit_digest, end_commit)) = commits.last() else {
230            return Err(ConsensusError::NoCommitReceived { peer });
231        };
232
233        // Parse and verify blocks. Then accumulate votes on the end commit.
234        let end_commit_ref = CommitRef::new(end_commit.index(), *end_commit_digest);
235        let mut stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
236        let mut verified_voting_headers = Vec::new();
237        for serialized_block_header in serialized_vote_blocks_headers {
238            let signed_block_header: SignedBlockHeader = bcs::from_bytes(&serialized_block_header)
239                .map_err(ConsensusError::MalformedHeader)?;
240            // The block signature needs to be verified.
241            self.block_verifier.verify(&signed_block_header)?;
242            for vote in signed_block_header.commit_votes() {
243                if *vote == end_commit_ref {
244                    stake_aggregator.add(signed_block_header.author(), &self.context.committee);
245                }
246            }
247            // Store the verified voting block header
248            verified_voting_headers.push(VerifiedBlockHeader::new_verified(
249                signed_block_header,
250                serialized_block_header,
251            ));
252        }
253
254        // Check if the end commit has enough votes.
255        if !stake_aggregator.reached_threshold(&self.context.committee) {
256            return Err(ConsensusError::NotEnoughCommitVotes {
257                stake: stake_aggregator.stake(),
258                peer,
259                commit: Box::new(end_commit.clone()),
260            });
261        }
262
263        let trusted_commits = commits
264            .into_iter()
265            .zip(serialized_commits)
266            .map(|((_d, c), s)| TrustedCommit::new_trusted(c, s))
267            .collect();
268        Ok((trusted_commits, verified_voting_headers))
269    }
270}

272/// Verifies transactions against their block headers and returns a map of
273/// BlockRef to VerifiedTransactions.
274pub(crate) fn verify_transactions_with_headers(
275    context: Arc<Context>,
276    peer: AuthorityIndex,
277    serialized_transactions: BTreeMap<GenericTransactionRef, Bytes>,
278    block_headers: BTreeMap<BlockRef, VerifiedBlockHeader>,
279) -> ConsensusResult<BTreeMap<GenericTransactionRef, VerifiedTransactions>> {
280    let mut verified_transactions_map = BTreeMap::new();
281    let mut encoder = create_encoder(&context);
282    for (committed_transactions_ref, inner_serialized_transactions) in serialized_transactions {
283        let block_ref = match committed_transactions_ref {
284            GenericTransactionRef::BlockRef(br) => br,
285            _ => {
286                return Err(ConsensusError::TransactionRefVariantMismatch {
287                    protocol_flag_enabled: false,
288                    expected_variant: "BlockRef",
289                    received_variant: "TransactionRef",
290                });
291            }
292        };
293        // Step 1: Get the block header and verify that the transactions commitment
294        // matches. This ensures the transactions we received are exactly
295        // the ones that were included in the block when it was created.
296        let block_header = block_headers
297            .get(&block_ref)
298            .expect("header for fetched transactions must exist");
299
300        if block_header.transactions_commitment()
301            != TransactionsCommitment::compute_transactions_commitment(
302                &inner_serialized_transactions,
303                &context,
304                &mut encoder,
305            )
306            .expect("correct computation of the transactions commitment should be successful")
307        {
308            return Err(ConsensusError::TransactionCommitmentFailure {
309                round: block_ref.round,
310                author: block_ref.author,
311                peer,
312            });
313        }
314
315        // Step 2: Deserialize the actual transactions vector.
316        let transactions: Vec<Transaction> = bcs::from_bytes(&inner_serialized_transactions)
317            .map_err(ConsensusError::MalformedTransactions)?;
318
319        // Step 3: Create a VerifiedTransactions instance and insert into map
320        let verified_transactions = VerifiedTransactions::new(
321            transactions,
322            TransactionRef::new(block_ref, block_header.transactions_commitment()),
323            Some(block_ref.digest),
324            inner_serialized_transactions,
325        );
326
327        verified_transactions_map.insert(
328            GenericTransactionRef::BlockRef(block_ref),
329            verified_transactions,
330        );
331    }
332
333    Ok(verified_transactions_map)
334}

336/// Verifies transactions against their transaction refs and returns a map of
337/// BlockRef to VerifiedTransactions.
338pub(crate) fn verify_transactions_with_transactions_refs(
339    context: &Arc<Context>,
340    peer: AuthorityIndex,
341    serialized_transactions: BTreeMap<GenericTransactionRef, Bytes>,
342) -> ConsensusResult<BTreeMap<GenericTransactionRef, VerifiedTransactions>> {
343    let mut verified_transactions_map = BTreeMap::new();
344    let mut encoder = create_encoder(context);
345    for (committed_transactions_ref, inner_serialized_transactions) in serialized_transactions {
346        let transaction_ref = match committed_transactions_ref {
347            GenericTransactionRef::TransactionRef(tx_ref) => tx_ref,
348            _ => {
349                return Err(ConsensusError::TransactionRefVariantMismatch {
350                    protocol_flag_enabled: true,
351                    expected_variant: "TransactionRef",
352                    received_variant: "BlockRef",
353                });
354            }
355        };
356        // Step 1: Verify that the transaction commitment matches.
357        if transaction_ref.transactions_commitment
358            != TransactionsCommitment::compute_transactions_commitment(
359                &inner_serialized_transactions,
360                context,
361                &mut encoder,
362            )
363            .expect("correct computation of the transactions commitment should be successful")
364        {
365            return Err(ConsensusError::TransactionCommitmentFailure {
366                round: transaction_ref.round,
367                author: transaction_ref.author,
368                peer,
369            });
370        }
371
372        // Step 2: Deserialize the actual transactions vector.
373        let transactions: Vec<Transaction> = bcs::from_bytes(&inner_serialized_transactions)
374            .map_err(ConsensusError::MalformedTransactions)?;
375
376        // Step 3: Create a VerifiedTransactions instance and insert into map
377        let verified_transactions = VerifiedTransactions::new(
378            transactions,
379            transaction_ref,
380            None,
381            inner_serialized_transactions,
382        );
383
384        verified_transactions_map.insert(
385            GenericTransactionRef::TransactionRef(transaction_ref),
386            verified_transactions,
387        );
388    }
389
390    Ok(verified_transactions_map)
391}

393/// Generic fetch loop that retries fetching data from available authorities
394/// until a request succeeds. This is shared between RegularCommitSyncer and
395/// FastCommitSyncer.
396///
397/// # Type Parameters
398/// - `C`: Network client type
399/// - `T`: Fetched data type (CertifiedCommits for regular, (Vec<TrustedCommit>,
400///   Vec<CommittedSubDag>) for fast)
401/// - `F`: Fetch function type
402/// - `Fut`: Future returned by fetch function
403///
404/// # Parameters
405/// - `inner`: Shared context and dependencies
406/// - `commit_range`: The range of commits to fetch
407/// - `fetch_timeout_multiplier`: Multiplier for timeout calculation (4 for
408///   regular, 2 for fast)
409/// - `fetch_once_fn`: Implementation-specific fetch function
410///
411/// # Returns
412/// Tuple of (end_commit_index, fetched_data)
413#[cfg_attr(test, tracing::instrument(skip_all, fields(authority = %inner.context.own_index)))]
414pub(crate) async fn fetch_loop<C, T, F, Fut>(
415    inner: Arc<Inner<C>>,
416    commit_range: CommitRange,
417    fetch_timeout_multiplier: u32,
418    fetch_once_fn: F,
419) -> (CommitIndex, T)
420where
421    C: NetworkClient,
422    T: Send,
423    F: Fn(Arc<Inner<C>>, AuthorityIndex, CommitRange, Duration) -> Fut,
424    Fut: std::future::Future<Output = ConsensusResult<T>> + Send,
425{
426    // Individual request base timeout.
427    #[cfg(not(test))]
428    const TIMEOUT: Duration = Duration::from_secs(10);
429    #[cfg(test)]
430    const TIMEOUT: Duration = Duration::from_millis(500);
431    // Max per-request timeout will be base timeout times a multiplier.
432    // At the extreme, this means there will be 120s timeout to fetch
433    // max_blocks_per_fetch blocks.
434    const MAX_TIMEOUT_MULTIPLIER: u32 = 12;
435    // timeout * max number of targets should be reasonably small, so the
436    // system can adjust to slow network or large data sizes quickly.
437    const MAX_NUM_TARGETS: usize = 24;
438    let mut timeout_multiplier = 0;
439
440    let _timer = inner
441        .context
442        .metrics
443        .node_metrics
444        .commit_sync_fetch_loop_latency
445        .start_timer();
446    info!(
447        "[{}] Starting to fetch commits in {commit_range:?} ...",
448        inner.sync_type.as_str()
449    );
450    loop {
451        // Attempt to fetch commits and blocks through min(committee size,
452        // MAX_NUM_TARGETS) peers.
453        let mut target_authorities = inner
454            .context
455            .committee
456            .authorities()
457            .filter_map(|(i, _)| {
458                if i != inner.context.own_index {
459                    Some(i)
460                } else {
461                    None
462                }
463            })
464            .collect_vec();
465        #[cfg(not(test))]
466        target_authorities.shuffle(&mut ThreadRng::default());
467        target_authorities.truncate(MAX_NUM_TARGETS);
468        // Increase timeout multiplier for each loop until MAX_TIMEOUT_MULTIPLIER.
469        timeout_multiplier = (timeout_multiplier + 1).min(MAX_TIMEOUT_MULTIPLIER);
470        let request_timeout = TIMEOUT * timeout_multiplier;
471
472        let fetch_timeout = request_timeout * fetch_timeout_multiplier;
473        // Try fetching from the selected target authority.
474        for authority in target_authorities {
475            match tokio::time::timeout(
476                fetch_timeout,
477                fetch_once_fn(
478                    inner.clone(),
479                    authority,
480                    commit_range.clone(),
481                    request_timeout,
482                ),
483            )
484            .await
485            {
486                Ok(Ok(data)) => {
487                    info!(
488                        "[{}] Finished fetching commits in {commit_range:?}",
489                        inner.sync_type.as_str()
490                    );
491                    return (commit_range.end(), data);
492                }
493                Ok(Err(e)) => {
494                    let hostname = inner
495                        .context
496                        .committee
497                        .authority(authority)
498                        .hostname
499                        .clone();
500                    warn!(
501                        "[{}] Failed to fetch {commit_range:?} from {hostname}: {}",
502                        inner.sync_type.as_str(),
503                        e
504                    );
505                    let error: &'static str = e.into();
506                    inner
507                        .context
508                        .metrics
509                        .node_metrics
510                        .commit_sync_fetch_once_errors
511                        .with_label_values(&[hostname.as_str(), error, inner.sync_type.as_str()])
512                        .inc();
513                }
514                Err(_) => {
515                    let hostname = inner
516                        .context
517                        .committee
518                        .authority(authority)
519                        .hostname
520                        .clone();
521                    warn!(
522                        "[{}] Timed out fetching {commit_range:?} from {authority}",
523                        inner.sync_type.as_str()
524                    );
525                    inner
526                        .context
527                        .metrics
528                        .node_metrics
529                        .commit_sync_fetch_once_errors
530                        .with_label_values(&[
531                            hostname.as_str(),
532                            "FetchTimeout",
533                            inner.sync_type.as_str(),
534                        ])
535                        .inc();
536                }
537            }
538        }
539        // Avoid busy looping, by waiting for a while before retrying.
540        sleep(TIMEOUT).await;
541    }
542}

544/// Generic function to start pending fetches while respecting parallelism
545/// limits. This is shared between RegularCommitSyncer and FastCommitSyncer.
546///
547/// # Parameters
548/// - `inner`: Shared context and dependencies
549/// - `pending_fetches`: Set of commit ranges pending fetch
550/// - `fetched_ranges_count`: Number of fetched ranges waiting to be processed
551/// - `inflight_fetches_count`: Number of currently in-flight fetch tasks
552/// - `synced_commit_index`: Latest synced commit index
553/// - `spawn_fn`: Closure to spawn a new fetch task
554///
555/// # Returns
556/// Updated counts after spawning new fetches
557pub(crate) fn try_start_fetches<C, F>(
558    inner: &Arc<Inner<C>>,
559    pending_fetches: &mut BTreeSet<CommitRange>,
560    fetched_ranges_count: usize,
561    inflight_fetches_count: usize,
562    synced_commit_index: CommitIndex,
563    mut spawn_fn: F,
564) -> (usize, usize)
565where
566    C: NetworkClient,
567    F: FnMut(CommitRange),
568{
569    // Cap parallel fetches based on configured limit and committee size, to avoid
570    // overloading the network. Also when there are too many fetched block headers
571    // that cannot be sent to Core before an earlier fetch has not finished,
572    // reduce parallelism so the earlier fetch can retry on a better host and
573    // succeed.
574    let target_parallel_fetches = inner
575        .context
576        .parameters
577        .commit_sync_parallel_fetches
578        .min(inner.context.committee.size() * 2 / 3)
579        .min(
580            inner
581                .context
582                .parameters
583                .commit_sync_batches_ahead
584                .saturating_sub(fetched_ranges_count),
585        )
586        .max(1);
587
588    let mut new_inflight_count = inflight_fetches_count;
589
590    // Start new fetches if there are pending batches and available slots.
591    loop {
592        if new_inflight_count >= target_parallel_fetches {
593            break;
594        }
595        if !pending_fetches.is_empty() {
596            info!(
597                "[{}] Pending fetches: {:?}, target parallel fetches: {}, inflight fetch number: {}",
598                inner.sync_type.as_str(),
599                pending_fetches,
600                target_parallel_fetches,
601                new_inflight_count
602            );
603        }
604        let Some(commit_range) = pending_fetches.pop_first() else {
605            break;
606        };
607        spawn_fn(commit_range);
608        new_inflight_count += 1;
609    }
610
611    let metrics = &inner.context.metrics.node_metrics;
612    let sync_label = inner.sync_type.as_str();
613    metrics
614        .commit_sync_inflight_fetches
615        .with_label_values(&[sync_label])
616        .set(new_inflight_count as i64);
617    metrics
618        .commit_sync_pending_fetches
619        .with_label_values(&[sync_label])
620        .set(pending_fetches.len() as i64);
621    metrics
622        .commit_sync_highest_synced_index
623        .with_label_values(&[sync_label])
624        .set(synced_commit_index as i64);
625
626    (new_inflight_count, pending_fetches.len())
627}

633/// Result of scheduling commit range fetches.
634pub(crate) struct ScheduleResult {
635    /// The new highest scheduled commit index (if any ranges were scheduled).
636    pub new_highest_scheduled: Option<CommitIndex>,
637    /// The commit ranges that were scheduled for fetching.
638    pub ranges_scheduled: Vec<CommitRange>,
639}

641/// Creates commit range batches for fetching, respecting backpressure.
642///
643/// This function is shared between RegularCommitSyncer and FastCommitSyncer.
644/// It calculates which commit ranges should be fetched next based on:
645/// - The current gap between local and quorum commit indices
646/// - Backpressure from unhandled commits
647///
648/// # Parameters
649/// - `inner`: Shared context and dependencies
650/// - `fetch_after_index`: Start fetching from commits after this index
651/// - `quorum_commit_index`: The commit index that quorum has reached
652/// - `highest_handled_index`: The highest commit index that has been processed
653/// - `unhandled_commits_threshold`: Threshold for applying backpressure
654///
655/// # Returns
656/// A `ScheduleResult` with the new highest scheduled index and ranges to fetch.
657pub(crate) fn schedule_commit_ranges<C: NetworkClient>(
658    inner: &Inner<C>,
659    fetch_after_index: CommitIndex,
660    quorum_commit_index: CommitIndex,
661    highest_handled_index: CommitIndex,
662    unhandled_commits_threshold: CommitIndex,
663) -> ScheduleResult {
664    let step = inner.sync_type.commit_sync_batch_size(&inner.context);
665    let mut result = ScheduleResult {
666        new_highest_scheduled: None,
667        ranges_scheduled: Vec::new(),
668    };
669
670    for prev_end in (fetch_after_index..=quorum_commit_index).step_by(step as usize) {
671        let range_start = prev_end + 1;
672        let range_end = prev_end + step;
673
674        // Don't schedule incomplete batches
675        if quorum_commit_index < range_end {
676            break;
677        }
678
679        // Apply backpressure if handler is lagging
680        if highest_handled_index + unhandled_commits_threshold < range_end {
681            warn!(
682                "[{}] Skip scheduling new commit fetches: handler lagging. \
683                 highest_handled={}, threshold={}",
684                inner.sync_type.as_str(),
685                highest_handled_index,
686                unhandled_commits_threshold
687            );
688            break;
689        }
690
691        result
692            .ranges_scheduled
693            .push((range_start..=range_end).into());
694        result.new_highest_scheduled = Some(range_end);
695    }
696
697    result
698}

700/// Handles JoinError from fetch tasks.
701///
702/// # Returns
703/// `true` if the syncer should shutdown, `false` otherwise.
704pub(crate) fn handle_fetch_join_error(
705    error: &tokio::task::JoinError,
706    sync_type: &CommitSyncType,
707) -> bool {
708    if error.is_panic() {
709        // Re-panic in the main task
710        return true;
711    }
712    warn!(
713        "[{}] Fetch cancelled. Shutting down: {}",
714        sync_type.as_str(),
715        error
716    );
717    true // Signal to shutdown
718}

720/// Re-queues unfetched portion of a commit range for retry.
721///
722/// When a fetch returns partial results (fewer commits than requested),
723/// this function queues the remaining range for another fetch attempt.
724pub(crate) fn requeue_partial_range(
725    pending_fetches: &mut BTreeSet<CommitRange>,
726    commit_end: CommitIndex,
727    target_end: CommitIndex,
728) {
729    if commit_end < target_end {
730        pending_fetches.insert((commit_end + 1..=target_end).into());
731    }
