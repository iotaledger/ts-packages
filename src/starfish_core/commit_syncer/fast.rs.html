42/// Output from fast sync fetch operations containing commits, subdags, and
43/// voting headers.
44#[derive(Clone, Debug, Default)]
45pub struct FastSyncOutput {
46    pub commits: Vec<TrustedCommit>,
47    pub committed_subdags: Vec<CommittedSubDag>,
48    pub voting_block_headers: Vec<VerifiedBlockHeader>,
49}

51pub(crate) struct FastCommitSyncer<C: NetworkClient> {
52    // States shared by scheduler and fetch tasks.
53
54    // Shared components wrapper.
55    inner: Arc<Inner<C>>,
56
57    // States only used by the scheduler.
58
59    // Inflight requests to fetch commits from different authorities.
60    inflight_fetches: JoinSet<(u32, FastSyncOutput)>,
61    // Additional ranges of commits to fetch.
62    pending_fetches: BTreeSet<CommitRange>,
63    // Fetched commits and blocks by commit range.
64    fetched_ranges: BTreeMap<CommitRange, FastSyncOutput>,
65    // Highest commit index among inflight and pending fetches.
66    // Used to determine the start of new ranges to be fetched.
67    highest_scheduled_index: Option<CommitIndex>,
68    // Highest index among fetched commits, after commits and blocks are verified.
69    // Used for metrics.
70    highest_fetched_commit_index: CommitIndex,
71    // The commit index that is the max of highest local commit index and commit index inflight to
72    // Core. Used to determine if fetched blocks can be sent to Core without gaps.
73    synced_commit_index: CommitIndex,
74    // Whether the syncer is in "close to quorum" mode, meaning remaining gap < batch size.
75    // When this is true, the syncer will fetch block headers and transactions for cached rounds
76    // before completing fast sync.
77    close_to_quorum_mode: bool,
78    // Whether the fast syncer has actually fetched any data. Close-to-quorum mode only
79    // activates after this is true. Reset to false after reinitialization completes.
80    has_fetched_data: bool,
81}

97            commit_consumer_monitor,
98            network_client,
99            block_verifier,
100            dag_state,
101            sync_type: CommitSyncType::Fast,
102        });
103        let last_solid_commit_index = inner.dag_state.read().last_solid_commit_index();
104        info!(
105            "[fast_commit_sync] Initialized with synced_commit_index={}",
106            last_solid_commit_index
107        );
108        FastCommitSyncer {
109            inner,
110            inflight_fetches: JoinSet::new(),
111            pending_fetches: BTreeSet::new(),
112            fetched_ranges: BTreeMap::new(),
113            highest_scheduled_index: None,
114            highest_fetched_commit_index: 0,
115            synced_commit_index: last_solid_commit_index,
116            close_to_quorum_mode: false,
117            has_fetched_data: false,
118        }
119    }

121    pub(crate) fn start(self) -> CommitSyncerHandle {
122        let (tx_shutdown, rx_shutdown) = oneshot::channel();
123        let schedule_task = spawn_logged_monitored_task!(self.schedule_loop(rx_shutdown,));
124        CommitSyncerHandle {
125            schedule_task,
126            tx_shutdown,
127        }
128    }
129    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %self.inner.context.own_index)))]
130    async fn schedule_loop(mut self, mut rx_shutdown: oneshot::Receiver<()>) {
131        let mut interval = tokio::time::interval(Duration::from_millis(500));
132        interval.set_missed_tick_behavior(MissedTickBehavior::Skip);
133
134        loop {
135            tokio::select! {
136                // Periodically, schedule new fetches if the node is falling behind.
137                _ = interval.tick() => {
138                    self.try_schedule_once();
139                }
140                // Handles results from fetch tasks.
141                Some(result) = self.inflight_fetches.join_next(), if !self.inflight_fetches.is_empty() => {
142                    if let Err(ref e) = result {
143                        if e.is_panic() {
144                            std::panic::resume_unwind(result.unwrap_err().into_panic());
145                        }
146                        if handle_fetch_join_error(e, &self.inner.sync_type) {
147                            // If any fetch is cancelled or panicked, try to shutdown and exit the loop.
148                            self.inflight_fetches.shutdown().await;
149                            return;
150                        }
151                    }
152                    let (target_end, output) = result.unwrap();
153                    self.handle_fetch_result(target_end, output).await;
154                }
155                _ = &mut rx_shutdown => {
156                    // Shutdown requested.
157                    info!("[{}] FastCommitSyncer shutting down ...", self.inner.sync_type.as_str());
158                    self.inflight_fetches.shutdown().await;
159                    return;
160                }
161            }
162
163            self.try_start_fetches();
164
165            // Handle close-to-quorum mode: when all fetches complete and we're close
166            // to the quorum, fetch block headers for a large enough number of rounds and
167            // reinitialize.
168            if self.close_to_quorum_mode
169                && self.inflight_fetches.is_empty()
170                && self.pending_fetches.is_empty()
171                && self.fetched_ranges.is_empty()
172            {
173                info!(
174                    "[{}] Close-to-quorum: all fetches complete, fetching headers for cached_rounds",
175                    self.inner.sync_type.as_str()
176                );
177
178                match Self::fetch_headers_for_reinitialization(self.inner.clone()).await {
179                    Ok(headers) => {
180                        if let Err(e) = self
181                            .inner
182                            .core_thread_dispatcher
183                            .reinitialize_components(headers)
184                            .await
185                        {
186                            warn!(
187                                "[{}] Failed to reinitialize components: {}",
188                                self.inner.sync_type.as_str(),
189                                e
190                            );
191                        } else {
192                            info!(
193                                "[{}] Components reinitialized, fast sync complete",
194                                self.inner.sync_type.as_str()
195                            );
196                        }
197                    }
198                    Err(e) => {
199                        warn!(
200                            "[{}] Failed to fetch headers for cached rounds: {}",
201                            self.inner.sync_type.as_str(),
202                            e
203                        );
204                    }
205                }
206
207                // Reset state - regular syncer takes over for now.
208                // Keep the loop running so we can re-activate if the node
209                // falls behind significantly again.
210                self.close_to_quorum_mode = false;
211                self.has_fetched_data = false;
212                self.highest_scheduled_index = None;
213                self.synced_commit_index = self.inner.dag_state.read().last_solid_commit_index();
214
215                info!(
216                    "[{}] Fast sync complete, staying active for potential reactivation",
217                    self.inner.sync_type.as_str()
218                );
219            }
220        }
221    }

223    fn try_schedule_once(&mut self) {
224        let quorum_commit_index = self.inner.commit_vote_monitor.quorum_commit_index();
225        let last_solid_commit_index = self.inner.dag_state.read().last_solid_commit_index();
226        let highest_handled_index = self.inner.commit_consumer_monitor.highest_handled_commit();
227        let highest_scheduled_index = self.highest_scheduled_index.unwrap_or(0);
228        let unhandled_commits_threshold = self.inner.unhandled_commits_threshold();
229        let step = self
230            .inner
231            .sync_type
232            .commit_sync_batch_size(&self.inner.context);
233
234        // Skip scheduling depending on sync type and gap threshold.
235        let gap = quorum_commit_index.saturating_sub(last_solid_commit_index);
236        let should_schedule = self.has_fetched_data
237            || self.inner.sync_type.should_schedule(
238                gap,
239                self.inner.context.parameters.commit_sync_gap_threshold,
240                self.inner
241                    .context
242                    .protocol_config
243                    .consensus_fast_commit_sync()
244                    && self.inner.context.parameters.enable_fast_commit_syncer,
245            );
246
247        if should_schedule {
248            let metrics = &self.inner.context.metrics.node_metrics;
249            metrics
250                .commit_sync_quorum_index
251                .set(quorum_commit_index as i64);
252            metrics
253                .commit_sync_local_index
254                .set(last_solid_commit_index as i64);
255            // Update synced_commit_index periodically to make sure it is not smaller than
256            // local solid commit index.
257            self.synced_commit_index = self.synced_commit_index.max(last_solid_commit_index);
258
259            // TODO: cleanup inflight fetches that are no longer needed.
260            let fetch_after_index = self
261                .synced_commit_index
262                .max(self.highest_scheduled_index.unwrap_or(0));
263
264            debug!(
265                "[{}] Checking to schedule fetches: synced_commit_index={}, highest_handled_index={}, highest_scheduled_index={}, quorum_commit_index={}, unhandled_commits_threshold={}, fetch_after_index={}",
266                self.inner.sync_type.as_str(),
267                self.synced_commit_index,
268                highest_handled_index,
269                highest_scheduled_index,
270                quorum_commit_index,
271                unhandled_commits_threshold,
272                fetch_after_index,
273            );
274
275            // Schedule commit ranges for fetching using shared helper
276            let schedule_result = schedule_commit_ranges(
277                &self.inner,
278                fetch_after_index,
279                quorum_commit_index,
280                highest_handled_index,
281                unhandled_commits_threshold,
282            );
283
284            // Add scheduled ranges to pending fetches
285            for range in schedule_result.ranges_scheduled {
286                debug!(
287                    "[{}] Scheduling fetch for commit range {}..={}",
288                    self.inner.sync_type.as_str(),
289                    range.start(),
290                    range.end()
291                );
292                self.pending_fetches.insert(range);
293            }
294
295            // Update highest scheduled index
296            if let Some(new_highest) = schedule_result.new_highest_scheduled {
297                self.highest_scheduled_index = Some(new_highest);
298            }
299        }
300
301        // Detect close-to-quorum mode: when remaining gap is less than a full batch.
302        // Only activate if we've actually fetched data during this fast sync session.
303        //
304        // When close_to_quorum_mode is activated, the schedule_loop() will:
305        // 1. Wait for all inflight/pending fetches to complete
306        // 2. Fetch block headers for ~cached_rounds worth of commits
307        // 3. Send ReinitializeComponents to core thread to properly initialize DAG
308        //    state
309        // 4. Reset fast sync state so regular syncer can take over
310        if self.has_fetched_data && !self.close_to_quorum_mode {
311            let current_fetch_after = self
312                .synced_commit_index
313                .max(self.highest_scheduled_index.unwrap_or(0));
314            let remaining_gap = quorum_commit_index.saturating_sub(current_fetch_after);
315            if remaining_gap > 0 && remaining_gap < step {
316                let range_start = current_fetch_after + 1;
317                let range_end = quorum_commit_index;
318                debug!(
319                    "[{}] Scheduling final partial fetch for commit range {}..={} (remaining_gap={})",
320                    self.inner.sync_type.as_str(),
321                    range_start,
322                    range_end,
323                    remaining_gap
324                );
325                self.pending_fetches
326                    .insert((range_start..=range_end).into());
327                self.highest_scheduled_index = Some(range_end);
328            }
329            if remaining_gap < step {
330                self.close_to_quorum_mode = true;
331                info!(
332                    "[{}] Entering close-to-quorum mode: remaining_gap={}, step={}",
333                    self.inner.sync_type.as_str(),
334                    remaining_gap,
335                    step
336                );
337            }
338        }
339    }

341    async fn handle_fetch_result(&mut self, target_end: CommitIndex, output: FastSyncOutput) {
342        assert!(!output.committed_subdags.is_empty());
343
344        // Track that we have actually fetched data during this fast sync session.
345        self.has_fetched_data = true;
346
347        let total_transactions_size_bytes = output
348            .committed_subdags
349            .iter()
350            .flat_map(|subdag| &subdag.transactions)
351            .map(|txns| txns.serialized().len() as u64)
352            .sum();
353
354        let metrics = &self.inner.context.metrics.node_metrics;
355        let sync_label = self.inner.sync_type.as_str();
356        metrics
357            .commit_sync_fetched_commits
358            .with_label_values(&[sync_label])
359            .inc_by(output.committed_subdags.len() as u64);
360        metrics
361            .commit_sync_total_fetched_transactions_size
362            .with_label_values(&[sync_label])
363            .inc_by(total_transactions_size_bytes);
364
365        let (commit_start, commit_end) = (
366            output.committed_subdags.first().unwrap().commit_ref.index,
367            output.committed_subdags.last().unwrap().commit_ref.index,
368        );
369        self.highest_fetched_commit_index = self.highest_fetched_commit_index.max(commit_end);
370        metrics
371            .commit_sync_highest_fetched_index
372            .with_label_values(&[sync_label])
373            .set(self.highest_fetched_commit_index as i64);
374
375        // Allow returning partial results and try fetching the rest separately.
376        requeue_partial_range(&mut self.pending_fetches, commit_end, target_end);
377        // Make sure the synced_commit_index is up to date.
378        self.synced_commit_index = self
379            .synced_commit_index
380            .max(self.inner.dag_state.read().last_solid_commit_index());
381        // Only add new blocks if at least some of them are not already synced.
382        if self.synced_commit_index < commit_end {
383            self.fetched_ranges
384                .insert((commit_start..=commit_end).into(), output);
385        }
386        // Try to process as many fetched blocks as possible.
387        while let Some((fetched_commit_range, _)) = self.fetched_ranges.first_key_value() {
388            // Only pop fetched_ranges if there is no gap with blocks already synced.
389            // Note: start, end and synced_commit_index are all inclusive.
390            let (fetched_commit_range, output) =
391                if fetched_commit_range.start() <= self.synced_commit_index + 1 {
392                    self.fetched_ranges.pop_first().unwrap()
393                } else {
394                    // Found a gap between the earliest fetched block and the latest synced block,
395                    // so not sending additional blocks to Core.
396                    metrics
397                        .commit_sync_gap_on_processing
398                        .with_label_values(&[sync_label])
399                        .inc();
400                    break;
401                };
402            // Avoid sending to Core a whole batch of already synced blocks.
403            if fetched_commit_range.end() <= self.synced_commit_index {
404                continue;
405            }
406
407            debug!(
408                "[{}] Fetched {} subdags with transactions for commit range {:?}",
409                sync_label,
410                output.committed_subdags.len(),
411                fetched_commit_range,
412            );
413
414            // If the core thread cannot handle the incoming blocks, it is ok to block here.
415            if let Err(e) = self
416                .inner
417                .core_thread_dispatcher
418                .add_subdags_from_fast_sync(output.clone())
419                .await
420            {
421                info!(
422                    "[{}] Failed to dispatch subdags to core, shutting down: {}",
423                    sync_label, e
424                );
425                return;
426            }
427
428            // Once subdags are sent to Core, ratchet up synced_commit_index
429            self.synced_commit_index = self.synced_commit_index.max(fetched_commit_range.end());
430        }
431
432        metrics
433            .commit_sync_inflight_fetches
434            .with_label_values(&[sync_label])
435            .set(self.inflight_fetches.len() as i64);
436        metrics
437            .commit_sync_pending_fetches
438            .with_label_values(&[sync_label])
439            .set(self.pending_fetches.len() as i64);
440        metrics
441            .commit_sync_highest_synced_index
442            .with_label_values(&[sync_label])
443            .set(self.synced_commit_index as i64);
444    }

446    fn try_start_fetches(&mut self) {
447        let inner = self.inner.clone();
448        shared_try_start_fetches(
449            &self.inner,
450            &mut self.pending_fetches,
451            self.fetched_ranges.len(),
452            self.inflight_fetches.len(),
453            self.synced_commit_index,
454            |commit_range| {
455                self.inflight_fetches
456                    .spawn(Self::fetch_loop(inner.clone(), commit_range));
457            },
458        );
459    }

461    // Retries fetching commits and block headers from available authorities, until
462    // a request succeeds where at least a prefix of the commit range is
463    // fetched. Returns the fetched commits and block headers referenced by the
464    // commits.
465    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %inner.context.own_index)))]
466    async fn fetch_loop(
467        inner: Arc<Inner<C>>,
468        commit_range: CommitRange,
469    ) -> (CommitIndex, FastSyncOutput) {
470        shared_fetch_loop(inner, commit_range, 2, Self::fetch_once).await
471    }

473    // Fetches commits and transactions from a single authority.
474    async fn fetch_once(
475        inner: Arc<Inner<C>>,
476        target_authority: AuthorityIndex,
477        commit_range: CommitRange,
478        timeout: Duration,
479    ) -> ConsensusResult<FastSyncOutput> {
480        let _timer = inner
481            .context
482            .metrics
483            .node_metrics
484            .commit_sync_fetch_once_latency
485            .with_label_values(&[inner.sync_type.as_str()])
486            .start_timer();
487        assert!(inner.context.protocol_config.consensus_fast_commit_sync());
488
489        // 1. Fetch commits, voting headers, and transactions in the commit range from
490        //    the target authority. Each transaction is serialized as
491        //    SerializedTransactionsV2 which includes the TransactionRef.
492        let (serialized_commits, serialized_proof_for_last_commit, serialized_transactions) = inner
493            .network_client
494            .fetch_commits_and_transactions(target_authority, commit_range.clone(), timeout)
495            .await?;
496
497        // 2. Verify the response contains block headers that can certify the last
498        //    returned commit, and the returned commits are chained by digest,
499        // so earlier commits are certified as well.
500        let batch_size = inner.sync_type.commit_sync_batch_size(&inner.context) as usize;
501        let (commits, voting_block_headers) = Handle::current()
502            .spawn_blocking({
503                let inner = inner.clone();
504                move || {
505                    inner.verify_commits(
506                        target_authority,
507                        commit_range,
508                        serialized_commits,
509                        serialized_proof_for_last_commit,
510                        2 * batch_size,
511                    )
512                }
513            })
514            .await
515            .expect("Spawn blocking should not fail")?;
516
517        // 3. Collect all committed transaction block refs from commits
518        let mut committed_tx_refs: BTreeSet<TransactionRef> = commits
519            .iter()
520            .flat_map(|c| c.committed_transactions())
521            .filter_map(|gen_tr_ref| gen_tr_ref.expect_transaction_ref().ok())
522            .collect();
523
524        // 4. Process fetched transactions. Each serialized_transaction is a
525        //    SerializedTransactionsV2 containing both the TransactionRef and the actual
526        //    transaction data.
527        let mut fetched_transactions = BTreeMap::new();
528        for serialized_transaction in serialized_transactions {
529            if let Ok(tx_v2) = bcs::from_bytes::<SerializedTransactionsV2>(&serialized_transaction)
530            {
531                let transaction_ref = tx_v2.transaction_ref;
532                if !committed_tx_refs.contains(&transaction_ref) {
533                    return Err(ConsensusError::UnexpectedTransactionForCommit {
534                        peer: target_authority,
535                        received: GenericTransactionRef::TransactionRef(transaction_ref),
536                    });
537                }
538                fetched_transactions.insert(
539                    GenericTransactionRef::TransactionRef(transaction_ref),
540                    tx_v2.serialized_transactions,
541                );
542                committed_tx_refs.remove(&transaction_ref);
543            } else {
544                debug!(
545                    "[{}] Failed to deserialize SerializedTransactionsV2: {:?}",
546                    inner.sync_type.as_str(),
547                    serialized_transaction
548                );
549                continue;
550            }
551        }
552
553        // Check if any committed transactions were not fetched (committed_tx_refs
554        // should be empty now)
555        if !committed_tx_refs.is_empty() {
556            // TODO: create subdags for prefix of commits
557            return Err(ConsensusError::FetchedTransactionsMismatch {
558                peer: target_authority,
559                expected: committed_tx_refs.len() + fetched_transactions.len(),
560                received: fetched_transactions.len(),
561            });
562        }
563
564        // 5. Verify transactions
565        let mut transactions_map = if !fetched_transactions.is_empty() {
566            Handle::current()
567                .spawn_blocking({
568                    let context = inner.context.clone();
569
570                    move || {
571                        verify_transactions_with_transactions_refs(
572                            &context,
573                            target_authority,
574                            fetched_transactions,
575                        )
576                    }
577                })
578                .await
579                .expect("Spawn blocking should not fail")?
580        } else {
581            BTreeMap::new()
582        };
583
584        // 6. Now create the CommittedSubDags with the fetched transactions.
585        // For fast commit sync, we use block headers refs and reputation scores from
586        // the commit.
587        let mut committed_subdags = Vec::new();
588        for commit in &commits {
589            // Get block headers from the commit
590            let committed_header_refs = commit.block_headers().to_vec();
591
592            // Get reputation scores from the commit
593            let reputation_scores = commit.reputation_scores().to_vec();
594
595            // Collect transactions for this commit
596            let commit_transactions: Vec<VerifiedTransactions> = commit
597                .committed_transactions()
598                .iter()
599                .filter_map(|tx_ref| transactions_map.remove(tx_ref))
600                .collect();
601
602            committed_subdags.push(CommittedSubDag::new(
603                commit.leader(),
604                vec![], // headers - VerifiedBlockHeader, we don't have these in fast sync
605                committed_header_refs,
606                commit_transactions,
607                commit.timestamp_ms(),
608                commit.reference(),
609                reputation_scores,
610            ));
611        }
612
613        Ok(FastSyncOutput {
614            commits,
615            committed_subdags,
616            voting_block_headers,
617        })
618    }

620    /// Fetches block headers needed for component reinitialization from the
621    /// network. This is called when close_to_quorum mode is active and all
622    /// pending fetches complete. Fetches headers for the maximum of
623    /// cached_rounds, gc_depth * 2, leader_schedule_window, and
624    /// commits_since_schedule_update to satisfy DagState cache, linearizer,
625    /// and leader schedule recovery requirements.
626    async fn fetch_headers_for_reinitialization(
627        inner: Arc<Inner<C>>,
628    ) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
629        // We need headers for three purposes:
630        // 1. DagState cache: at least cached_rounds commits back
631        // 2. Linearizer recovery: at least gc_depth * 2 commits back
632        // 3. Leader schedule recovery: at least leader_schedule_window commits back, or
633        //    all commits since the last stored commit info
634        //    (commits_since_schedule_update)
635        // Fetch the maximum to satisfy all requirements
636        let cached_rounds = inner.context.parameters.dag_state_cached_rounds;
637        let gc_depth = inner.context.protocol_config.gc_depth();
638        let leader_schedule_window = crate::leader_schedule::CONSENSUS_COMMITS_PER_SCHEDULE as u32;
639        // Get block refs from recent commits stored during fast sync
640        // TODO: The commits might not yet stored, but only fetched and pending
641        // processing.
642        let (commits_since_schedule_update, block_refs) = {
643            let dag_state = inner.dag_state.read();
644            let last_commit_index = dag_state.last_commit_index();
645            let last_commit_info_index = dag_state.last_commit_info_index();
646            let commits_since_schedule_update =
647                last_commit_index.saturating_sub(last_commit_info_index);
648            let num_commits = max(
649                commits_since_schedule_update,
650                max(leader_schedule_window, max(cached_rounds, gc_depth * 2)),
651            );
652            let block_refs = dag_state.get_block_refs_for_recent_commits(num_commits);
653            (commits_since_schedule_update, block_refs)
654        };
655
656        let max_headers_per_fetch = inner.context.parameters.max_headers_per_commit_sync_fetch;
657
658        info!(
659            "[{}] Fetching {} block headers for reinitialization (cached_rounds={}, gc_depth*2={}, leader_schedule_window={}, commits_since_schedule_update={})",
660            inner.sync_type.as_str(),
661            block_refs.len(),
662            cached_rounds,
663            gc_depth * 2,
664            leader_schedule_window,
665            commits_since_schedule_update
666        );
667
668        // Shuffle target authorities for load balancing
669        #[cfg_attr(test, expect(unused_mut))]
670        let mut target_authorities: Vec<_> = inner
671            .context
672            .committee
673            .authorities()
674            .filter_map(|(i, _)| {
675                if i != inner.context.own_index {
676                    Some(i)
677                } else {
678                    None
679                }
680            })
681            .collect();
682        #[cfg(not(test))]
683        target_authorities.shuffle(&mut ThreadRng::default());
684
685        // Fetch headers in chunks to avoid overwhelming the network
686        let mut all_headers = Vec::new();
687        for chunk in block_refs.chunks(max_headers_per_fetch) {
688            let chunk_refs: Vec<_> = chunk.to_vec();
689
690            // Try fetching from different authorities until successful
691            let mut fetched = false;
692            for &authority in &target_authorities {
693                match tokio::time::timeout(
694                    FETCH_HEADERS_TIMEOUT,
695                    inner.network_client.fetch_block_headers(
696                        authority,
697                        chunk_refs.clone(),
698                        vec![],
699                        FETCH_HEADERS_TIMEOUT,
700                    ),
701                )
702                .await
703                {
704                    Ok(Ok(serialized_headers)) => {
705                        // Verify headers match requested refs
706                        match verify_fetched_headers(authority, &chunk_refs, serialized_headers) {
707                            Ok(headers) => {
708                                info!(
709                                    "[{}] Fetched {} headers from authority {}",
710                                    inner.sync_type.as_str(),
711                                    headers.len(),
712                                    authority
713                                );
714                                all_headers.extend(headers);
715                                fetched = true;
716                                break;
717                            }
718                            Err(e) => {
719                                warn!(
720                                    "[{}] Failed to verify headers from {}: {}",
721                                    inner.sync_type.as_str(),
722                                    authority,
723                                    e
724                                );
725                            }
726                        }
727                    }
728                    Ok(Err(e)) => {
729                        warn!(
730                            "[{}] Failed to fetch headers from {}: {}",
731                            inner.sync_type.as_str(),
732                            authority,
733                            e
734                        );
735                    }
736                    Err(_) => {
737                        warn!(
738                            "[{}] Timed out fetching headers from {}",
739                            inner.sync_type.as_str(),
740                            authority
741                        );
742                    }
743                }
744            }
745
746            if !fetched {
747                return Err(ConsensusError::FailedToFetchBlockHeaders {
748                    num_requested: chunk_refs.len(),
749                });
750            }
751        }
752
753        info!(
754            "[{}] Successfully fetched {} total block headers for reinitialization",
755            inner.sync_type.as_str(),
756            all_headers.len()
757        );
758
759        Ok(all_headers)
760    }

1238        let blocked_validator_index: usize = 1;
1239
1240        // Phase 1: Start all authorities and let them create initial commits.
1241        // Disable fast commit syncer for the test validator so it won't resolve
1242        // pending subdags during Phase 3 (the fast syncer uses last_solid_commit_index
1243        // for gap detection, which would trigger fetching when pending subdags exist).
1244        // Phase 6 restarts the test validator with enable_fast_commit_syncer: true.
1245        for (index, _) in committee.authorities() {
1246            let parameters = Parameters {
1247                db_path: temp_dirs[index.value()].path().to_path_buf(),
1248                dag_state_cached_rounds: 5,
1249                commit_sync_parallel_fetches: 2,
1250                commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
1251                commit_sync_gap_threshold: COMMIT_GAP_THRESHOLD,
1252                fast_commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
1253                enable_fast_commit_syncer: index.value() != test_validator_index,
1254                sync_last_known_own_block_timeout: Duration::from_millis(2_000),
1255                ..Default::default()
1256            };
1257            let (authority, receiver, monitor) = make_authority_with_params(
1258                index,
1259                &temp_dirs[index.value()],
1260                committee.clone(),
1261                keypairs.clone(),
1262                boot_counters[index],
1263                protocol_config.clone(),
1264                parameters,
1265                0,
1266            )
1267            .await;
1268            boot_counters[index] += 1;
1269            authorities.push(authority);
1270            output_receivers.push(receiver);
1271            consumer_monitors.push(monitor);
1272        }
1273
1274        let mut txn_counter = 0u64;
1275        let start_time = Instant::now();
1276        let mut committed_index = [0u32; NUM_AUTHORITIES];
1277        while start_time.elapsed() < stable_work_duration {
1278            // Submit transactions to all validators (rotating)
1279            let authority_index = txn_counter as usize % authorities.len();
1280            let txn = vec![txn_counter as u8; 16];
1281            authorities[authority_index]
1282                .transaction_client()
1283                .submit(vec![txn])
1284                .await
1285                .unwrap();
1286            txn_counter += 1;
1287
1288            for (index, receiver) in output_receivers.iter_mut().enumerate() {
1289                while let Ok(committed_subdag) = receiver.try_recv() {
1290                    let commit_index = committed_subdag.commit_ref.index;
1291                    assert!(
1292                        commit_index > committed_index[index],
1293                        "Commit index {} should be greater than previous {}",
1294                        commit_index,
1295                        committed_index[index]
1296                    );
1297                    committed_index[index] = commit_index;
1298                    consumer_monitors[index].set_highest_handled_commit(commit_index);
1299                }
1300            }
1301            sleep(Duration::from_millis(50)).await;
1302        }
1303
1304        // Phase 2: Dynamically unsubscribe from validator 1 + stop txn synchronizer +
1305        // stop shard reconstructor This will create pending subdags as headers
1306        // arrive via cordial dissemination but transactions from validator 1's
1307        // blocks are missing and shards cannot be reconstructed. Commit syncers
1308        // won't activate during Phase 3 because there's no commit gap - the
1309        // validator keeps up with commits, just missing transactions.
1310        authorities[test_validator_index].unsubscribe_from_peer_for_test(
1311            committee
1312                .to_authority_index(blocked_validator_index)
1313                .unwrap(),
1314        );
1315        authorities[test_validator_index]
1316            .stop_transactions_synchronizer_for_test()
1317            .await
1318            .expect("Transaction synchronizer should stop");
1319        authorities[test_validator_index]
1320            .stop_shard_reconstructor_for_test()
1321            .await
1322            .expect("Shard reconstructor should stop");
1323
1324        // Phase 3: Wait for headers to arrive via cordial dissemination and commits to
1325        // be created. Submit transactions to all validators (rotating).
1326        // This should create pending subdags (gap between
1327        // last_commit and last_solid_commit_leader_round)
1328
1329        // Track commits before the wait
1330        let commits_before = committed_index[test_validator_index];
1331
1332        // Submit transactions to all validators during Phase 3.
1333        // Validator 0 can process transactions from itself and validators 2 & 3.
1334        // However, validator 0 can't fetch transactions from validator 1's blocks
1335        // because:
1336        // - It's unsubscribed from validator 1
1337        // - Transaction synchronizer is stopped (blocks active transaction fetching)
1338        // - Shard reconstructor is stopped (blocks erasure-coded shard reconstruction)
1339        // This creates pending subdags.
1340        // The fast commit syncer is disabled for the test validator (see Phase 1).
1341        let phase3_start = Instant::now();
1342        while phase3_start.elapsed() < stable_work_duration {
1343            // Submit transactions to all validators (rotating)
1344            let authority_index = txn_counter as usize % authorities.len();
1345            let txn = vec![txn_counter as u8; 16];
1346            authorities[authority_index]
1347                .transaction_client()
1348                .submit(vec![txn])
1349                .await
1350                .unwrap();
1351            txn_counter += 1;
1352
1353            // Drain receivers
1354            for (index, receiver) in output_receivers.iter_mut().enumerate() {
1355                while let Ok(committed_subdag) = receiver.try_recv() {
1356                    let commit_index = committed_subdag.commit_ref.index;
1357                    if commit_index > committed_index[index] {
1358                        committed_index[index] = commit_index;
1359                        consumer_monitors[index].set_highest_handled_commit(commit_index);
1360                    }
1361                }
1362            }
1363            sleep(Duration::from_millis(50)).await;
1364        }
1365
1366        let commits_after = committed_index[test_validator_index];
1367        let new_commits = commits_after - commits_before;
1368        assert!(
1369            new_commits > 0,
1370            "Expected new commits during Phase 3, got 0"
1371        );
1372
1373        // Verify pending subdags gap exists
1374        let dag_state = authorities[test_validator_index].dag_state_for_test();
1375        let last_commit = dag_state.read().last_commit_round();
1376        let last_solid = dag_state.read().last_solid_commit_leader_round();
1377        let dag_round = dag_state.read().threshold_clock_round();
1378
1379        // Verify gap exists - we expect pending subdags
1380        let has_gap = last_commit > last_solid.unwrap_or(0);
1381        assert!(
1382            has_gap,
1383            "Expected pending subdags gap: last_commit={}, last_solid={:?}. \
1384             DAG round={}, new_commits={}. \
1385             Validator 1's new blocks should have missing transactions.",
1386            last_commit, last_solid, dag_round, new_commits
1387        );
1388
1389        // Record where the validator is now (with pending subdags)
1390        let last_processed_with_pending =
1391            consumer_monitors[test_validator_index].highest_handled_commit();
1392
1393        // Phase 4: Stop test validator (preserves pending subdags to disk)
1394        authorities.remove(test_validator_index).stop().await;
1395
1396        // Phase 5: Let other validators continue while the test validator is stopped
1397        // (creates fast sync gap > threshold)
1398        let start_time = Instant::now();
1399        while start_time.elapsed() < stable_work_duration * 2 {
1400            for (index, receiver) in output_receivers.iter_mut().enumerate() {
1401                if index == test_validator_index {
1402                    continue; // Skip stopped validator
1403                }
1404                while let Ok(committed_subdag) = receiver.try_recv() {
1405                    let commit_index = committed_subdag.commit_ref.index;
1406                    if commit_index > committed_index[index] {
1407                        committed_index[index] = commit_index;
1408                        consumer_monitors[index].set_highest_handled_commit(commit_index);
1409                    }
1410                }
1411            }
1412            sleep(Duration::from_millis(50)).await;
1413        }
1414
1415        let max_other = consumer_monitors
1416            .iter()
1417            .enumerate()
1418            .filter(|(i, _)| *i != test_validator_index)
1419            .map(|(_, m)| m.highest_handled_commit())
1420            .max()
1421            .unwrap_or(0);
1422
1423        let gap = max_other.saturating_sub(last_processed_with_pending);
1424        assert!(
1425            gap > COMMIT_GAP_THRESHOLD,
1426            "Gap {} should be greater than threshold {}",
1427            gap,
1428            COMMIT_GAP_THRESHOLD
1429        );
1430
1431        // Phase 6: Restart test validator with full connectivity and fast commit
1432        // syncer enabled.
1433        let parameters = Parameters {
1434            db_path: temp_dirs[test_validator_index].path().to_path_buf(),
1435            dag_state_cached_rounds: 5,
1436            commit_sync_parallel_fetches: 2,
1437            commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
1438            commit_sync_gap_threshold: COMMIT_GAP_THRESHOLD,
1439            fast_commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
1440            sync_last_known_own_block_timeout: Duration::from_millis(2_000),
1441            enable_fast_commit_syncer: true,
1442            ..Default::default()
1443        };
1444        let (authority, receiver, monitor) = make_authority_with_params(
1445            committee.to_authority_index(test_validator_index).unwrap(),
1446            &temp_dirs[test_validator_index],
1447            committee.clone(),
1448            keypairs.clone(),
1449            boot_counters[test_validator_index],
1450            protocol_config.clone(),
1451            parameters,
1452            last_processed_with_pending,
1453        )
1454        .await;
1455        output_receivers[test_validator_index] = receiver;
1456        consumer_monitors[test_validator_index] = monitor;
1457        authorities.insert(test_validator_index, authority);
1458
1459        // Phase 7: Wait for the validator to catch up via fast sync
1460        let start_time = Instant::now();
1461        let mut caught_up = false;
1462        while start_time.elapsed() < Duration::from_secs(60) {
1463            for (index, receiver) in output_receivers.iter_mut().enumerate() {
1464                while let Ok(committed_subdag) = receiver.try_recv() {
1465                    let commit_index = committed_subdag.commit_ref.index;
1466                    if commit_index > committed_index[index] {
1467                        committed_index[index] = commit_index;
1468                        consumer_monitors[index].set_highest_handled_commit(commit_index);
1469                    }
1470                }
1471            }
1472
1473            let test_index = consumer_monitors[test_validator_index].highest_handled_commit();
1474            let max_other = consumer_monitors
1475                .iter()
1476                .enumerate()
1477                .filter(|(i, _)| *i != test_validator_index)
1478                .map(|(_, m)| m.highest_handled_commit())
1479                .max()
1480                .unwrap_or(0);
1481
1482            if test_index > last_processed_with_pending && test_index + 20 >= max_other {
1483                caught_up = true;
1484                break;
1485            }
1486            sleep(Duration::from_millis(50)).await;
1487        }
1488
1489        assert!(
1490            caught_up,
1491            "Validator {} should have caught up via fast sync",
1492            test_validator_index
1493        );
1494
1495        // Verify the validator progressed significantly after restart with pending
1496        // subdags
1497        let final_index = consumer_monitors[test_validator_index].highest_handled_commit();
1498        assert!(
1499            final_index > last_processed_with_pending,
1500            "Validator should have progressed after restart: with_pending_subdags={}, final={}",
1501            last_processed_with_pending,
1502            final_index
1503        );
1504
1505        // Verify that fast sync was actually used by checking the fetched commits
1506        // metric
1507        let commit_sync_fetched_commits: u64 = authorities
1508            .iter()
1509            .map(|a| {
1510                a.context()
1511                    .metrics
1512                    .node_metrics
1513                    .commit_sync_fetched_commits
1514                    .with_label_values(&["fast_commit_sync"])
1515                    .get()
1516            })
1517            .sum();
1518
1519        assert!(
1520            commit_sync_fetched_commits > 0,
1521            "Expected commits fetched via fast sync > 0, got {}",
1522            commit_sync_fetched_commits
1523        );
1524
1525        // Stop all authorities
1526        for authority in authorities {
1527            authority.stop().await;
1528        }
1529    }
