15struct SuspendedBlockHeader {
16    block_header: VerifiedBlockHeader,
17    missing_ancestors: BTreeSet<BlockRef>,
18    timestamp: Instant,
19}
20
21impl SuspendedBlockHeader {
22    fn new(block: VerifiedBlockHeader, missing_ancestors: &BTreeSet<BlockRef>) -> Self {
23        let mut ma = BTreeSet::new();
24        for ancestor in missing_ancestors {
25            ma.insert(*ancestor);
26        }
27        Self {
28            block_header: block,
29            missing_ancestors: ma,
30            timestamp: Instant::now(),
31        }
32    }
33}

35pub(crate) struct BlockSuspender {
36    context: Arc<Context>,
37    /// Keeps all the suspended block headers. A suspended block header is a
38    /// header that is missing part of its causal history and thus can't be
39    /// immediately processed. A block header will remain in this map until
40    /// all its causal history has been successfully processed.
41    suspended_headers: BTreeMap<BlockRef, SuspendedBlockHeader>,
42    /// A map that keeps all the blocks that we are missing (keys) and the
43    /// corresponding blocks that reference the missing blocks as ancestors
44    /// and need them to get unsuspended. It is possible for a missing
45    /// dependency (key) to be a suspended block, so the block has been
46    /// already fetched but itself is still missing some of its ancestors to be
47    /// processed.
48    missing_ancestors: BTreeMap<BlockRef, BTreeSet<BlockRef>>,
49    /// A map of blocks that need to be fetched to the set of authorities
50    /// expected to have them available locally. This set is approximated
51    /// based on the block's author and the authors of its direct children.
52    /// A block is considered missing if it appears in `missing_ancestors`
53    /// and has not yet been fetched. Blocks already stored or present in
54    /// `suspended_headers` are excluded.
55    headers_to_fetch: BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
56}
57
58impl BlockSuspender {
59    pub(crate) fn new(context: Arc<Context>) -> Self {
60        Self {
61            context,
62            suspended_headers: BTreeMap::new(),
63            missing_ancestors: BTreeMap::new(),
64            headers_to_fetch: BTreeMap::new(),
65        }
66    }
67
68    /// Reinitialize BlockSuspender after fast sync completes.
69    /// Clears all suspended blocks, missing ancestors, and fetch queues.
70    pub(crate) fn reinitialize(&mut self) {
71        self.suspended_headers.clear();
72        self.missing_ancestors.clear();
73        self.headers_to_fetch.clear();
74    }
75
76    /// Accept or suspend a batch of received block headers based on their
77    /// missing ancestors.
78    ///
79    /// Each header is either:
80    /// - Accepted immediately if all its ancestors are resolved (directly or
81    ///   through this batch),
82    /// - Or suspended if it still has missing or unresolved ancestors.
83    ///
84    /// After initially processing the batch, this function also recursively
85    /// unsuspends any other blocks that were previously suspended and are
86    /// now unblocked due to newly accepted ancestors.
87    ///
88    /// # Arguments
89    /// * `missing_ancestors_map` - A map of verified block headers to the set
90    ///   of their unresolved ancestor references as read from the DagState by
91    ///   the caller.
92    ///
93    /// # Returns
94    /// * `Vec<VerifiedBlockHeader>` — All headers that were accepted, either
95    ///   directly or via recursive unsuspension.
96    /// * `BTreeSet<BlockRef>` — Set of ancestor block references that are still
97    ///   missing and should be fetched.
98    ///
99    /// # Side Effects
100    /// - Updates the suspension state of blocks.
101    /// - May recursively unsuspend dependent blocks.
102    /// - Updates internal fetch statistics.
103    pub(crate) fn accept_or_suspend_received_headers(
104        &mut self,
105        missing_ancestors_map: BTreeMap<VerifiedBlockHeader, BTreeSet<BlockRef>>,
106    ) -> (Vec<VerifiedBlockHeader>, BTreeSet<BlockRef>) {
107        // Decide which headers are resolved now, which need to be suspended, and which
108        // ancestors must still be fetched
109        let (fully_resolved_headers, ancestors_to_fetch) =
110            self.resolve_or_suspend_headers(missing_ancestors_map);
111        let mut accepted_headers = vec![];
112
113        for fully_resolved_header in fully_resolved_headers {
114            let fully_resolved_header_reference = fully_resolved_header.reference();
115            // Accept the resolved header
116            accepted_headers.push(fully_resolved_header);
117            // Recursively attempt to unsuspend any blocks that were waiting on this one
118            let unsuspended =
119                self.recursively_unsuspend_dependents(fully_resolved_header_reference);
120            // Add them to the list of accepted headers
121            accepted_headers.extend(unsuspended);
122        }
123        self.update_stats(ancestors_to_fetch.len() as u64);
124        (accepted_headers, ancestors_to_fetch)
125    }
126    /// Processes a batch of verified block headers and updates their suspension
127    /// state based on missing ancestors.
128    ///
129    /// This function evaluates each incoming block to determine whether it can
130    /// be accepted immediately (i.e., all its ancestors are resolved), or
131    /// if it must be suspended due to unresolved or missing ancestors.
132    /// Suspended blocks are tracked, and any additional ancestor blocks that
133    /// need to be fetched are identified.
134    ///
135    /// # Arguments
136    /// * `missing_ancestors_map` - A map from the incoming verified block
137    ///   headers to the set of their missing ancestor references as read from
138    ///   the DagState by the calling code in BlockManager.
139    ///
140    /// # Returns
141    /// A tuple containing:
142    /// * `Vec<VerifiedBlockHeader>` - List of block headers that are fully
143    ///   resolved and ready for further processing to check if their children
144    ///   can be unsuspended.
145    /// * `BTreeSet<BlockRef>` - Set of ancestor block references that still
146    ///   need to be fetched.
147    ///
148    /// # Side Effects
149    /// This function:
150    /// - Removes incoming batch block references from `self.missing_blocks`.
151    /// - Registers suspended blocks and their missing ancestors in internal
152    ///   tracking structures.
153    /// - Triggers reporting mechanisms for suspended blocks and missing
154    ///   ancestors.
155    /// - Identifies which ancestor blocks must still be fetched externally.
156    fn resolve_or_suspend_headers(
157        &mut self,
158        missing_ancestors_map: BTreeMap<VerifiedBlockHeader, BTreeSet<BlockRef>>,
159    ) -> (Vec<VerifiedBlockHeader>, BTreeSet<BlockRef>) {
160        // Collect references of all incoming blocks in this batch
161        let incoming_block_refs: BTreeSet<BlockRef> = missing_ancestors_map
162            .keys()
163            .map(|b| b.reference())
164            .collect();
165        debug!(
166            "Trying to accept block headers: {}",
167            incoming_block_refs.iter().map(|b| b.to_string()).join(",")
168        );
169        let mut fully_resolved_headers: Vec<VerifiedBlockHeader> = vec![];
170        let mut ancestors_to_fetch = BTreeSet::new();
171        for (incoming_block_header, missing_ancestors) in missing_ancestors_map {
172            let block_ref = incoming_block_header.reference();
173            let block_header_author = incoming_block_header.author();
174
175            // We're now processing this block, so remove it from the missing_blocks list
176            self.headers_to_fetch.remove(&block_ref);
177
178            // If there are no missing ancestors, we can mark the block header as fully
179            // resolved
180            if missing_ancestors.is_empty() {
181                fully_resolved_headers.push(incoming_block_header);
182                continue;
183            }
184
185            // Otherwise, suspend the block and track its missing ancestors
186            self.register_suspended_block(incoming_block_header, &missing_ancestors);
187            self.report_suspended_block(&block_ref);
188            for ancestor in missing_ancestors {
189                self.register_missing_ancestor(block_ref, ancestor);
190                self.report_missing_ancestor(&ancestor);
191
192                // If this ancestor is not already suspended and is not part of the incoming
193                // batch, we may need to fetch it
194                if !self.suspended_headers.contains_key(&ancestor)
195                    && !incoming_block_refs.contains(&ancestor)
196                {
197                    ancestors_to_fetch.insert(ancestor);
198                    // Only report the block as missing if we’re not already fetching it
199                    if !self.headers_to_fetch.contains_key(&ancestor) {
200                        self.report_block_to_fetch(ancestor.author);
201                    }
202                    self.register_block_to_fetch(&ancestor, block_header_author);
203                }
204            }
205        }
206        (fully_resolved_headers, ancestors_to_fetch)
207    }
208    /// Recursively unsuspends all blocks that were dependent on a now-accepted
209    /// block.
210    ///
211    /// Starting from `resolved_block`, this function walks the dependency graph
212    /// and attempts to unsuspend any suspended blocks that were blocked on
213    /// it. Successfully unsuspended blocks are added to the result, and
214    /// their own dependents are checked in turn.
215    ///
216    /// # Arguments
217    /// * `resolved_block_ref` - The block that was just accepted and may
218    ///   unsuspend other blocks.
219    ///
220    /// # Returns
221    /// * A list of verified block headers that have been unsuspended.
222    fn recursively_unsuspend_dependents(
223        &mut self,
224        resolved_block_ref: BlockRef,
225    ) -> Vec<VerifiedBlockHeader> {
226        let mut ready_blocks = vec![];
227        let mut stack = vec![resolved_block_ref];
228
229        while let Some(popped_resolved_block_ref) = stack.pop() {
230            // And try to check if its direct children can be unsuspended
231            if let Some(suspended_block_refs_with_missing_deps) =
232                self.missing_ancestors.remove(&popped_resolved_block_ref)
233            {
234                for suspended_block_ref in suspended_block_refs_with_missing_deps {
235                    // For each dependency try to unsuspend it. If that's successful then we add it
236                    // to the stack so we can recursively try to unsuspend its
237                    // children.
238                    if let Some(unsuspended_block) =
239                        self.unsuspend_if_ready(&suspended_block_ref, &popped_resolved_block_ref)
240                    {
241                        self.report_unsuspended_block(&unsuspended_block);
242                        stack.push(unsuspended_block.block_header.reference());
243                        ready_blocks.push(unsuspended_block.block_header);
244                    }
245                }
246            }
247        }
248        ready_blocks
249    }
250    /// Attempts to unsuspend a block if one of its missing dependencies has
251    /// just been accepted.
252    ///
253    /// This function is called when a block that was previously suspended (due
254    /// to missing ancestors) may now be unblocked because one of its
255    /// dependencies (`resolved_dependency`) has been resolved.
256    /// The dependency is removed from the suspended block’s set of missing
257    /// ancestors. If this was the final missing dependency, the block is
258    /// removed from the suspension map and returned.
259    ///
260    /// # Arguments
261    /// * `suspended_block_ref` - Reference to the suspended block being
262    ///   re-evaluated.
263    /// * `resolved_dependency` - A block reference that was just accepted and
264    ///   may resolve a dependency.
265    ///
266    /// # Returns
267    /// * `Some(SuspendedBlockHeader)` if the block no longer has missing
268    ///   ancestors and is ready to be reprocessed.
269    /// * `None` if the block still has unresolved ancestors.
270    ///
271    /// # Panics
272    /// - If the block is not found in the `suspended_block_headers` map.
273    /// - If `resolved_dependency` is not in the block's `missing_ancestors`
274    ///   set.
275    ///
276    /// # Side Effects
277    /// - Mutates internal state by removing resolved dependencies and possibly
278    ///   removing the suspended block.
279    fn unsuspend_if_ready(
280        &mut self,
281        suspended_block_ref: &BlockRef,
282        resolved_dependency: &BlockRef,
283    ) -> Option<SuspendedBlockHeader> {
284        let block = self
285            .suspended_headers
286            .get_mut(suspended_block_ref)
287            .expect("Block should be in suspended map");
288
289        assert!(
290            block.missing_ancestors.remove(resolved_dependency),
291            "Block reference {} should be present in missing dependencies of {:?}",
292            suspended_block_ref,
293            block.block_header
294        );
295        // If there are no more missing ancestors, unsuspend and return the block
296        if block.missing_ancestors.is_empty() {
297            return self.suspended_headers.remove(suspended_block_ref);
298        }
299        // Otherwise, keep it suspended
300        None
301    }
302    fn register_suspended_block(
303        &mut self,
304        block_header: VerifiedBlockHeader,
305        missing_ancestors: &BTreeSet<BlockRef>,
306    ) {
307        let block_ref = block_header.reference();
308        match self.suspended_headers.entry(block_ref) {
309            Entry::Vacant(v) => {
310                let suspended_block = SuspendedBlockHeader::new(block_header, missing_ancestors);
311                v.insert(suspended_block);
312            }
313            Entry::Occupied(mut o) => {
314                o.get_mut().missing_ancestors.extend(missing_ancestors);
315            }
316        };
317    }
318    fn register_missing_ancestor(&mut self, block_ref: BlockRef, ancestor: BlockRef) {
319        self.missing_ancestors
320            .entry(ancestor)
321            .or_default()
322            .insert(block_ref);
323    }
324    fn report_suspended_block(&self, block_ref: &BlockRef) {
325        self.context
326            .metrics
327            .node_metrics
328            .block_headers_suspensions
329            .with_label_values(&[self.context.authority_hostname(block_ref.author)])
330            .inc();
331    }
332    fn report_missing_ancestor(&mut self, ancestor: &BlockRef) {
333        self.context
334            .metrics
335            .node_metrics
336            .block_manager_missing_ancestors_by_authority
337            .with_label_values(&[self.context.authority_hostname(ancestor.author)])
338            .inc();
339    }
340    fn register_block_to_fetch(
341        &mut self,
342        missing_ancestor: &BlockRef,
343        child_author: AuthorityIndex,
344    ) {
345        match self.headers_to_fetch.entry(*missing_ancestor) {
346            Entry::Vacant(v) => {
347                // Both the child block's author and the actual missing ancestor's author
348                // are expected to have this block available.
349                v.insert(BTreeSet::from([missing_ancestor.author, child_author]));
350            }
351            Entry::Occupied(mut o) => {
352                o.get_mut().insert(child_author);
353            }
354        }
355    }
356
357    fn report_block_to_fetch(&mut self, block_to_fetch_author: AuthorityIndex) {
358        self.context
359            .metrics
360            .node_metrics
361            .block_manager_missing_block_headers_by_authority
362            .with_label_values(&[self.context.authority_hostname(block_to_fetch_author)])
363            .inc();
364    }
365    fn report_unsuspended_block(&self, unsuspended_block: &SuspendedBlockHeader) {
366        let now = Instant::now();
367        self.context
368            .metrics
369            .node_metrics
370            .block_header_unsuspensions
371            .with_label_values(&[self
372                .context
373                .authority_hostname(unsuspended_block.block_header.author())])
374            .inc();
375        self.context
376            .metrics
377            .node_metrics
378            .suspended_block_header_time
379            .with_label_values(&[self
380                .context
381                .authority_hostname(unsuspended_block.block_header.author())])
382            .observe(
383                now.saturating_duration_since(unsuspended_block.timestamp)
384                    .as_secs_f64(),
385            );
386    }
387    pub(crate) fn is_block_ref_suspended(&self, block_ref: &BlockRef) -> bool {
388        self.suspended_headers.contains_key(block_ref)
389    }
390    pub(crate) fn headers_to_fetch(&self) -> BTreeMap<BlockRef, BTreeSet<AuthorityIndex>> {
391        self.headers_to_fetch.clone()
392    }
393    fn update_stats(&mut self, blocks_to_fetch: u64) {
394        let metrics = &self.context.metrics.node_metrics;
395        metrics.missing_block_headers_total.inc_by(blocks_to_fetch);
396        metrics
397            .block_manager_suspended_block_headers
398            .set(self.suspended_headers.len() as i64);
399        metrics
400            .block_manager_missing_ancestors
401            .set(self.missing_ancestors.len() as i64);
402        metrics
403            .block_manager_missing_block_headers
404            .set(self.headers_to_fetch.len() as i64);
405    }
406    #[cfg(test)]
407    pub(crate) fn blocks_to_fetch_len(&self) -> usize {
408        self.headers_to_fetch.len()
409    }
410    #[cfg(test)]
411    pub(crate) fn insert_block_to_fetch(
412        &mut self,
413        block_ref: BlockRef,
414        set: BTreeSet<AuthorityIndex>,
415    ) -> Option<BTreeSet<AuthorityIndex>> {
416        self.headers_to_fetch.insert(block_ref, set)
417    }
418    #[cfg(test)]
419    pub(crate) fn set_missing_ancestors_with_no_children(&mut self, block_ref: BlockRef) {
420        self.missing_ancestors.entry(block_ref).or_default();
421    }
422    #[cfg(test)]
423    pub(crate) fn suspended_blocks_refs(&self) -> BTreeSet<BlockRef> {
424        self.suspended_headers.keys().cloned().collect()
425    }
426    #[cfg(test)]
427    pub(crate) fn is_empty(&self) -> bool {
428        self.suspended_headers.is_empty()
429            && self.missing_ancestors.is_empty()
430            && self.headers_to_fetch.is_empty()
431    }
432    #[cfg(test)]
433    pub(crate) fn blocks_to_fetch_refs(&self) -> BTreeSet<BlockRef> {
434        self.headers_to_fetch.keys().cloned().collect()
435    }

440    use super::*;
441    /// Evaluates a set of verified block headers to determine which blocks
442    /// should be suspended and which are still missing.
443    ///
444    /// This function is called in randomized tests to verify the
445    /// suspension logic of the `BlockSuspender`.
446    /// returns:
447    /// tuple of:
448    /// - `Vec<BlockRef>`: all block references that should be suspended
449    /// - `Vec<BlockRef>`: all block references that are still missing and need
450    ///   to be fetched
451    pub(crate) fn evaluate_block_headers(
452        seen_so_far: &[VerifiedBlockHeader],
453    ) -> (BTreeSet<BlockRef>, BTreeSet<BlockRef>) {
454        let mut suspended = BTreeSet::new();
455        let mut missing = BTreeSet::new();
456        let mut seen_so_far_ordered = seen_so_far.iter().collect::<Vec<_>>();
457        let seen_so_far_refs: BTreeSet<BlockRef> =
458            seen_so_far.iter().map(|b| b.reference()).collect();
459        seen_so_far_ordered.sort_by_key(|b| b.round());
460        for block in seen_so_far_ordered {
461            if block.round() == 1 {
462                // Skip the first block as it has no ancestors.
463                continue;
464            }
465            let block_ref = block.reference();
466            for ancestor in block.ancestors() {
467                let is_missing = !seen_so_far_refs.contains(ancestor);
468                let is_suspended = suspended.contains(ancestor);
469
470                if is_missing {
471                    missing.insert(*ancestor);
472                }
473
474                if is_missing || is_suspended {
475                    suspended.insert(block_ref);
476                }
477            }
478        }
479        (suspended, missing)
480    }
