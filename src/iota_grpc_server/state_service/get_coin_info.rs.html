4use std::sync::Arc;
5
6use iota_grpc_types::{
7    google::rpc::bad_request::FieldViolation,
8    v1::{
9        coin::{
10            CoinMetadata, CoinTreasury, RegulatedCoinMetadata, coin_treasury::SupplyState,
11            regulated_coin_metadata::CoinRegulatedState,
12        },
13        error_reason::ErrorReason,
14        state_service::{GetCoinInfoRequest, GetCoinInfoResponse},
15    },
16};
17
18use crate::{error::RpcError, types::GrpcReader, validation::object_id_proto};
19
20/// Get coin info for a given coin type.
21///
22/// This endpoint does not use the Merge trait / read_mask pattern because the
23/// proto `GetCoinInfoRequest` has no `read_mask` field — all response fields
24/// are always returned. The response is small and fixed-shape, so field
25/// filtering would add complexity without meaningful savings.
26#[tracing::instrument(skip(reader))]
27pub(crate) fn get_coin_info(
28    reader: Arc<GrpcReader>,
29    GetCoinInfoRequest { coin_type, .. }: GetCoinInfoRequest,
30) -> Result<GetCoinInfoResponse, RpcError> {
31    // Validate coin_type
32    let coin_type_str = coin_type.as_deref().ok_or_else(|| {
33        FieldViolation::new("coin_type")
34            .with_description("coin_type is required")
35            .with_reason(ErrorReason::FieldMissing)
36    })?;
37
38    let core_coin_type = iota_types::parse_iota_struct_tag(coin_type_str).map_err(|e| {
39        FieldViolation::new("coin_type")
40            .with_description(format!("invalid coin_type: {e}"))
41            .with_reason(ErrorReason::FieldInvalid)
42    })?;
43
44    let coin_info = reader.get_coin_v2_info(&core_coin_type)?;
45    let coin_info = coin_info.ok_or_else(|| {
46        RpcError::new(
47            tonic::Code::NotFound,
48            format!(
49                "No coin info found for type {coin_type_str} \
50                 — CoinMetadata or TreasuryCap may not be published"
51            ),
52        )
53    })?;
54
55    let mut response = GetCoinInfoResponse::default().with_coin_type(coin_type_str.to_string());
56
57    // Populate metadata if available
58    if let Some(coin_metadata_object_id) = coin_info.coin_metadata_object_id {
59        if let Some(object) = reader.get_object(&coin_metadata_object_id)? {
60            match iota_types::coin::CoinMetadata::try_from(object) {
61                Ok(metadata) => {
62                    let mut grpc_metadata = CoinMetadata::default()
63                        .with_id(object_id_proto(&metadata.id.id.bytes))
64                        .with_decimals(metadata.decimals as u32)
65                        .with_name(metadata.name)
66                        .with_symbol(metadata.symbol)
67                        .with_description(metadata.description);
68                    if let Some(icon_url) = metadata.icon_url {
69                        grpc_metadata = grpc_metadata.with_icon_url(icon_url);
70                    }
71                    response.metadata = Some(grpc_metadata);
72                }
73                Err(e) => {
74                    tracing::error!(
75                        "Unable to read object {coin_metadata_object_id} as CoinMetadata \
76                         for coin type {coin_type_str}: {e}"
77                    );
78                }
79            }
80        }
81    }
82
83    // Populate treasury if available
84    if let Some(treasury_object_id) = coin_info.treasury_object_id {
85        if let Some(object) = reader.get_object(&treasury_object_id)? {
86            // Determine supply_state from ownership before consuming the object.
87            // Immutable or owned by 0x0 means the TreasuryCap can never be used
88            // to mint, so supply is fixed.
89            let supply_state = match &object.owner {
90                iota_types::object::Owner::Immutable => SupplyState::Fixed,
91                iota_types::object::Owner::AddressOwner(addr)
92                    if *addr == iota_types::base_types::IotaAddress::ZERO =>
93                {
94                    SupplyState::Fixed
95                }
96                _ => SupplyState::Unknown,
97            };
98
99            match iota_types::coin::TreasuryCap::try_from(object) {
100                Ok(treasury) => {
101                    response.treasury = Some(
102                        CoinTreasury::default()
103                            .with_id(object_id_proto(&treasury.id.id.bytes))
104                            .with_total_supply(treasury.total_supply.value)
105                            .with_supply_state(supply_state),
106                    );
107                }
108                Err(e) => {
109                    tracing::error!(
110                        "Unable to read object {treasury_object_id} as TreasuryCap \
111                         for coin type {coin_type_str}: {e}"
112                    );
113                }
114            }
115        }
116    } else if iota_types::gas_coin::GAS::is_gas(&core_coin_type) {
117        // Special case for native GAS coin: get treasury from system state.
118        // The native gas coin's total supply is fixed (no TreasuryCap can mint more).
119        let summary = reader.get_system_state_summary()?;
120        response.treasury = Some(
121            CoinTreasury::default()
122                .with_id(object_id_proto(&summary.iota_treasury_cap_id()))
123                .with_total_supply(summary.iota_total_supply())
124                .with_supply_state(SupplyState::Fixed),
125        );
126    }
127
128    // Populate regulated metadata.
129    //
130    // NOTE: Unlike `CoinMetadata` and `TreasuryCap` above which use the
131    // type-safe `TryFrom<Object>` conversion, `RegulatedCoinMetadata` is
132    // deserialized via raw `bcs::from_bytes`. This is because
133    // `iota_types::deny_list_v1::RegulatedCoinMetadata` does not implement
134    // `TryFrom<Object>`. If that impl is added upstream, this should be
135    // updated to use it for consistency and better error handling.
136    if let Some(regulated_object_id) = coin_info.regulated_coin_metadata_object_id {
137        if let Some(object) = reader.get_object(&regulated_object_id)? {
138            if let Some(move_obj) = object.data.try_as_move() {
139                match bcs::from_bytes::<iota_types::deny_list_v1::RegulatedCoinMetadata>(
140                    move_obj.contents(),
141                ) {
142                    Ok(regulated) => {
143                        // Proto fields `allow_global_pause` and `variant` are not populated
144                        // because the on-chain `RegulatedCoinMetadata` struct does not store
145                        // these values — it only contains `id`, `coin_metadata_object`, and
146                        // `deny_cap_object`.
147                        response.regulated_metadata = Some(
148                            RegulatedCoinMetadata::default()
149                                .with_id(object_id_proto(&regulated.id.id.bytes))
150                                .with_coin_metadata_object(object_id_proto(
151                                    &regulated.coin_metadata_object.bytes,
152                                ))
153                                .with_deny_cap_object(object_id_proto(
154                                    &regulated.deny_cap_object.bytes,
155                                ))
156                                .with_coin_regulated_state(CoinRegulatedState::Regulated),
157                        );
158                    }
159                    Err(e) => {
160                        tracing::error!(
161                            "Unable to read object {regulated_object_id} as \
162                             RegulatedCoinMetadata for coin type {coin_type_str}: {e}"
163                        );
164                    }
165                }
166            } else {
167                tracing::error!("Object {regulated_object_id} is not a Move object");
168            }
169        }
170    } else {
171        // No RegulatedCoinMetadata object exists for this coin type — explicitly
172        // mark as unregulated so clients can distinguish "not regulated" from
173        // "field not populated".
174        response.regulated_metadata = Some(
175            RegulatedCoinMetadata::default()
176                .with_coin_regulated_state(CoinRegulatedState::Unregulated),
177        );
178    }
179
180    Ok(response)
181}
