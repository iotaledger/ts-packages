28/// A small parser used for parsing programmable transaction commands for
29/// transactional tests
30pub struct CommandParser<
31    'a,
32    I: Iterator<Item = (CommandToken, &'a str)>,
33    P: BorrowMut<Parser<'a, CommandToken, I>>,
34> {
35    inner: P,
36    _a: PhantomData<&'a ()>,
37    _i: PhantomData<I>,
38}
39
40#[derive(Debug, Clone)]
41pub struct ParsedMoveCall {
42    pub package: Identifier,
43    pub module: Identifier,
44    pub function: Identifier,
45    pub type_arguments: Vec<ParsedType>,
46    pub arguments: Vec<Argument>,
47}
48
49#[derive(Debug, Clone)]
50pub enum ParsedCommand {
51    MoveCall(Box<ParsedMoveCall>),
52    TransferObjects(Vec<Argument>, Argument),
53    SplitCoins(Argument, Vec<Argument>),
54    MergeCoins(Argument, Vec<Argument>),
55    MakeMoveVec(Option<ParsedType>, Vec<Argument>),
56    Publish(String, Vec<String>),
57    Upgrade(String, Vec<String>, String, Argument),
58}
59
60impl<'a, I: Iterator<Item = (CommandToken, &'a str)>>
61    CommandParser<'a, I, Parser<'a, CommandToken, I>>
62{
63    pub fn new<T: IntoIterator<Item = (CommandToken, &'a str), IntoIter = I>>(v: T) -> Self {
64        Self::from_parser(Parser::new(v))
65    }
66}

68impl<'a, I, P> CommandParser<'a, I, P>
69where
70    I: Iterator<Item = (CommandToken, &'a str)>,
71    P: BorrowMut<Parser<'a, CommandToken, I>>,
72{
73    pub fn from_parser(inner: P) -> Self {
74        Self {
75            inner,
76            _a: PhantomData,
77            _i: PhantomData,
78        }
79    }

81    pub fn parse_commands(&mut self) -> Result<Vec<ParsedCommand>> {
82        let commands = self.inner().parse_list(
83            |p| CommandParser::from_parser(p).parse_command_start(),
84            CommandToken::Semi,
85            // not checked
86            CommandToken::Void,
87            // allow_trailing_delim
88            true,
89        )?;
90        let commands = commands
91            .into_iter()
92            .enumerate()
93            .map(|(actual, (annotated, c))| {
94                if let Some(annotated) = annotated {
95                    if actual != annotated {
96                        anyhow::bail!(
97                            "Actual command index of {actual} \
98                            does not match annotated index {annotated}",
99                        );
100                    }
101                }
102                Ok(c)
103            })
104            .collect::<Result<_>>()?;
105        Ok(commands)
106    }

108    pub fn parse_command_start(&mut self) -> Result<(Option<usize>, ParsedCommand)> {
109        self.inner().advance(CommandToken::CommandStart)?;
110        let idx = if let Some(CommandToken::Number) = self.inner().peek_tok() {
111            let num = self.inner().advance(CommandToken::Number)?;
112            let idx = usize::from_str(num).context("Invalid command index annotation")?;
113            self.inner().advance(CommandToken::Colon)?;
114            Some(idx)
115        } else {
116            None
117        };
118        let cmd = self.parse_command()?;
119        Ok((idx, cmd))
120    }
121
122    pub fn parse_command(&mut self) -> Result<ParsedCommand> {
123        use super::token::CommandToken as Tok;
124        Ok(match self.inner().advance_any()? {
125            (Tok::Ident, TRANSFER_OBJECTS) => {
126                self.inner().advance(Tok::LParen)?;
127                let args = self.parse_command_args(Tok::LBracket, Tok::RBracket)?;
128                self.inner().advance(Tok::Comma)?;
129                let arg = self.parse_command_arg()?;
130                self.maybe_trailing_comma()?;
131                self.inner().advance(Tok::RParen)?;
132                ParsedCommand::TransferObjects(args, arg)
133            }
134            (Tok::Ident, SPLIT_COINS) => {
135                self.inner().advance(Tok::LParen)?;
136                let coin = self.parse_command_arg()?;
137                self.inner().advance(Tok::Comma)?;
138                let amts = self.parse_command_args(Tok::LBracket, Tok::RBracket)?;
139                self.maybe_trailing_comma()?;
140                self.inner().advance(Tok::RParen)?;
141                ParsedCommand::SplitCoins(coin, amts)
142            }
143            (Tok::Ident, MERGE_COINS) => {
144                self.inner().advance(Tok::LParen)?;
145                let target = self.parse_command_arg()?;
146                self.inner().advance(Tok::Comma)?;
147                let coins = self.parse_command_args(Tok::LBracket, Tok::RBracket)?;
148                self.maybe_trailing_comma()?;
149                self.inner().advance(Tok::RParen)?;
150                ParsedCommand::MergeCoins(target, coins)
151            }
152            (Tok::Ident, MAKE_MOVE_VEC) => {
153                let type_opt = self.parse_type_arg_opt()?;
154                self.inner().advance(Tok::LParen)?;
155                let args = self.parse_command_args(Tok::LBracket, Tok::RBracket)?;
156                self.maybe_trailing_comma()?;
157                self.inner().advance(Tok::RParen)?;
158                ParsedCommand::MakeMoveVec(type_opt, args)
159            }
160            (Tok::Ident, PUBLISH) => {
161                self.inner().advance(Tok::LParen)?;
162                let staged_package = self.inner().advance(Tok::Ident)?;
163                self.inner().advance(Tok::Comma)?;
164                self.inner().advance(Tok::LBracket)?;
165                let dependencies = self.inner().parse_list(
166                    |p| Ok(p.advance(Tok::Ident)?.to_owned()),
167                    CommandToken::Comma,
168                    Tok::RBracket,
169                    // allow_trailing_delim
170                    true,
171                )?;
172                self.inner().advance(Tok::RBracket)?;
173                self.maybe_trailing_comma()?;
174                self.inner().advance(Tok::RParen)?;
175                ParsedCommand::Publish(staged_package.to_owned(), dependencies)
176            }
177            (Tok::Ident, UPGRADE) => {
178                self.inner().advance(Tok::LParen)?;
179                let staged_package = self.inner().advance(Tok::Ident)?;
180                self.inner().advance(Tok::Comma)?;
181                self.inner().advance(Tok::LBracket)?;
182                let dependencies = self.inner().parse_list(
183                    |p| Ok(p.advance(Tok::Ident)?.to_owned()),
184                    CommandToken::Comma,
185                    Tok::RBracket,
186                    // allow_trailing_delim
187                    true,
188                )?;
189                self.inner().advance(Tok::RBracket)?;
190                self.inner().advance(Tok::Comma)?;
191                let upgraded_package = self.inner().advance(Tok::Ident)?;
192                self.inner().advance(Tok::Comma)?;
193                let upgrade_ticket = self.parse_command_arg()?;
194                self.maybe_trailing_comma()?;
195                self.inner().advance(Tok::RParen)?;
196                ParsedCommand::Upgrade(
197                    staged_package.to_owned(),
198                    dependencies,
199                    upgraded_package.to_owned(),
200                    upgrade_ticket,
201                )
202            }
203            (Tok::Ident, contents) => {
204                let package = Identifier::new(contents)?;
205                self.inner().advance(Tok::ColonColon)?;
206                let module = Identifier::new(self.inner().advance(Tok::Ident)?)?;
207                self.inner().advance(Tok::ColonColon)?;
208                let function = Identifier::new(self.inner().advance(Tok::Ident)?)?;
209                let type_arguments = self.parse_type_args_opt()?.unwrap_or_default();
210                let arguments = self.parse_command_args(Tok::LParen, Tok::RParen)?;
211                let call = ParsedMoveCall {
212                    package,
213                    module,
214                    function,
215                    type_arguments,
216                    arguments,
217                };
218                ParsedCommand::MoveCall(Box::new(call))
219            }
220
221            (tok, _) => bail!("unexpected token {}, expected command identifier", tok),
222        })
223    }
224
225    pub fn maybe_trailing_comma(&mut self) -> Result<()> {
226        if let Some(CommandToken::Comma) = self.inner().peek_tok() {
227            self.inner().advance(CommandToken::Comma)?;
228        }
229        Ok(())
230    }
231
232    pub fn parse_command_args(
233        &mut self,
234        start: CommandToken,
235        end: CommandToken,
236    ) -> Result<Vec<Argument>> {
237        self.inner().advance(start)?;
238        let args = self.inner().parse_list(
239            |p| CommandParser::from_parser(p).parse_command_arg(),
240            CommandToken::Comma,
241            end,
242            // allow_trailing_delim
243            true,
244        )?;
245        self.inner().advance(end)?;
246        Ok(args)
247    }
248
249    pub fn parse_command_arg(&mut self) -> Result<Argument> {
250        use super::token::CommandToken as Tok;
251        Ok(match self.inner().advance_any()? {
252            (Tok::Ident, GAS_COIN) => Argument::GasCoin,
253            (Tok::Ident, INPUT) => {
254                self.inner().advance(Tok::LParen)?;
255                let num = self.parse_u16()?;
256                self.maybe_trailing_comma()?;
257                self.inner().advance(Tok::RParen)?;
258                Argument::Input(num)
259            }
260            (Tok::Ident, RESULT) => {
261                self.inner().advance(Tok::LParen)?;
262                let num = self.parse_u16()?;
263                self.maybe_trailing_comma()?;
264                self.inner().advance(Tok::RParen)?;
265                Argument::Result(num)
266            }
267            (Tok::Ident, NESTED_RESULT) => {
268                self.inner().advance(Tok::LParen)?;
269                let i = self.parse_u16()?;
270                self.inner().advance(Tok::Comma)?;
271                let j = self.parse_u16()?;
272                self.maybe_trailing_comma()?;
273                self.inner().advance(Tok::RParen)?;
274                Argument::NestedResult(i, j)
275            }
276            (tok, _) => bail!("unexpected token {}, expected argument identifier", tok),
277        })
278    }
279
280    pub fn parse_u16(&mut self) -> Result<u16> {
281        let contents = self.inner().advance(CommandToken::Number)?;
282        u16::from_str(contents).context("Expected u16 for Argument")
283    }
284
285    pub fn parse_type_arg_opt(&mut self) -> Result<Option<ParsedType>> {
286        match self.parse_type_args_opt()? {
287            None => Ok(None),
288            Some(v) if v.len() != 1 => bail!(
289                "unexpected multiple type arguments. Expected 1 type argument but got {}",
290                v.len()
291            ),
292            Some(mut v) => Ok(Some(v.pop().unwrap())),
293        }
294    }
295
296    pub fn parse_type_args_opt(&mut self) -> Result<Option<Vec<ParsedType>>> {
297        if !matches!(self.inner().peek_tok(), Some(CommandToken::TypeArgString)) {
298            return Ok(None);
299        }
300        let contents = self.inner().advance(CommandToken::TypeArgString)?;
301        let type_tokens: Vec<_> = TypeToken::tokenize(contents)?
302            .into_iter()
303            .filter(|(tok, _)| !tok.is_whitespace())
304            .collect();
305        let mut parser = Parser::new(type_tokens);
306        parser.advance(TypeToken::Lt)?;
307        let res = parser.parse_list(|p| p.parse_type(), TypeToken::Comma, TypeToken::Gt, true)?;
308        parser.advance(TypeToken::Gt)?;
309        if let Ok((_, contents)) = parser.advance_any() {
310            bail!("Expected end of token stream. Got: {}", contents)
311        }
312        Ok(Some(res))
313    }
314
315    pub fn inner(&mut self) -> &mut Parser<'a, CommandToken, I> {
316        self.inner.borrow_mut()
317    }

320impl ParsedCommand {
321    pub fn parse_vec(s: &str) -> Result<Vec<Self>> {
322        let tokens: Vec<_> = CommandToken::tokenize(s)?
323            .into_iter()
324            .filter(|(tok, _)| !tok.is_whitespace())
325            .collect();
326        let mut parser = CommandParser::new(tokens);
327        let res = parser.parse_commands()?;
328        if let Ok((_, contents)) = parser.inner().advance_any() {
329            bail!("Expected end of token stream. Got: {}", contents)
330        }
331        Ok(res)
332    }

334    pub fn into_command(
335        self,
336        staged_packages: &impl Fn(&str) -> Option<Vec<Vec<u8>>>,
337        address_mapping: &impl Fn(&str) -> Option<AccountAddress>,
338    ) -> Result<Command> {
339        Ok(match self {
340            ParsedCommand::MoveCall(c) => {
341                Command::MoveCall(Box::new(c.into_move_call(address_mapping)?))
342            }
343            ParsedCommand::TransferObjects(objs, recipient) => {
344                Command::TransferObjects(objs, recipient)
345            }
346            ParsedCommand::SplitCoins(coin, amts) => Command::SplitCoins(coin, amts),
347            ParsedCommand::MergeCoins(target, coins) => Command::MergeCoins(target, coins),
348            ParsedCommand::MakeMoveVec(ty_opt, args) => Command::make_move_vec(
349                ty_opt
350                    .map(|t| t.into_type_tag(address_mapping))
351                    .transpose()?,
352                args,
353            ),
354            ParsedCommand::Publish(staged_package, dependencies) => {
355                let Some(package_contents) = staged_packages(&staged_package) else {
356                    bail!("No staged package '{staged_package}'");
357                };
358                let dependencies = dependencies
359                    .into_iter()
360                    .map(|d| match address_mapping(&d) {
361                        Some(a) => Ok(a.into()),
362                        None => bail!("Unbound dependency '{d}"),
363                    })
364                    .collect::<Result<Vec<ObjectID>>>()?;
365                Command::Publish(package_contents, dependencies)
366            }
367            ParsedCommand::Upgrade(staged_package, dependencies, upgraded_package, ticket) => {
368                let Some(package_contents) = staged_packages(&staged_package) else {
369                    bail!("No staged package '{staged_package}'");
370                };
371                let dependencies = dependencies
372                    .into_iter()
373                    .map(|d| match address_mapping(&d) {
374                        Some(a) => Ok(a.into()),
375                        None => bail!("Unbound dependency '{d}"),
376                    })
377                    .collect::<Result<Vec<ObjectID>>>()?;
378                let Some(upgraded_package) = address_mapping(&upgraded_package) else {
379                    bail!("Unbound upgraded package '{upgraded_package}'");
380                };
381                let upgraded_package = upgraded_package.into();
382                Command::Upgrade(package_contents, dependencies, upgraded_package, ticket)
383            }
384        })
385    }
386}

388impl ParsedMoveCall {
389    pub fn into_move_call(
390        self,
391        address_mapping: &impl Fn(&str) -> Option<AccountAddress>,
392    ) -> Result<ProgrammableMoveCall> {
393        let Self {
394            package,
395            module,
396            function,
397            type_arguments,
398            arguments,
399        } = self;
400        let Some(package) = address_mapping(package.as_str()) else {
401            bail!("Unable to resolve package {}", package)
402        };
403        let type_arguments = type_arguments
404            .into_iter()
405            .map(|t| t.into_type_tag(address_mapping).map(TypeInput::from))
406            .collect::<Result<_>>()?;
407        Ok(ProgrammableMoveCall {
408            package: package.into(),
409            module: module.to_string(),
410            function: function.to_string(),
411            type_arguments,
412            arguments,
413        })
414    }
