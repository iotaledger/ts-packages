19/// Helper function to filter out empty strings and join them with a separator.
20pub(crate) fn join_non_empty_strings<S: AsRef<str>>(items: &[S], separator: &str) -> String {
21    items
22        .iter()
23        .map(|s| s.as_ref())
24        .filter(|f| !f.is_empty())
25        .collect::<Vec<_>>()
26        .join(separator)
27}

29pub fn build_cargo_command<S1: AsRef<str>, S2: AsRef<str>, S3: AsRef<str>>(
30    subcommand: &str,
31    toolchain: Option<String>,
32    features: Vec<String>,
33    binaries: &[S1],
34    setup_commands: &[S2],
35    additional_args: &[S3],
36) -> String {
37    let toolchain_arg = toolchain
38        .as_ref()
39        .filter(|t| t.as_str() != "stable")
40        .map(|t| format!("+{t}"))
41        .unwrap_or_default();
42
43    let target_dir_arg = toolchain
44        .filter(|t| t != "stable")
45        .map(|t| format!("--target-dir target_{t}"))
46        .unwrap_or_default();
47
48    let features_arg = if features.is_empty() {
49        "".to_string()
50    } else {
51        format!("--features \"{}\"", features.join(" "))
52    };
53
54    let binaries_args: Vec<String> = binaries
55        .iter()
56        .map(|name| format!("--bin {}", name.as_ref()))
57        .collect();
58    let binaries_args = join_non_empty_strings(&binaries_args, " ");
59
60    let additional_args_str = join_non_empty_strings(additional_args, " ");
61
62    let mut cargo_args = vec![
63        "cargo",
64        &toolchain_arg,
65        subcommand,
66        &target_dir_arg,
67        "--release",
68        &binaries_args,
69        &features_arg,
70    ];
71
72    if !additional_args_str.is_empty() {
73        cargo_args.extend(&["--", &additional_args_str]);
74    }
75
76    let cargo_command = join_non_empty_strings(&cargo_args, " ");
77
78    let default_setup = [
79        "source \"$HOME/.cargo/env\"",
80        "export RUSTFLAGS='-C target-cpu=native'",
81    ];
82
83    let all_commands: Vec<String> = default_setup
84        .iter()
85        .map(|s| s.to_string())
86        .chain(setup_commands.iter().map(|s| s.as_ref().to_string()))
87        .chain(std::iter::once(cargo_command))
88        .collect();
89    join_non_empty_strings(&all_commands, " && ")
90}

94pub struct Repository {
95    /// The url of the repository.

97    pub url: Url,

99    pub commit: String,

102fn parse_url<'de, D>(deserializer: D) -> Result<Url, D::Error>
103where
104    D: Deserializer<'de>,
105{
106    let s: &str = Deserialize::deserialize(deserializer)?;
107    let url = Url::parse(s).map_err(D::Error::custom)?;
108
109    match url.path_segments().map(|x| x.count() >= 2) {
110        None | Some(false) => Err(D::Error::custom(SettingsError::MalformedRepositoryUrl(url))),
111        _ => Ok(url),
112    }
113}

157/// The testbed settings. Those are topically specified in a file.
158#[derive(Deserialize, Clone)]
159pub struct Settings {
160    /// The testbed unique id. This allows multiple users to run concurrent
161    /// testbeds on the same cloud provider's account without interference
162    /// with each others.
163    pub testbed_id: String,
164    /// The cloud provider hosting the testbed.
165    pub cloud_provider: CloudProvider,
166    /// The path to the secret token for authentication with the cloud provider.
167    pub token_file: PathBuf,
168    /// The ssh private key to access the instances.
169    pub ssh_private_key_file: PathBuf,
170    /// The corresponding ssh public key registered on the instances. If not
171    /// specified. the public key defaults the same path as the private key
172    /// with an added extension 'pub'.
173    pub ssh_public_key_file: Option<PathBuf>,
174    /// The list of cloud provider regions to deploy on the testbed. If the
175    /// metrics server is used, it will be located in the first region in
176    /// the list.
177    pub regions: Vec<String>,
178    /// The specs of the instances to deploy for nodes. Those are dependent
179    /// on the cloud provider, e.g., specifying 't3.medium' creates
180    /// instances with 2 vCPU and 4GB of ram on AWS.
181    pub node_specs: String,
182    /// The list of cloud provider regions to deploy for clients on the
183    /// testbed.
184    pub client_specs: String,
185    /// Region to deploy the metrics instance.
186    pub metrics_specs: String,
187    /// Optional build cache configuration.
188    pub build_cache: Option<BuildCache>,
189    /// The details of the git repository to deploy.
190    pub repository: Repository,
191    /// The working directory on the remote instance (containing all
192    /// configuration files).
193    #[serde(default = "default_working_dir")]
194    pub working_dir: PathBuf,
195    /// Pass '--use-fullnode-for-execution' and '--fullnode-rpc-addresses' to
196    /// stress binary.
197    #[serde(default)]
198    pub use_fullnode_for_execution: bool,
199    /// The directory (on the local machine) where to save benchmarks
200    /// results.
201    #[serde(default = "default_results_dir")]
202    pub results_dir: PathBuf,
203    /// Binary build configuration.
204    #[serde(default)]
205    pub build_configs: HashMap<String, BinaryBuildConfig>,
206    /// Enable flamegraphs when running nodes.
207    #[serde(default)]
208    pub enable_flamegraph: bool,
209    /// Enable prometheus snapshots.
210    #[serde(default)]
211    pub enable_prometheus_snapshots: bool,
212}

222impl Settings {
223    /// Load the settings from a json file.
224    pub fn load<P>(path: P) -> SettingsResult<Self>
225    where
226        P: AsRef<Path> + Display + Clone,
227    {
228        let reader = || -> Result<Self, std::io::Error> {
229            let data = fs::read(path.clone())?;
230            let data = resolve_env(std::str::from_utf8(&data).unwrap());
231            let settings: Settings = serde_json::from_slice(data.as_bytes())?;
232
233            fs::create_dir_all(&settings.results_dir)?;
234
235            Ok(settings)
236        };
237
238        reader().map_err(|e| SettingsError::InvalidSettings {
239            file: path.to_string(),
240            message: e.to_string(),
241        })
242    }
243
244    /// Get the name of the repository (from its url).
245    pub fn repository_name(&self) -> String {
246        self.repository
247            .url
248            .path_segments()
249            .expect("Url should already be checked when loading settings")
250            .collect::<Vec<_>>()[1]
251            .split('.')
252            .next()
253            .unwrap()
254            .to_string()
255    }
256
257    /// Load the secret token to authenticate with the cloud provider.
258    pub fn load_token(&self) -> SettingsResult<String> {
259        match fs::read_to_string(&self.token_file) {
260            Ok(token) => Ok(token.trim_end_matches('\n').to_string()),
261            Err(e) => Err(SettingsError::InvalidTokenFile {
262                file: self.token_file.display().to_string(),
263                message: e.to_string(),
264            }),
265        }
266    }
267
268    /// Load the ssh public key from file.
269    pub fn load_ssh_public_key(&self) -> SettingsResult<String> {
270        let ssh_public_key_file = self.ssh_public_key_file.clone().unwrap_or_else(|| {
271            let mut private = self.ssh_private_key_file.clone();
272            private.set_extension("pub");
273            private
274        });
275        match fs::read_to_string(&ssh_public_key_file) {
276            Ok(token) => Ok(token.trim_end_matches('\n').to_string()),
277            Err(e) => Err(SettingsError::InvalidSshPublicKeyFile {
278                file: ssh_public_key_file.display().to_string(),
279                message: e.to_string(),
280            }),
281        }
282    }

284    /// Check if the build cache is enabled.
285    pub fn build_cache_enabled(&self) -> bool {
286        self.build_cache
287            .as_ref()
288            .map(|b| b.enabled)
289            .unwrap_or(false)
290    }

292    /// Get build groups for the default binaries (iota, iota-node, stress).
293    /// Groups binaries by toolchain and features to minimize build steps.
294    pub fn build_groups(&self) -> BuildGroups {
295        let mut groups: BuildGroups = HashMap::new();
296
297        for name in ["iota", "iota-node", "stress"] {
298            let config = self.build_configs.get(name);
299
300            let mut features = config.map(|c| c.features.clone()).unwrap_or_default();
301            features.sort(); // Sort for consistent grouping
302
303            let group = BuildGroup {
304                toolchain: config.and_then(|c| c.toolchain.clone()),
305                features,
306            };
307
308            groups.entry(group).or_default().push(name.to_string());
309        }
310
311        groups
312    }

314    /// Get the build cache server configuration for a specific CPU target.
315    pub fn build_cache_server_for_target(&self, cpu_target: &str) -> Option<&BuildCacheServer> {
316        self.build_cache.as_ref().and_then(|build_cache| {
317            if build_cache.enabled {
318                // Find a server that supports this CPU target
319                build_cache
320                    .servers
321                    .values()
322                    .find(|server| server.targets.contains(&cpu_target.to_string()))
323            } else {
324                None
325            }
326        })
327    }

396// Resolves ${ENV} into it's value for each env variable.
397fn resolve_env(s: &str) -> String {
398    let mut s = s.to_string();
399    for (name, value) in env::vars() {
400        s = s.replace(&format!("${{{name}}}"), &value);
401    }
402    if s.contains("${") {
403        eprintln!("settings.json:\n{s}\n");
404        panic!("Unresolved env variables in the settings.json");
405    }
406    s
407}
