12#[derive(Debug, Clone, Deserialize, Serialize)]
13pub enum TopologyLayout {
14    /// All nodes are randomly distributed with their own latencies, no clusters
15    RandomGeographical,
16    /// Nodes are randomly distributed in number_of_clusters clusters
17    RandomClustered { number_of_clusters: usize },
18    /// Uses a hardcoded 10x10 matrix with 10 equal-sized regions
19    HardCodedClustered,
20    /// Use mainnet validator region distribution with shuffled order
21    Mainnet,
22}
23
24#[derive(Debug, Clone, Deserialize, Serialize)]
25pub enum PerturbationSpec {
26    /// No Perturbation introduced
27    None,
28    /// Broken Triangle introduced for number_of_triangles Triangles of nodes
29    /// latency(A,B) + latency(B,C) + added_latency =  Latency(A.C)
30    BrokenTriangle {
31        number_of_triangles: u16,
32        added_latency: u16,
33    },
34    /// Blocking connections
35    Blocking {
36        number_of_blocked_connections: usize,
37    },
38}

40pub struct NetworkLatencyCommandBuilder<'a> {
41    instances: &'a [Instance],
42    topology_layout: TopologyLayout,
43    perturbation_spec: PerturbationSpec,
44    max_latency: u16,
45}
46
47pub fn latency_command(latency_vector: &Vec<(&Instance, u16)>) -> String {
48    // Clean existing rules
49    let mut cmd = "export IFACE=$(ip -j route get 172 | jq -r '.[0].dev') && ".to_string();
50    cmd.push_str("sudo tc qdisc del dev $IFACE root 2>/dev/null || true && ");
51    cmd.push_str("sudo tc qdisc add dev $IFACE root handle 1: htb default 1 && ");
52    // Root prio qdisc
53    cmd.push_str("sudo tc class add dev $IFACE parent 1: classid 1:1 htb rate 1gbit && ");
54
55    // Add one netem band per IP
56    for (i, (instance, latency)) in latency_vector.iter().enumerate() {
57        let ip = instance.private_ip;
58        let handle = i + 10; // avoid conflict with default bands
59        cmd.push_str(&format!(
60            "sudo tc class add dev $IFACE parent 1:1 classid 1:{handle} htb rate 1gbit && "
61        ));
62        cmd.push_str(&format!(
63            "sudo tc qdisc add dev $IFACE parent 1:{handle} handle {handle}: netem delay {latency}ms && "
64        ));
65        // Add filters that map MARKs to the prio bands
66        cmd.push_str(&format!(
67            "sudo tc filter add dev $IFACE protocol ip parent 1: prio 1 u32 match ip dst {ip}/32 flowid 1:{handle} && "
68        ));
69    }
70
71    // Remove trailing " && "
72    cmd.trim_end_matches(" && ").to_string()
73}
74impl<'a> NetworkLatencyCommandBuilder<'a> {
75    pub fn new(instances: &'a [Instance]) -> Self {
76        Self {
77            instances,
78            topology_layout: TopologyLayout::RandomGeographical,
79            perturbation_spec: PerturbationSpec::None,
80            max_latency: 500,
81        }
82    }
83
84    pub fn with_topology_layout(mut self, topology_layout: TopologyLayout) -> Self {
85        self.topology_layout = topology_layout;
86        self
87    }
88
89    pub fn with_perturbation_spec(mut self, perturbation_spec: PerturbationSpec) -> Self {
90        self.perturbation_spec = perturbation_spec;
91        self
92    }
93
94    pub fn with_max_latency(mut self, max_latency: u16) -> Self {
95        self.max_latency = max_latency;
96        self
97    }
98
99    pub fn build_network_latency_matrix(self) -> Vec<(Instance, String)> {
100        let latency_matrix = LatencyMatrixBuilder::new(self.instances.len())
101            .with_max_latency(self.max_latency)
102            .with_topology_layout(self.topology_layout)
103            .with_perturbation_spec(self.perturbation_spec)
104            .build();
105        // print out the latency matrix
106        println!("\n\n{:?}\n\n", latency_matrix);
107        let mut instance2instance_latency_map: HashMap<&Instance, Vec<(&Instance, u16)>> =
108            HashMap::new();
109        for (i, instance) in self.instances.iter().enumerate() {
110            let entry = instance2instance_latency_map.entry(instance).or_default();
111            #[expect(clippy::needless_range_loop)]
112            for j in 0..self.instances.len() {
113                if latency_matrix[i][j] == 0 {
114                    // no need to generate latency commands where latency is 0
115                    // same cluster or same node
116                    continue;
117                }
118                entry.push((&self.instances[j], latency_matrix[i][j]));
119            }
120        }
121        instance2instance_latency_map
122            .iter()
123            .map(|(instance, vector)| ((*instance).clone(), latency_command(vector)))
124            .collect::<Vec<_>>()
125    }
126}
