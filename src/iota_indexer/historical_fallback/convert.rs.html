4//! Conversion utilities for historical fallback data.
5//!
6//! This module provides wrapper types that enable conversion from raw data
7//! fetched from historical fallback storage into the `Stored*` or JSON-RPC
8//! compatible types used by the Indexer's JSON-RPC API layer.
9
10use std::sync::Arc;
11
12use iota_json_rpc_types::IotaEvent;
13use iota_package_resolver::{PackageStore, Resolver};
14use iota_types::{
15    digests::TransactionDigest,
16    effects::TransactionEvents,
17    full_checkpoint_content::CheckpointTransaction,
18    messages_checkpoint::{CertifiedCheckpointSummary, CheckpointContents},
19    object::Object,
20};
21use prometheus::Registry;
22
23use crate::{
24    errors::{IndexerError, IndexerResult},
25    ingestion::{common::prepare::extract_df_kind, primary::prepare::PrimaryWorker},
26    metrics::IndexerMetrics,
27    models::{
28        checkpoints::StoredCheckpoint,

30        transactions::{StoredTransaction, tx_events_to_iota_tx_events},
31    },
32    types::{IndexedCheckpoint, IndexedObject},
33};

35/// Alias for an [`Object`] fetched from historical fallback storage.
36///
37/// Contains all data needed to reconstruct a [`StoredObject`].
38pub(crate) type HistoricalFallbackObject = Object;
39
40/// Alias for [`CertifiedCheckpointSummary`] with its [`CheckpointContents`]
41/// data fetched from historical fallback storage.
42///
43/// Contains all data needed to reconstruct a [`StoredCheckpoint`].
44pub(crate) type HistoricalFallbackCheckpoint = (CertifiedCheckpointSummary, CheckpointContents);
45
46impl From<HistoricalFallbackObject> for StoredObject {
47    fn from(object: HistoricalFallbackObject) -> Self {
48        let df_kind = extract_df_kind(&object);
49        let indexed = IndexedObject::from_object(None, object, df_kind);
50        StoredObject::from(indexed)
51    }
52}
53
54impl From<HistoricalFallbackCheckpoint> for StoredCheckpoint {
55    fn from(checkpoint: HistoricalFallbackCheckpoint) -> Self {
56        let (checkpoint_summary, checkpoint_contents) = checkpoint;
57        // StoredCheckpoint::from implementation does not use the `successful_tx_num`
58        // param in IndexedCheckpoint::from_iota_checkpoint, in this regard it is safe
59        // to hardcode to 0.
60        let indexed =
61            IndexedCheckpoint::from_iota_checkpoint(&checkpoint_summary, &checkpoint_contents, 0);
62        StoredCheckpoint::from(&indexed)
63    }
64}

66/// Wrapper for [`TransactionEvents`] and additional data fetched from
67/// historical fallback storage.
68///
69/// Contains all data needed to reconstruct [`IotaEvent`]s.
70#[derive(Debug, Clone)]
71pub struct HistoricalFallbackEvents {
72    /// Events emitted during transaction execution.
73    events: TransactionEvents,
74    /// Checkpoint timestamp.
75    timestamp: u64,
76}
77
78impl HistoricalFallbackEvents {
79    pub fn new(events: TransactionEvents, checkpoint_summary: CertifiedCheckpointSummary) -> Self {
80        Self {
81            events,
82            timestamp: checkpoint_summary.timestamp_ms,
83        }
84    }
85
86    /// Converts the raw [`TransactionEvents`] into JSON RPC compatible
87    /// [`IotaEvent`]s.
88    pub(crate) async fn into_iota_events(
89        self,
90        package_resolver: &Arc<Resolver<impl PackageStore>>,
91        tx_digest: TransactionDigest,
92    ) -> IndexerResult<Vec<IotaEvent>> {
93        tx_events_to_iota_tx_events(
94            self.events,
95            package_resolver,
96            tx_digest,
97            Some(self.timestamp),
98        )
99        .await
100        .map(|tx_block_event| tx_block_event.data)
101    }
102}

104/// Wrapper for a complete transaction fetched from historical fallback storage.
105///
106/// Contains all data needed to reconstruct a [`StoredTransaction`].
107#[derive(Debug, Clone)]
108pub struct HistoricalFallbackTransaction {
109    /// Checkpointed transaction data.
110    checkpoint_transaction: CheckpointTransaction,
111    /// Checkpoint sequence number the transaction is part of.
112    historical_checkpoint: HistoricalFallbackCheckpoint,
113}
114
115impl HistoricalFallbackTransaction {
116    pub fn new(
117        checkpoint_transaction: CheckpointTransaction,
118        historical_checkpoint: HistoricalFallbackCheckpoint,
119    ) -> Self {
120        Self {
121            checkpoint_transaction,
122            historical_checkpoint,
123        }
124    }
125
126    /// Converts the historical fallback transaction into a
127    /// [`StoredTransaction`].
128    pub(crate) async fn into_stored_transaction(self) -> IndexerResult<StoredTransaction> {
129        let tx_digest = self.checkpoint_transaction.transaction.digest();
130        let (summary, contents) = self.historical_checkpoint;
131
132        let Some(tx_sequence_number) = contents
133            .enumerate_transactions(&summary)
134            .find(|(_seq, execution_digest)| &execution_digest.transaction == tx_digest)
135            .map(|(seq, _execution_digest)| seq)
136        else {
137            return Err(IndexerError::HistoricalFallbackStorageError(format!(
138                "cannot find transaction sequence number to transaction: {tx_digest}"
139            )));
140        };
141
142        let indexed_tx = PrimaryWorker::index_transaction(
143            &self.checkpoint_transaction,
144            tx_sequence_number,
145            summary.sequence_number,
146            summary.timestamp_ms,
147            &IndexerMetrics::new(&Registry::new()),
148        )
149        .await?;
150
151        Ok(StoredTransaction::from(&indexed_tx))
152    }
153}
