72    async fn process_checkpoint(
73        &self,
74        checkpoint: Arc<CheckpointData>,
75    ) -> Result<Self::Message, Self::Error> {
76        self.metrics
77            .latest_fullnode_checkpoint_sequence_number
78            .set(checkpoint.checkpoint_summary.sequence_number as i64);
79        let time_now_ms = chrono::Utc::now().timestamp_millis();
80        let cp_download_lag = time_now_ms - checkpoint.checkpoint_summary.timestamp_ms as i64;
81        info!(
82            "checkpoint download lag for cp {}: {} ms",
83            checkpoint.checkpoint_summary.sequence_number, cp_download_lag
84        );
85        self.metrics.download_lag_ms.set(cp_download_lag);
86        self.metrics
87            .max_downloaded_checkpoint_sequence_number
88            .set(checkpoint.checkpoint_summary.sequence_number as i64);
89        self.metrics
90            .downloaded_checkpoint_timestamp_ms
91            .set(checkpoint.checkpoint_summary.timestamp_ms as i64);
92        info!(
93            "Indexer lag: downloaded checkpoint {} with time now {} and checkpoint time {}",
94            checkpoint.checkpoint_summary.sequence_number,
95            time_now_ms,
96            checkpoint.checkpoint_summary.timestamp_ms
97        );
98
99        let checkpoint_data = Self::index_checkpoint(
100            &checkpoint,
101            Arc::new(self.metrics.clone()),
102            Self::index_packages(slice::from_ref(&checkpoint), &self.metrics),
103        )
104        .await?;
105        self.indexed_checkpoint_sender
106            .send(checkpoint_data)
107            .await
108            .map_err(|_| {
109                IndexerError::MpscChannel(
110                    "failed to send checkpoint data, receiver half closed".into(),
111                )
112            })?;
113        Ok(())
114    }
115}

128    async fn index_epoch(data: &CheckpointData) -> Result<Option<EpochToCommit>, IndexerError> {
129        let checkpoint_object_store = EpochEndIndexingObjectStore::new(data);
130
131        let CheckpointData {
132            transactions,
133            checkpoint_summary,
134            checkpoint_contents: _,
135        } = data;
136
137        // Genesis epoch
138        if *checkpoint_summary.sequence_number() == 0 {
139            info!("Processing genesis epoch");
140            let system_state =
141                get_iota_system_state(&checkpoint_object_store)?.into_iota_system_state_summary();
142            return Ok(Some(EpochToCommit {
143                last_epoch: None,
144                new_epoch: StartOfEpochUpdate::new(
145                    &system_state,
146                    0, // first_checkpoint_id
147                    0, // first_tx_sequence_number
148                    None,
149                ),
150            }));
151        }
152
153        // If not end of epoch, return
154        if checkpoint_summary.end_of_epoch_data.is_none() {
155            return Ok(None);
156        }
157
158        let event = transactions
159            .iter()
160            .flat_map(|t| t.events.as_ref().map(|e| &e.data))
161            .flatten()
162            .find(|ev| ev.is_system_epoch_info_event_v1() || ev.is_system_epoch_info_event_v2())
163            .map(|ev| {
164                if ev.is_system_epoch_info_event_v2() {
165                    SystemEpochInfoEvent::V2(
166                        bcs::from_bytes::<SystemEpochInfoEventV2>(&ev.contents).expect(
167                            "event deserialization should succeed as type was pre-validated",
168                        ),
169                    )
170                } else {
171                    SystemEpochInfoEvent::V1(
172                        bcs::from_bytes::<SystemEpochInfoEventV1>(&ev.contents).expect(
173                            "event deserialization should succeed as type was pre-validated",
174                        ),
175                    )
176                }
177            });
178
179        let system_state = get_iota_system_state(&checkpoint_object_store)?;
180        if event.is_none() {
181            warn!(
182                "no SystemEpochInfoEvent found at end of epoch {}, some epoch data will be set to default.",
183                checkpoint_summary.epoch,
184            );
185            assert!(
186                system_state.safe_mode(),
187                "iota is not in safe mode but no SystemEpochInfoEvent found at end of epoch {}",
188                checkpoint_summary.epoch
189            );
190        }
191
192        let event = event
193            .as_ref()
194            .map_or_else(Default::default, IndexedEpochInfoEvent::from);
195        let new_epoch_first_checkpoint_id = checkpoint_summary.sequence_number + 1;
196        let new_epoch_first_tx_sequence_number = checkpoint_summary.network_total_transactions;
197        Ok(Some(EpochToCommit {
198            last_epoch: Some(EndOfEpochUpdate::new(checkpoint_summary, &event)),
199            new_epoch: StartOfEpochUpdate::new(
200                &system_state.into_iota_system_state_summary(),
201                new_epoch_first_checkpoint_id,
202                new_epoch_first_tx_sequence_number,
203                Some(&event),
204            ),
205        }))
206    }

208    fn derive_object_versions(
209        object_history_changes: &TransactionObjectChangesToCommit,
210    ) -> IndexerResult<Vec<StoredObjectVersion>> {
211        let mut object_versions = vec![];
212        for changed_obj in object_history_changes.changed_objects.iter() {
213            object_versions.push(changed_obj.try_into()?);
214        }
215        for deleted_obj in object_history_changes.deleted_objects.iter() {
216            object_versions.push(deleted_obj.into());
217        }
218        Ok(object_versions)
219    }

221    async fn index_checkpoint(
222        data: &CheckpointData,
223        metrics: Arc<IndexerMetrics>,
224        packages: Vec<IndexedPackage>,
225    ) -> Result<CheckpointDataToCommit, IndexerError> {
226        let checkpoint_seq = data.checkpoint_summary.sequence_number;
227        info!(checkpoint_seq, "Indexing checkpoint data blob");
228
229        // Index epoch
230        let epoch = Self::index_epoch(data).await?;
231
232        // Index Objects
233        let object_changes = Self::index_checkpoint_objects(data, &metrics).await?;
234        let object_history_changes: TransactionObjectChangesToCommit =
235            Self::index_objects_history(data).await?;
236        let object_versions = Self::derive_object_versions(&object_history_changes)?;
237
238        let (checkpoint, db_transactions, db_events, db_tx_indices, db_event_indices, db_displays) = {
239            let CheckpointData {
240                transactions,
241                checkpoint_summary,
242                checkpoint_contents,
243            } = data;
244
245            let (db_transactions, db_events, db_tx_indices, db_event_indices, db_displays) =
246                Self::index_transactions(
247                    transactions,
248                    checkpoint_summary,
249                    checkpoint_contents,
250                    &metrics,
251                )
252                .await?;
253
254            let successful_tx_num: u64 = db_transactions.iter().map(|t| t.successful_tx_num).sum();
255            (
256                IndexedCheckpoint::from_iota_checkpoint(
257                    checkpoint_summary,
258                    checkpoint_contents,
259                    successful_tx_num as usize,
260                ),
261                db_transactions,
262                db_events,
263                db_tx_indices,
264                db_event_indices,
265                db_displays,
266            )
267        };
268        let time_now_ms = chrono::Utc::now().timestamp_millis();
269        metrics
270            .index_lag_ms
271            .set(time_now_ms - checkpoint.timestamp_ms as i64);
272        metrics
273            .max_indexed_checkpoint_sequence_number
274            .set(checkpoint.sequence_number as i64);
275        metrics
276            .indexed_checkpoint_timestamp_ms
277            .set(checkpoint.timestamp_ms as i64);
278        info!(
279            "Indexer lag: indexed checkpoint {} with time now {} and checkpoint time {}",
280            checkpoint.sequence_number, time_now_ms, checkpoint.timestamp_ms
281        );
282
283        Ok(CheckpointDataToCommit {
284            checkpoint,
285            transactions: db_transactions,
286            events: db_events,
287            tx_indices: db_tx_indices,
288            event_indices: db_event_indices,
289            display_updates: db_displays,
290            object_changes,
291            object_history_changes,
292            object_versions,
293            packages,
294            epoch,
295        })
296    }

298    async fn index_transactions(
299        transactions: &[CheckpointTransaction],
300        checkpoint_summary: &CertifiedCheckpointSummary,
301        checkpoint_contents: &CheckpointContents,
302        metrics: &IndexerMetrics,
303    ) -> IndexerResult<(
304        Vec<IndexedTransaction>,
305        Vec<IndexedEvent>,
306        Vec<TxIndex>,
307        Vec<EventIndex>,
308        BTreeMap<String, StoredDisplay>,
309    )> {
310        let checkpoint_seq = checkpoint_summary.sequence_number();
311
312        let mut tx_seq_num_iter = checkpoint_contents
313            .enumerate_transactions(checkpoint_summary)
314            .map(|(seq, execution_digest)| (execution_digest.transaction, seq));
315
316        if checkpoint_contents.size() != transactions.len() {
317            return Err(IndexerError::FullNodeReading(format!(
318                "checkpointContents has different size {} compared to Transactions {} for checkpoint {checkpoint_seq}",
319                checkpoint_contents.size(),
320                transactions.len()
321            )));
322        }
323
324        let mut db_transactions = Vec::new();
325        let mut db_events = Vec::new();
326        let mut db_displays = BTreeMap::new();
327        let mut db_tx_indices = Vec::new();
328        let mut db_event_indices = Vec::new();
329
330        for tx in transactions {
331            // Unwrap safe - we checked they have equal length above
332            let (tx_digest, tx_sequence_number) = tx_seq_num_iter.next().unwrap();
333            let actual_tx_digest = tx.transaction.digest();
334            if tx_digest != *actual_tx_digest {
335                return Err(IndexerError::FullNodeReading(format!(
336                    "transactions has different ordering from CheckpointContents, for checkpoint {checkpoint_seq}, Mismatch found at {tx_digest} v.s. {actual_tx_digest}",
337                )));
338            }
339
340            let (indexed_tx, tx_indices, indexed_events, events_indices, stored_displays) =
341                Self::index_transaction_components(
342                    tx,
343                    tx_sequence_number,
344                    *checkpoint_seq,
345                    checkpoint_summary.timestamp_ms,
346                    metrics,
347                )
348                .await?;
349            db_transactions.push(indexed_tx);
350            db_tx_indices.push(tx_indices);
351            db_events.extend(indexed_events);
352            db_event_indices.extend(events_indices);
353            db_displays.extend(stored_displays);
354        }
355        Ok((
356            db_transactions,
357            db_events,
358            db_tx_indices,
359            db_event_indices,
360            db_displays,
361        ))
362    }

364    pub(crate) async fn index_transaction_components(
365        tx: &CheckpointTransaction,
366        tx_sequence_number: u64,
367        checkpoint_seq: CheckpointSequenceNumber,
368        checkpoint_timestamp_ms: u64,
369        metrics: &IndexerMetrics,
370    ) -> IndexerResult<IndexedTransactionComponents> {
371        let db_txn = Self::index_transaction(
372            tx,
373            tx_sequence_number,
374            checkpoint_seq,
375            checkpoint_timestamp_ms,
376            metrics,
377        )
378        .await?;
379
380        let CheckpointTransaction {
381            transaction: sender_signed_data,
382            effects: fx,
383            events,
384            ..
385        } = tx;
386
387        let tx_digest = sender_signed_data.digest();
388        let tx = sender_signed_data.transaction_data();
389        let events = events
390            .as_ref()
391            .map(|events| events.data.clone())
392            .unwrap_or_default();
393
394        let transaction_kind = IotaTransactionKind::from(tx.kind());
395
396        let db_events = events
397            .iter()
398            .enumerate()
399            .map(|(idx, event)| {
400                IndexedEvent::from_event(
401                    tx_sequence_number,
402                    idx as u64,
403                    checkpoint_seq,
404                    *tx_digest,
405                    event,
406                    checkpoint_timestamp_ms,
407                )
408            })
409            .collect();
410
411        let db_event_indices = events
412            .iter()
413            .enumerate()
414            .map(|(idx, event)| EventIndex::from_event(tx_sequence_number, idx as u64, event))
415            .collect();
416
417        let db_displays = events
418            .iter()
419            .flat_map(StoredDisplay::try_from_event)
420            .map(|display| (display.object_type.clone(), display))
421            .collect();
422
423        // Input Objects
424        let input_objects = tx
425            .input_objects()
426            .expect("committed txns have been validated")
427            .into_iter()
428            .map(|obj_kind| obj_kind.object_id())
429            .collect::<Vec<_>>();
430
431        // Changed Objects
432        let changed_objects = fx
433            .all_changed_objects()
434            .into_iter()
435            .map(|(object_ref, _owner, _write_kind)| object_ref.0)
436            .collect::<Vec<_>>();
437
438        // Wrapped or deleted objects
439        let wrapped_or_deleted_objects = fx
440            .all_tombstones()
441            .into_iter()
442            .chain(fx.created_then_wrapped_objects())
443            .map(|(object_id, _)| object_id)
444            .collect::<Vec<_>>();
445
446        // Payers
447        let payers = vec![tx.gas_owner()];
448
449        // Sender
450        let sender = tx.sender();
451
452        // Recipients
453        let recipients = fx
454            .all_changed_objects()
455            .into_iter()
456            .filter_map(|(_object_ref, owner, _write_kind)| match owner {
457                Owner::AddressOwner(address) => Some(address),
458                _ => None,
459            })
460            .unique()
461            .collect::<Vec<_>>();
462
463        // Move Calls
464        let move_calls = tx
465            .move_calls()
466            .iter()
467            .map(|(p, m, f)| (*<&ObjectID>::clone(p), m.to_string(), f.to_string()))
468            .collect();
469
470        let db_tx_indices = TxIndex {
471            tx_sequence_number,
472            transaction_digest: *tx_digest,
473            checkpoint_sequence_number: checkpoint_seq,
474            input_objects,
475            changed_objects,
476            sender,
477            payers,
478            recipients,
479            move_calls,
480            tx_kind: transaction_kind,
481            wrapped_or_deleted_objects,
482        };
483
484        Ok((
485            db_txn,
486            db_tx_indices,
487            db_events,
488            db_event_indices,
489            db_displays,
490        ))
491    }

493    /// Creates a new [`IndexedTransaction`]
494    pub(crate) async fn index_transaction(
495        tx: &CheckpointTransaction,
496        tx_sequence_number: u64,
497        checkpoint_seq: CheckpointSequenceNumber,
498        checkpoint_timestamp_ms: u64,
499        metrics: &IndexerMetrics,
500    ) -> IndexerResult<IndexedTransaction> {
501        let tx_digest = tx.transaction.digest();
502        let tx_data = tx.transaction.transaction_data();
503
504        let events = tx
505            .events
506            .as_ref()
507            .map(|events| events.data.clone())
508            .unwrap_or_default();
509
510        let transaction_kind = IotaTransactionKind::from(tx_data.kind());
511
512        let objects = tx
513            .input_objects
514            .iter()
515            .chain(tx.output_objects.iter())
516            .collect::<Vec<_>>();
517
518        let (balance_change, object_changes) = InMemTxChanges::new(&objects, metrics.clone())
519            .get_changes(tx_data, &tx.effects, tx_digest)
520            .await?;
521
522        Ok(IndexedTransaction {
523            tx_sequence_number,
524            tx_digest: *tx_digest,
525            checkpoint_sequence_number: checkpoint_seq,
526            timestamp_ms: checkpoint_timestamp_ms,
527            sender_signed_data: tx.transaction.data().clone(),
528            successful_tx_num: if tx.effects.status().is_ok() {
529                tx_data.kind().tx_count() as u64
530            } else {
531                0
532            },
533            effects: tx.effects.clone(),
534            object_changes,
535            balance_change,
536            events,
537            transaction_kind,
538        })
539    }

549    pub(crate) async fn index_objects(
550        data: &CheckpointData,
551        metrics: &IndexerMetrics,
552    ) -> Result<TransactionObjectChangesToCommit, IndexerError> {
553        let _timer = metrics.indexing_objects_latency.start_timer();
554        let checkpoint_seq = data.checkpoint_summary.sequence_number;
555
556        let eventually_removed_object_refs_post_version =
557            data.eventually_removed_object_refs_post_version();
558        let indexed_eventually_removed_objects = eventually_removed_object_refs_post_version
559            .into_iter()
560            .map(|obj_ref| IndexedDeletedObject {
561                object_id: obj_ref.0,
562                object_version: obj_ref.1.into(),
563                checkpoint_sequence_number: checkpoint_seq,
564            })
565            .collect();
566
567        let latest_live_output_objects = data.latest_live_output_objects();
568        let changed_objects = latest_live_output_objects
569            .into_iter()
570            .map(|o| {
571                let df_kind = extract_df_kind(o);
572                IndexedObject::from_object(Some(checkpoint_seq), o.clone(), df_kind)
573            })
574            .collect::<Vec<_>>();
575        Ok(TransactionObjectChangesToCommit {
576            changed_objects,
577            deleted_objects: indexed_eventually_removed_objects,
578        })
579    }

581    // similar to index_objects, but objects_history keeps all versions of objects
582    async fn index_objects_history(
583        data: &CheckpointData,
584    ) -> Result<TransactionObjectChangesToCommit, IndexerError> {
585        let checkpoint_seq = data.checkpoint_summary.sequence_number;
586        let deleted_objects = data
587            .transactions
588            .iter()
589            .flat_map(|tx| tx.removed_object_refs_post_version())
590            .collect::<Vec<_>>();
591        let indexed_deleted_objects: Vec<IndexedDeletedObject> = deleted_objects
592            .into_iter()
593            .map(|obj_ref| IndexedDeletedObject {
594                object_id: obj_ref.0,
595                object_version: obj_ref.1.into(),
596                checkpoint_sequence_number: checkpoint_seq,
597            })
598            .collect();
599
600        let output_objects: Vec<_> = data
601            .transactions
602            .iter()
603            .flat_map(|tx| &tx.output_objects)
604            .collect();
605        // TODO(gegaowp): the current df_info implementation is not correct,
606        // but we have decided remove all df_* except df_kind.
607        let changed_objects = output_objects
608            .into_iter()
609            .map(|o| {
610                let df_kind = extract_df_kind(o);
611                IndexedObject::from_object(Some(checkpoint_seq), o.clone(), df_kind)
612            })
613            .collect::<Vec<_>>();
614
615        Ok(TransactionObjectChangesToCommit {
616            changed_objects,
617            deleted_objects: indexed_deleted_objects,
618        })
619    }

621    fn index_packages(
622        checkpoint_data: &[CheckpointData],
623        metrics: &IndexerMetrics,
624    ) -> Vec<IndexedPackage> {
625        let _timer = metrics.indexing_packages_latency.start_timer();
626        checkpoint_data
627            .iter()
628            .flat_map(|data| {
629                let checkpoint_sequence_number = data.checkpoint_summary.sequence_number;
630                data.transactions
631                    .iter()
632                    .flat_map(|tx| &tx.output_objects)
633                    .filter_map(|o| {
634                        if let iota_types::object::Data::Package(p) = &o.data {
635                            Some(IndexedPackage {
636                                package_id: o.id(),
637                                move_package: p.clone(),
638                                checkpoint_sequence_number,
639                            })
640                        } else {
641                            None
642                        }
643                    })
644                    .collect::<Vec<_>>()
645            })
646            .collect()
647    }

649    pub(crate) fn pg_blocking_cp(state: PgIndexerStore) -> Result<ConnectionPool, IndexerError> {
650        let state_as_any = state.as_any();
651        if let Some(pg_state) = state_as_any.downcast_ref::<PgIndexerStore>() {

653        }
654        Err(IndexerError::Uncategorized(anyhow::anyhow!(
655            "failed to downcast state to PgIndexerStore"
656        )))

660pub struct InMemObjectCache {
661    id_map: HashMap<ObjectID, Object>,
662    seq_map: HashMap<(ObjectID, SequenceNumber), Object>,
663}
664
665impl InMemObjectCache {
666    pub fn new() -> Self {
667        Self {
668            id_map: HashMap::new(),
669            seq_map: HashMap::new(),
670        }
671    }
672
673    pub fn insert_object(&mut self, obj: Object) {
674        self.id_map.insert(obj.id(), obj.clone());
675        self.seq_map.insert((obj.id(), obj.version()), obj);
676    }
677
678    pub fn get(&self, id: &ObjectID, version: Option<&SequenceNumber>) -> Option<&Object> {
679        if let Some(version) = version {
680            self.seq_map.get(&(*id, *version))
681        } else {
682            self.id_map.get(id)
683        }
684    }
685}
686
687impl Default for InMemObjectCache {
688    fn default() -> Self {
689        Self::new()
690    }
691}

710        }
711    }
712
713    pub(crate) async fn get_changes(
714        &self,
715        tx: &TransactionData,
716        effects: &TransactionEffects,
717        tx_digest: &TransactionDigest,
718    ) -> IndexerResult<(
719        Vec<iota_json_rpc_types::BalanceChange>,
720        Vec<IndexedObjectChange>,
721    )> {
722        let _timer = self
723            .metrics
724            .indexing_tx_object_changes_latency
725            .start_timer();
726        let object_change: Vec<_> = get_object_changes(
727            self,
728            tx.sender(),
729            effects.modified_at_versions(),
730            effects.all_changed_objects(),
731            effects.all_removed_objects(),
732        )
733        .await?
734        .into_iter()
735        .map(IndexedObjectChange::from)
736        .collect();
737        let balance_change = get_balance_changes_from_effect(
738            self,
739            effects,
740            tx.input_objects().unwrap_or_else(|e| {
741                panic!("checkpointed tx {tx_digest:?} has invalid input objects: {e}")
742            }),
743            None,
744        )
745        .await?;
746        Ok((balance_change, object_change))
747    }
748}

750#[async_trait]
751impl ObjectProvider for InMemTxChanges {
752    type Error = IndexerError;
753
754    async fn get_object(
755        &self,
756        id: &ObjectID,
757        version: &SequenceNumber,
758    ) -> Result<Object, Self::Error> {
759        let object = self
760            .object_cache
761            .get(id, Some(version))
762            .as_ref()
763            .map(|o| <&Object>::clone(o).clone());
764        if let Some(o) = object {
765            self.metrics.indexing_get_object_in_mem_hit.inc();
766            return Ok(o);
767        }
768
769        panic!(
770            "object {id} is not found in TxChangesProcessor as an ObjectProvider (fn get_object)"
771        );
772    }
773
774    async fn find_object_lt_or_eq_version(
775        &self,
776        id: &ObjectID,
777        version: &SequenceNumber,
778    ) -> Result<Option<Object>, Self::Error> {
779        // First look up the exact version in object_cache.
780        let object = self
781            .object_cache
782            .get(id, Some(version))
783            .as_ref()
784            .map(|o| <&Object>::clone(o).clone());
785        if let Some(o) = object {
786            self.metrics.indexing_get_object_in_mem_hit.inc();
787            return Ok(Some(o));
788        }
789
790        // Second look up the latest version in object_cache. This may be
791        // called when the object is deleted hence the version at deletion
792        // is given.
793        let object = self
794            .object_cache
795            .get(id, None)
796            .as_ref()
797            .map(|o| <&Object>::clone(o).clone());
798        if let Some(o) = object {
799            if o.version() > *version {
800                panic!(
801                    "found a higher version {} for object {id}, expected lt_or_eq {version}",
802                    o.version(),
803                );
804            }
805            if o.version() <= *version {
806                self.metrics.indexing_get_object_in_mem_hit.inc();
807                return Ok(Some(o));
808            }
809        }
810
811        panic!(
812            "object {id} is not found in TxChangesProcessor as an ObjectProvider (fn find_object_lt_or_eq_version)"
813        );
814    }
815}
