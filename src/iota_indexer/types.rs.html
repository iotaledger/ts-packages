49    pub checkpoint_commitments: Vec<CheckpointCommitment>,
50    pub validator_signature: AggregateAuthoritySignature,
51    // Note: not used in StoredCheckpoint conversion and in code overall.
52    pub successful_tx_num: usize,
53    pub end_of_epoch_data: Option<EndOfEpochData>,
54    pub end_of_epoch: bool,
55    pub min_tx_sequence_number: u64,
56    pub max_tx_sequence_number: u64,

59impl IndexedCheckpoint {
60    pub fn from_iota_checkpoint(
61        checkpoint: &iota_types::messages_checkpoint::CertifiedCheckpointSummary,
62        contents: &iota_types::messages_checkpoint::CheckpointContents,
63        successful_tx_num: usize,
64    ) -> Self {
65        let total_gas_cost = checkpoint.epoch_rolling_gas_cost_summary.computation_cost as i64
66            + checkpoint.epoch_rolling_gas_cost_summary.storage_cost as i64
67            - checkpoint.epoch_rolling_gas_cost_summary.storage_rebate as i64;
68        let tx_digests = contents.iter().map(|t| t.transaction).collect::<Vec<_>>();
69        let max_tx_sequence_number = checkpoint.network_total_transactions - 1;
70        // NOTE: + 1u64 first to avoid subtraction with overflow
71        let min_tx_sequence_number = max_tx_sequence_number + 1u64 - tx_digests.len() as u64;
72        let auth_sig = &checkpoint.auth_sig().signature;
73        Self {
74            sequence_number: checkpoint.sequence_number,
75            checkpoint_digest: *checkpoint.digest(),
76            epoch: checkpoint.epoch,
77            tx_digests,
78            previous_checkpoint_digest: checkpoint.previous_digest,
79            end_of_epoch_data: checkpoint.end_of_epoch_data.clone(),
80            end_of_epoch: checkpoint.end_of_epoch_data.clone().is_some(),
81            total_gas_cost,
82            computation_cost: checkpoint.epoch_rolling_gas_cost_summary.computation_cost,
83            computation_cost_burned: checkpoint
84                .epoch_rolling_gas_cost_summary
85                .computation_cost_burned,
86            storage_cost: checkpoint.epoch_rolling_gas_cost_summary.storage_cost,
87            storage_rebate: checkpoint.epoch_rolling_gas_cost_summary.storage_rebate,
88            non_refundable_storage_fee: checkpoint
89                .epoch_rolling_gas_cost_summary
90                .non_refundable_storage_fee,
91            successful_tx_num,
92            network_total_transactions: checkpoint.network_total_transactions,
93            timestamp_ms: checkpoint.timestamp_ms,
94            validator_signature: auth_sig.clone(),
95            checkpoint_commitments: checkpoint.checkpoint_commitments.clone(),
96            min_tx_sequence_number,
97            max_tx_sequence_number,
98        }
99    }
100}

269pub enum ObjectStatus {
270    Active = 0,
271    WrappedOrDeleted = 1,
272}
273
274impl TryFrom<i16> for ObjectStatus {
275    type Error = IndexerError;
276
277    fn try_from(value: i16) -> Result<Self, Self::Error> {
278        Ok(match value {
279            0 => ObjectStatus::Active,
280            1 => ObjectStatus::WrappedOrDeleted,
281            value => {
282                return Err(IndexerError::PersistentStorageDataCorruption(format!(
283                    "{value} as ObjectStatus"
284                )));
285            }
286        })
287    }
288}
289
290impl TryFrom<i16> for OwnerType {
291    type Error = IndexerError;
292
293    fn try_from(value: i16) -> Result<Self, Self::Error> {
294        Ok(match value {
295            0 => OwnerType::Immutable,
296            1 => OwnerType::Address,
297            2 => OwnerType::Object,
298            3 => OwnerType::Shared,
299            value => {
300                return Err(IndexerError::PersistentStorageDataCorruption(format!(
301                    "{value} as OwnerType"
302                )));
303            }
304        })
305    }
306}

308// Returns owner_type, owner_address
309pub fn owner_to_owner_info(owner: &Owner) -> (OwnerType, Option<IotaAddress>) {
310    match owner {
311        Owner::AddressOwner(address) => (OwnerType::Address, Some(*address)),
312        Owner::ObjectOwner(address) => (OwnerType::Object, Some(*address)),
313        Owner::Shared { .. } => (OwnerType::Shared, None),
314        Owner::Immutable => (OwnerType::Immutable, None),
315    }

763            transaction: options.show_input.then_some(response.transaction).flatten(),
764            raw_transaction: if options.show_raw_input {
765                response.raw_transaction
766            } else {
767                Default::default()
768            },
769            effects: options.show_effects.then_some(response.effects).flatten(),
770            events: options.show_events.then_some(response.events).flatten(),
771            object_changes: options
772                .show_object_changes
773                .then_some(response.object_changes)
774                .flatten(),
775            balance_changes: options
776                .show_balance_changes
777                .then_some(response.balance_changes)
778                .flatten(),
779            timestamp_ms: response.timestamp_ms,
780            confirmed_local_execution: response.confirmed_local_execution,
781            checkpoint: response.checkpoint,
782            errors: response.errors,
783            raw_effects: if options.show_raw_effects {
784                response.raw_effects
785            } else {
786                Default::default()
787            },
788        }
789    }
790}

792/// Provides conversion methods from gRPC types to iota core types.
793pub(crate) mod grpc_conversion {
794
795    use iota_grpc_types::v1::{
796        command::{CommandOutputs as GrpcCommandOutputs, CommandResults as GrpcCommandResults},
797        object::Objects as GrpcObjects,
798    };
799    use iota_json_rpc_types::{IotaArgument, IotaExecutionResult, IotaTypeTag};
800    use iota_types::{
801        iota_sdk_types_conversions::type_tag_sdk_to_core, object::Object, transaction::Argument,
802    };
803
804    use crate::types::IndexerResult;
805
806    /// Converts [`GrpcObjects`] into [`Vec<Object>`]
807    pub(crate) fn objects(objects: &GrpcObjects) -> IndexerResult<Vec<Object>> {
808        objects
809            .objects
810            .iter()
811            .map(|o| -> IndexerResult<_> { Ok(Object::try_from(o.object()?)?) })
812            .collect()
813    }
814
815    fn convert_command_outputs_into_mutated_by_ref(
816        command_outputs: GrpcCommandOutputs,
817    ) -> IndexerResult<Vec<(IotaArgument, Vec<u8>, IotaTypeTag)>> {
818        command_outputs
819            .outputs
820            .into_iter()
821            .map(|command_output| -> IndexerResult<_> {
822                Ok((
823                    IotaArgument::from(Argument::from(command_output.argument()?)),
824                    command_output.output_bcs()?.to_vec(),
825                    type_tag_sdk_to_core(&command_output.type_tag()?)?.into(),
826                ))
827            })
828            .collect()
829    }
830
831    fn convert_command_outputs_into_return_values(
832        command_outputs: GrpcCommandOutputs,
833    ) -> IndexerResult<Vec<(Vec<u8>, IotaTypeTag)>> {
834        command_outputs
835            .outputs
836            .into_iter()
837            .map(|command_output| -> IndexerResult<_> {
838                Ok((
839                    command_output.output_bcs()?.to_vec(),
840                    type_tag_sdk_to_core(&command_output.type_tag()?)?.into(),
841                ))
842            })
843            .collect()
844    }
845
846    /// Converts [`GrpcCommandResults`] into [`IotaExecutionResult`]
847    pub(crate) fn command_results(
848        command_results: GrpcCommandResults,
849    ) -> IndexerResult<Vec<IotaExecutionResult>> {
850        command_results
851            .results
852            .into_iter()
853            .map(|command_result| -> IndexerResult<_> {
854                let mutable_reference_outputs = command_result
855                    .mutated_by_ref()
856                    .map_err(Into::into)
857                    .and_then(|c| convert_command_outputs_into_mutated_by_ref(c.clone()))?;
858                let return_values = command_result
859                    .return_values()
860                    .map_err(Into::into)
861                    .and_then(|c| convert_command_outputs_into_return_values(c.clone()))?;
862
863                Ok(IotaExecutionResult {
864                    mutable_reference_outputs,
865                    return_values,
866                })
867            })
868            .collect()
869    }
