4use std::collections::{HashMap, HashSet};
5
6use prost_types::{DescriptorProto, field_descriptor_proto::Type};
7
8// Dependency graph for tracking message dependencies
9#[derive(Debug, Default)]
10pub(crate) struct DependencyGraph {
11    // Maps message name to set of messages it depends on
12    dependencies: HashMap<String, HashSet<String>>,
13    // Maps message name to its full qualified name
14    full_names: HashMap<String, String>,
15}
16
17impl DependencyGraph {
18    pub(crate) fn new() -> Self {
19        Self::default()
20    }
21
22    pub(crate) fn add_dependency(&mut self, from_message: &str, to_message: &str) {
23        self.dependencies
24            .entry(from_message.to_string())
25            .or_default()
26            .insert(to_message.to_string());
27    }
28
29    pub(crate) fn set_full_name(&mut self, message: &str, full_name: &str) {
30        self.full_names
31            .insert(message.to_string(), full_name.to_string());
32    }
33
34    // Check if there's a circular dependency between two messages
35    pub(crate) fn has_circular_dependency(&self, from_message: &str, to_message: &str) -> bool {
36        // If to_message can reach back to from_message, we have a cycle
37        self.can_reach(to_message, from_message, &mut HashSet::new())
38    }
39
40    // Check if we can reach target from source (used for cycle detection)
41    fn can_reach(&self, source: &str, target: &str, visited: &mut HashSet<String>) -> bool {
42        if source == target {
43            return true;
44        }
45
46        if visited.contains(source) {
47            return false; // Already visited this node
48        }
49
50        visited.insert(source.to_string());
51
52        if let Some(deps) = self.dependencies.get(source) {
53            for dep in deps {
54                if self.can_reach(dep, target, visited) {
55                    return true;
56                }
57            }
58        }
59
60        false
61    }
62}
63
64// Helper function to build dependency graph for all messages
65pub(crate) fn build_dependency_graph(
66    messages: &[DescriptorProto],
67    package: &str,
68    prefix: &str,
69) -> DependencyGraph {
70    let mut graph = DependencyGraph::new();
71
72    for message in messages {
73        let full_name = if prefix.is_empty() {
74            format!("{}.{}", package, message.name())
75        } else {
76            format!("{}.{}", prefix, message.name())
77        };
78
79        graph.set_full_name(message.name(), &full_name);
80
81        // Add dependencies for each message field
82        for field in &message.field {
83            if matches!(field.r#type(), Type::Message) {
84                let field_message_name = field.type_name().split('.').next_back().unwrap();
85                graph.add_dependency(message.name(), field_message_name);
86            }
87        }
88
89        // Recursively handle nested messages
90        let nested_graph = build_dependency_graph(&message.nested_type, package, &full_name);
91
92        // Merge nested graph dependencies
93        graph
94            .dependencies
95            .extend(nested_graph.dependencies.into_iter());
96
97        //// Merge nested graph full names
98        graph.full_names.extend(nested_graph.full_names);
99    }
100
101    graph
102}
