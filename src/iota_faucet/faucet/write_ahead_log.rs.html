16/// Persistent log of transactions paying out iota from the faucet, keyed by the
17/// coin serving the request.  Transactions are expected to be written to the
18/// log before they are sent to full-node, and removed after receiving a
19/// response back, before the coin becomes available for subsequent writes.
20///
21/// This allows the faucet to go down and back up, and not forget which requests
22/// were in-flight that it needs to confirm succeeded or failed.
23#[derive(DBMapUtils, Clone)]
24pub struct WriteAheadLog {
25    pub log: DBMap<ObjectID, Entry>,
26}
27
28#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone)]
29pub struct Entry {
30    pub uuid: uuid::Bytes,
31    // TODO (jian): remove recipient
32    pub recipient: IotaAddress,
33    pub tx: TransactionData,
34    pub retry_count: u64,
35    pub in_flight: bool,
36}
37
38impl WriteAheadLog {
39    pub(crate) fn open(path: &Path) -> Self {
40        Self::open_tables_read_write(
41            path.to_path_buf(),
42            typed_store::rocks::MetricConf::new("faucet_write_ahead_log"),
43            None,
44            None,
45        )
46    }
47
48    /// Mark `coin` as reserved for transaction `tx` sending coin to
49    /// `recipient`. Fails if `coin` is already in the WAL pointing to an
50    /// existing transaction.
51    pub(crate) fn reserve(
52        &mut self,
53        uuid: Uuid,
54        coin: ObjectID,
55        recipient: IotaAddress,
56        tx: TransactionData,
57    ) -> Result<(), TypedStoreError> {
58        if self.log.contains_key(&coin)? {
59            // Don't permit multiple writes against the same coin
60            // TODO: Use a better error type than `TypedStoreError`.
61            return Err(TypedStoreError::Serialization(format!(
62                "Duplicate WAL entry for coin {coin:?}",
63            )));
64        }
65
66        let uuid = *uuid.as_bytes();
67        self.log.insert(
68            &coin,
69            &Entry {
70                uuid,
71                recipient,
72                tx,
73                retry_count: 0,
74                in_flight: true,
75            },
76        )
77    }
78
79    /// Check whether `coin` has a pending transaction in the WAL.  Returns
80    /// `Ok(Some(entry))` if a pending transaction exists, `Ok(None)` if
81    /// not, and `Err(_)` if there was an internal error accessing the WAL.
82    pub(crate) fn reclaim(&self, coin: ObjectID) -> Result<Option<Entry>, TypedStoreError> {
83        match self.log.get(&coin) {
84            Ok(entry) => Ok(entry),
85            Err(TypedStoreError::Serialization(_)) => {
86                // Remove bad log from the store, so we don't crash on start up, this can happen
87                // if we update the WAL Entry and have some leftover Entry from
88                // the WAL.
89                self.log
90                    .remove(&coin)
91                    .unwrap_or_else(|_| panic!("coin: {coin:?} unable to be removed from log."));
92                Ok(None)
93            }
94            Err(err) => Err(err),
95        }
96    }
97
98    /// Indicate that the transaction in flight for `coin` has landed, and the
99    /// entry in the WAL can be removed.
100    pub(crate) fn commit(&mut self, coin: ObjectID) -> Result<(), TypedStoreError> {
101        self.log.remove(&coin)
102    }
103
104    pub(crate) fn increment_retry_count(&mut self, coin: ObjectID) -> Result<(), TypedStoreError> {
105        if let Some(mut entry) = self.log.get(&coin)? {
106            entry.retry_count += 1;
107            self.log.insert(&coin, &entry)?;
108        }
109        Ok(())
110    }
111
112    pub(crate) fn set_in_flight(
113        &mut self,
114        coin: ObjectID,
115        bool: bool,
116    ) -> Result<(), TypedStoreError> {
117        if let Some(mut entry) = self.log.get(&coin)? {
118            entry.in_flight = bool;
119            self.log.insert(&coin, &entry)?;
120        } else {
121            info!(
122                ?coin,
123                "Attempted to set inflight a coin that was not in the WAL."
124            );
125
126            return Err(TypedStoreError::RocksDB(format!(
127                "Coin object {coin:?} not found in WAL."
128            )));
129        }
130        Ok(())
131    }
132}

143    #[tokio::test]
144    async fn reserve_reclaim_reclaim() {
145        let tmp = tempfile::tempdir().unwrap();
146        let mut wal = WriteAheadLog::open(&tmp.path().join("wal"));
147
148        let uuid = Uuid::new_v4();
149        let coin = random_object_ref();
150        let (recv, tx) = random_request(coin);
151
152        assert!(wal.reserve(uuid, coin.0, recv, tx.clone()).is_ok());
153
154        // Reclaim once
155        let Some(entry) = wal.reclaim(coin.0).unwrap() else {
156            panic!("entry not found for {}", coin.0);
157        };
158
159        assert_eq!(uuid, Uuid::from_bytes(entry.uuid));
160        assert_eq!(recv, entry.recipient);
161        assert_eq!(tx, entry.tx);
162
163        // Reclaim again, should still be there.
164        let Some(entry) = wal.reclaim(coin.0).unwrap() else {
165            panic!("entry not found for {}", coin.0);
166        };
167
168        assert_eq!(uuid, Uuid::from_bytes(entry.uuid));
169        assert_eq!(recv, entry.recipient);
170        assert_eq!(tx, entry.tx);
171    }

175        let tmp = tempfile::tempdir().unwrap();
176        let mut wal = WriteAheadLog::open(&tmp.path().join("wal"));
177        let uuid = Uuid::new_v4();
178        let coin = random_object_ref();

181        // First write goes through
182        wal.reserve(uuid, coin.0, recv0, tx0).unwrap();
183        wal.increment_retry_count(coin.0).unwrap();
184
185        let entry = wal.reclaim(coin.0).unwrap().unwrap();
186        assert_eq!(entry.retry_count, 1);

209    #[tokio::test]
210    async fn reserve_reclaim_commit_reclaim() {
211        let tmp = tempfile::tempdir().unwrap();
212        let mut wal = WriteAheadLog::open(&tmp.path().join("wal"));
213
214        let uuid = Uuid::new_v4();
215        let coin = random_object_ref();
216        let (recv, tx) = random_request(coin);
217
218        wal.reserve(uuid, coin.0, recv, tx.clone()).unwrap();
219
220        // Reclaim to show that the entry is there
221        let Some(entry) = wal.reclaim(coin.0).unwrap() else {
222            panic!("entry not found for {}", coin.0);
223        };
224
225        assert_eq!(uuid, Uuid::from_bytes(entry.uuid));
226        assert_eq!(recv, entry.recipient);
227        assert_eq!(tx, entry.tx);
228
229        // Commit the transaction, which removes it from the log.
230        wal.commit(coin.0).unwrap();
231
232        // Expect it to now be gone
233        assert_eq!(Ok(None), wal.reclaim(coin.0));
234    }

256    fn random_request(coin: ObjectRef) -> (IotaAddress, TransactionData) {
257        let gas_price = 1;
258        let send = IotaAddress::random_for_testing_only();
259        let recv = IotaAddress::random_for_testing_only();
260        (
261            recv,
262            TransactionData::new_pay_iota(
263                send,
264                vec![coin],
265                vec![recv],
266                vec![1000],
267                coin,
268                gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER,
269                gas_price,
270            )
271            .unwrap(),
272        )
273    }
