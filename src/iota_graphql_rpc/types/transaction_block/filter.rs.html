7use async_graphql::InputObject;
8use iota_types::base_types::IotaAddress as NativeIotaAddress;
9
10use crate::types::{
11    digest::Digest, intersect, iota_address::IotaAddress,
12    transaction_block::TransactionBlockKindInput, type_filter::FqNameFilter, uint53::UInt53,

15/// Represents optional available filters for transaction blocks.
16#[derive(InputObject, Debug, Default, Clone)]
17pub(crate) struct TransactionBlockFilter {
18    /// Filter transactions by move function called.
19    ///
20    /// Calls can be filtered by the `package`, `package::module`, or the
21    /// `package::module::name` of their function.
22    pub function: Option<FqNameFilter>,
23
24    /// An input filter selecting for either system or programmable
25    /// transactions.
26    pub kind: Option<TransactionBlockKindInput>,
27    /// Limit to transactions that occurred strictly after the given checkpoint.
28    pub after_checkpoint: Option<UInt53>,
29    /// Limit to transactions in the given checkpoint.
30    pub at_checkpoint: Option<UInt53>,
31    /// Limit to transaction that occurred strictly before the given checkpoint.
32    pub before_checkpoint: Option<UInt53>,
33    /// Limit to transactions that were sent by the given address. NOTE: this
34    /// input filter has been deprecated in favor of `sentAddress` which has
35    /// clearer semantics. Both filters restrict transactions by their sender,
36    /// only, not signers in general.
37    ///
38    /// This filter will be removed after 6 months with the 1.24.0 release.
39    #[graphql(
40        deprecation = "Misleading semantics. Use `sentAddress` instead. This will be removed with the 1.24.0 release."
41    )]
42    pub sign_address: Option<IotaAddress>,
43    /// Limit to transactions that were sent by the given address.
44    pub sent_address: Option<IotaAddress>,
45    /// Limit to transactions that sent an object to the given address.
46    pub recv_address: Option<IotaAddress>,
47    /// Limit to transactions that accepted the given object as an input.
48    pub input_object: Option<IotaAddress>,
49    /// Limit to transactions that output a version of this object.
50    pub changed_object: Option<IotaAddress>,
51    /// Limit to transactions that wrapped or deleted the given object.
52    pub wrapped_or_deleted_object: Option<IotaAddress>,
53    /// Select transactions by their digest.
54    pub transaction_ids: Option<Vec<Digest>>,
55}
56
57impl TransactionBlockFilter {
58    /// Try to create a filter whose results are the intersection of transaction
59    /// blocks in `self`'s results and transaction blocks in `other`'s
60    /// results. This may not be possible if the resulting filter is
61    /// inconsistent in some way (e.g. a filter that requires one field to be
62    /// two different values simultaneously).
63    pub(crate) fn intersect(self, other: Self) -> Option<Self> {
64        macro_rules! intersect {
65            ($field:ident, $body:expr) => {
66                intersect::field(self.$field, other.$field, $body)
67            };
68        }
69
70        Some(Self {
71            function: intersect!(function, FqNameFilter::intersect)?,
72            kind: intersect!(kind, intersect::by_eq)?,
73
74            after_checkpoint: intersect!(after_checkpoint, intersect::by_max)?,
75            at_checkpoint: intersect!(at_checkpoint, intersect::by_eq)?,
76            before_checkpoint: intersect!(before_checkpoint, intersect::by_min)?,
77
78            sign_address: intersect!(sign_address, intersect::by_eq)?,
79            sent_address: intersect!(sent_address, intersect::by_eq)?,
80            recv_address: intersect!(recv_address, intersect::by_eq)?,
81            input_object: intersect!(input_object, intersect::by_eq)?,
82            changed_object: intersect!(changed_object, intersect::by_eq)?,
83            wrapped_or_deleted_object: intersect!(wrapped_or_deleted_object, intersect::by_eq)?,
84
85            transaction_ids: intersect!(transaction_ids, |a, b| {
86                let a = BTreeSet::from_iter(a.into_iter());
87                let b = BTreeSet::from_iter(b.into_iter());
88                Some(a.intersection(&b).cloned().collect())
89            })?,
90        })
91    }
92
93    /// Most filter conditions require a scan limit if used in tandem with other
94    /// filters. The exception to this is sender and checkpoint, since
95    /// sender is denormalized on all tables, and the corresponding tx range
96    /// can be determined for a checkpoint.
97    pub(crate) fn requires_scan_limit(&self) -> bool {
98        [
99            self.function.is_some(),
100            self.kind.is_some(),
101            self.recv_address.is_some(),
102            self.input_object.is_some(),
103            self.changed_object.is_some(),
104            self.wrapped_or_deleted_object.is_some(),
105            self.transaction_ids.is_some(),
106        ]
107        .into_iter()
108        .filter(|is_set| *is_set)
109        .count()
110            > 1
111    }
112
113    /// Returns the transaction sender to query `tx_sender`.
114    ///
115    /// If there are other filters set that would query tables with a `sender`
116    /// column, then this returns `None`.
117    pub(crate) fn explicit_sender(&self) -> Option<IotaAddress> {
118        if self.function.is_none()
119            && self.kind.is_none()
120            && self.recv_address.is_none()
121            && self.input_object.is_none()
122            && self.changed_object.is_none()
123            && self.wrapped_or_deleted_object.is_none()
124        {
125            self.sent_address.or(self.sign_address)
126        } else {
127            None
128        }
129    }
130
131    /// A TransactionBlockFilter is considered not to have any filters if no
132    /// filters are specified, or if the only filters are on `checkpoint`.
133    pub(crate) fn has_filters(&self) -> bool {
134        self.function.is_some()
135            || self.kind.is_some()
136            || self.sign_address.is_some()
137            || self.sent_address.is_some()
138            || self.recv_address.is_some()
139            || self.input_object.is_some()
140            || self.changed_object.is_some()
141            || self.wrapped_or_deleted_object.is_some()
142            || self.transaction_ids.is_some()
143    }
144
145    pub(crate) fn is_empty(&self) -> bool {
146        self.before_checkpoint == Some(UInt53::from(0))
147            || matches!(
148                (self.after_checkpoint, self.before_checkpoint),
149                (Some(after), Some(before)) if after >= before
150            )
151            || matches!(
152                (self.after_checkpoint, self.at_checkpoint),
153                (Some(after), Some(at)) if after >= at
154            )
155            || matches!(
156                (self.at_checkpoint, self.before_checkpoint),
157                (Some(at), Some(before)) if at >= before
158            )
159            // If SystemTx, sender if specified must be 0x0. Conversely, if sender is 0x0, kind must be SystemTx.
160            || matches!(
161                (self.kind, self.sent_address.or(self.sign_address)),
162                (Some(kind), Some(signer))
163                    if (kind == TransactionBlockKindInput::SystemTx)
164                        != (signer == IotaAddress::from(NativeIotaAddress::ZERO))
165            )
166            // Temporary while we deprecate `sign_address` in favor of `sent_address`.
167            || matches!(
168                (self.sign_address, self.sent_address),
169                (Some(signer), Some(sent))
170                    if signer != sent
171            )
172    }
173}
